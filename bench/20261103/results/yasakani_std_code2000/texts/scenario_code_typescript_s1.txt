# `metrics-service` — a complete, runnable TypeScript 5 service

Below is a full, self-contained implementation. It uses **only Node built-ins** (no third-party runtime dependencies) for the service, and **only `typescript` + `@types/node`** as dev dependencies. Tests use Node's built-in `node:test` runner.

## 4. `src/types.ts` — strict domain types

```ts
/**
 * Core domain types for the metrics service.
 *
 * Design notes:
 *  - `RunRecord` is the *validated* shape. Raw input is `unknown` until it
 *    passes `validateRunRecord`, which narrows it to `RunRecord`.
 *  - `GroupKey` is a canonical string so grouping is O(1) and stable.
 *  - `Aggregation` holds the running statistics for one (model, fixture) group.
 */

/** A single validated request-metrics record. */
export interface RunRecord {
  /** Unique run identifier (e.g. a UUID or a short slug). */
  id: string;
  /** Model name (e.g. "gpt-4", "llama-3-70b"). */
  model: string;
  /** Fixture / dataset name (e.g. "mmlu", "gsm8k"). */
  fixture: string;
  /** Wall-clock timestamp of the run, in milliseconds since epoch. */
  timestamp: number;
  /** Latency in milliseconds (non-negative). */
  latencyMs: number;
  /** Tokens processed (non-negative integer). */
  tokens: number;
  /** Outcome of the run. */
  outcome: RunOutcome;
  /** Optional free-form error message when outcome === "error". */
  error?: string;
}

export type RunOutcome = "success" | "error" | "timeout";

/** Canonical grouping key: `${model}::${fixture}`. */
export type GroupKey = string;

/**
 * Running aggregation for one (model, fixture) group.
 *
 * `sum` and `sumSq` are kept so the sample variance can be computed in
 * O(1) at query time. `values` is a bounded ring buffer used to compute the
 * median without storing every value ever seen (bounded memory).
 */
export interface Aggregation {
  key: GroupKey;
  model: string;
  fixture: string;
  count: number;
  sum: number;
  sumSq: number;
  min: number;
  max: number;
  /** Bounded sample of latency values (most recent `capacity` values). */
  values: number[];
  /** Number of values evicted from the ring buffer (for observability). */
  evicted: number;
  successCount: number;
  errorCount: number;
  timeoutCount: number;
  firstSeen: number;
  lastSeen: number;
}

/** Per-run metadata tracked for the /runs/:id endpoint. */
export interface RunMeta {
  id: string;
  model: string;
  fixture: string;
  timestamp: number;
  outcome: RunOutcome;
  latencyMs: number;
  tokens: number;
  /** Number of records observed for this run id (usually 1). */
  recordCount: number;
}

/** A fully computed summary row for one group. */
export interface GroupSummary {
  model: string;
  fixture: string;
  count: number;
  mean: number;
  median: number;
  /** Sample standard deviation (n-1 denominator). NaN when count < 2. */
  sampleStdDev: number;
  min: number;
  max: number;
  successCount: number;
  errorCount: number;
  timeoutCount: number;
  firstSeen: number;
  lastSeen: number;
}

/** The full /summary response body. */
export interface SummaryResponse {
  generatedAt: number;
  totalRecords: number;
  totalRuns: number;
  groups: GroupSummary[];
}

/** The /health response body. */
export interface HealthResponse {
  status: "ok" | "degraded";
  uptimeSec: number;
  totalRecords: number;
  totalRuns: number;
  groupCount: number;
  memory: {
    rssBytes: number;
    heapUsedBytes: number;
    heapTotalBytes: number;
  };
  version: string;
}

/** The /runs/:id response body. */
export interface RunResponse {
  id: string;
  model: string;
  fixture: string;
  timestamp: number;
  outcome: RunOutcome;
  latencyMs: number;
  tokens: number;
  recordCount: number;
}

/** Error thrown for malformed input. Carries a stable `code`. */
export class MalformedInputError extends Error {
  readonly code: string;
  readonly line?: number;
  constructor(message: string, code: string, line?: number) {
    super(message);
    this.name = "MalformedInputError";
    this.code = code;
    this.line = line;
  }
}

/** Error thrown when the in-memory store is full and a new run id arrives. */
export class StoreFullError extends Error {
  readonly code = "STORE_FULL";
  constructor(message: string) {
    super(message);
    this.name = "StoreFullError";
  }
}
```
