# CUDA 13.1 / C++20 — Row-wise BF16 Matrix Normalization on sm_120a

Below is a complete, self-contained program. It:

- Parses CLI arguments (matrix size, seed, warm-up iterations, timing iterations, tolerance).
- Generates a **deterministic** BF16 matrix on the host (LCG → FP32 → BF16).
- Uses **RAII** device buffers (`DeviceBuffer<T>`).
- Runs a **vectorized** CUDA kernel that normalizes each row in-place (divide by row max-abs), using `__nv_bfloat16` intrinsics and 128-bit vector loads/stores.
- Computes a **numerically stable FP32 reference** on the host (max-abs scaling, then normalize).
- Performs **warm-up**, **CUDA-event timing**, and **tolerance-based verification** (max absolute error + relative error).
- Includes a `CMakeLists.txt` targeting `sm_120a`.

template <typename T>
class DeviceBuffer {
public:
    DeviceBuffer() = default;

    explicit DeviceBuffer(std::size_t count) {
        allocate(count);
    }

    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    DeviceBuffer(DeviceBuffer&& other) noexcept
        : ptr_(other.ptr_), count_(other.count_) {
        other.ptr_ = nullptr;
        other.count_ = 0;
    }

    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            free();
            ptr_ = other.ptr_;
            count_ = other.count_;
            other.ptr_ = nullptr;
            other.count_ = 0;
        }
        return *this;
    }

    ~DeviceBuffer() { free(); }

    T* data() noexcept { return ptr_; }
    const T* data() const noexcept { return ptr_; }
    std::size_t count() const noexcept { return count_; }
    std::size_t bytes() const noexcept { return count_ * sizeof(T); }

    void allocate(std::size_t count) {
        free();
        count_ = count;
        if (count_ > 0) {
            CUDA_CHECK(cudaMalloc(reinterpret_cast<void**>(&ptr_), bytes()));
        }
    }

    void free() noexcept {
        if (ptr_ != nullptr) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            count_ = 0;
        }
    }

    // Copy host -> device
    void upload(const T* host_ptr, std::size_t count) {
        CUDA_CHECK(cudaMemcpy(ptr_, host_ptr, count * sizeof(T),
                              cudaMemcpyHostToDevice));
    }

    // Copy device -> host
    void download(T* host_ptr, std::size_t count) const {
        CUDA_CHECK(cudaMemcpy(host_ptr, ptr_, count * sizeof(T),
                              cudaMemcpyDeviceToHost));
    }

private:
    T* ptr_ = nullptr;
    std::size_t count_ = 0;
};

// Deterministic PRNG (64-bit LCG) — portable, no external deps

struct LCG {
    uint64_t state;
    explicit LCG(uint64_t seed) : state(seed) {}
    uint64_t next() {
        state = state * 6364136223846793005ULL + 1442695040888963407ULL;
        return state;
    }
    // Uniform float in [0, 1)
    float nextFloat() {
        return static_cast<float>(next() >> 40) * (1.0f / 16777216.0f);
    }
};

struct alignas(16) BF16x8 {
