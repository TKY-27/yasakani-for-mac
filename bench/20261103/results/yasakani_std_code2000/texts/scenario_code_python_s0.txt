Below is a complete, runnable Python 3.11 package. It is designed to be deterministic (no wall‑clock timestamps in state, no randomness, no thread races), to enforce a single active GPU job, to persist state atomically, to resume completed work, to stream structured JSONL events, to handle SIGINT, to redact configured secrets, and to include pytest tests for success, failure, cancellation, and resume.

```python
"""Configuration for the job runner."""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


@dataclass(frozen=True)
class RunnerConfig:
    """Immutable configuration for a runner run.

    Attributes:
        state_dir: Directory where atomic state files are written.
        events_path: Path to the JSONL event stream file.
        secrets: Mapping of secret name -> literal value to redact.
        max_retries: Maximum number of retries per job (0 disables retries).
        gpu_device: Logical GPU device identifier (informational).
    """

    state_dir: Path
    events_path: Path
    secrets: dict[str, str] = field(default_factory=dict)
    max_retries: int = 0
    gpu_device: str = "gpu:0"

    def __post_init__(self) -> None:
        # Ensure directories exist.
        self.state_dir = Path(self.state_dir)
        self.events_path = Path(self.events_path)
        self.state_dir.mkdir(parents=True, exist_ok=True)
        self.events_path.parent.mkdir(parents=True, exist_ok=True)
```

```python
"""Manifest parsing and validation."""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


class ManifestError(Exception):
    """Raised when the manifest is invalid."""


@dataclass(frozen=True)
class Job:
    """A single job in the manifest."""

    id: str
    command: list[str]
    gpu: bool = True
    env: dict[str, str] = field(default_factory=dict)
    timeout_seconds: float | None = None
    retries: int = 0


@dataclass(frozen=True)
class Manifest:
    """Parsed and validated manifest."""

    jobs: list[Job]
    raw: dict[str, Any] = field(default_factory=dict, repr=False)

    @property
    def job_ids(self) -> list[str]:
        return [j.id for j in self.jobs]


def load_manifest(path: Path) -> Manifest:
    """Load and validate a JSON manifest file.

    Expected schema:
    {
      "jobs": [
        {
          "id": "train-1",
          "command": ["python", "train.py", "--epochs", "10"],
          "gpu": true,
          "env": {"CUDA_VISIBLE_DEVICES": "0"},
          "timeout_seconds": 3600,
          "retries": 2
        }
      ]
    }
    """
    path = Path(path)
    if not path.exists():
        raise ManifestError(f"Manifest not found: {path}")

    try:
        raw = json.loads(path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ManifestError(f"Invalid JSON in manifest: {exc}") from exc

    if not isinstance(raw, dict):
        raise ManifestError("Manifest root must be a JSON object")

    jobs_raw = raw.get("jobs")
    if not isinstance(jobs_raw, list):
        raise ManifestError("Manifest must contain a 'jobs' list")

    jobs: list[Job] = []
    seen_ids: set[str] = set()

    for i, j in enumerate(jobs_raw):
        if not isinstance(j, dict):
            raise ManifestError(f"Job at index {i} must be an object")

        job_id = j.get("id")
        if not isinstance(job_id, str) or not job_id:
            raise ManifestError(f"Job at index {i} must have a non-empty string 'id'")
        if job_id in seen_ids:
            raise ManifestError(f"Duplicate job id: {job_id}")
        seen_ids.add(job_id)

        command = j.get("command")
        if not isinstance(command, list) or not command:
            raise ManifestError(f"Job '{job_id}' must have a non-empty 'command' list")
        if not all(isinstance(c, str) for c in command):
            raise ManifestError(f"Job '{job_id}' command entries must be strings")

        gpu = j.get("gpu", True)
        if not isinstance(gpu, bool):
            raise ManifestError(f"Job '{job_id}' 'gpu' must be a boolean")

        env = j.get("env", {})
        if not isinstance(env, dict):
            raise ManifestError(f"Job '{job_id}' 'env' must be an object")
        if not all(isinstance(k, str) and isinstance(v, str) for k, v in env.items()):
            raise ManifestError(f"Job '{job_id}' env keys and values must be strings")

        timeout = j.get("timeout_seconds")
        if timeout is not None and not isinstance(timeout, (int, float)):
            raise ManifestError(f"Job '{job_id}' 'timeout_seconds' must be a number")

        retries = j.get("retries", 0)
        if not isinstance(retries, int) or retries < 0:
            raise ManifestError(f"Job '{job_id}' 'retries' must be a non-negative integer")

        jobs.append(
            Job(
                id=job_id,
                command=command,
                gpu=gpu,
                env=env,
                timeout_seconds=float(timeout) if timeout is not None else None,
                retries=retries,
            )
        )

    return Manifest(jobs=jobs, raw=raw)
```
