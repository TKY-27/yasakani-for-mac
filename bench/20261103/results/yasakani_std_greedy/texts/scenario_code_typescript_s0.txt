# `metrics-service` — a complete, runnable TypeScript 5 service

Below is a full, self-contained implementation. It uses **only Node built-ins** (no third-party runtime dependencies) for the service, and **only `typescript` + `@types/node`** as dev dependencies. Tests use Node's built-in `node:test` runner.

## 4. `src/types.ts` — strict domain types

```ts
/**
 * Core domain types for the metrics service.
 *
 * Design notes:
 *  - `RunRecord` is the *validated* shape. Raw input is `unknown` until it
 *    passes `validateRunRecord`, which narrows it to `RunRecord`.
 *  - `GroupKey` is a canonical string so grouping is O(1) and stable.
 *  - `Aggregation` holds the running statistics for one (model, fixture) group.
 */

/** A single validated request-metrics record. */
export interface RunRecord {
  /** Unique run identifier (e.g. a UUID or a short slug). */
  id: string;
  /** Model name (e.g. "gpt-4", "llama-3-70b"). */
  model: string;
  /** Fixture / dataset name (e.g. "mmlu", "gsm8k"). */
  fixture: string;
  /** Wall-clock timestamp of the run, in milliseconds since epoch. */
  timestamp: number;
  /** Latency in milliseconds (non-negative). */
  latencyMs: number;
  /** Tokens processed (non-negative integer). */
  tokens: number;
  /** Outcome of the run. */
  outcome: RunOutcome;
  /** Optional free-form error message when outcome === "error". */
  error?: string;
}

export type RunOutcome = "success" | "error" | "timeout";

/** Canonical grouping key: `${model}::${fixture}`. */
export type GroupKey = string;

/**
 * Running aggregation for one (model, fixture) group.
 *
 * `sum` and `sumSq` are kept so the sample variance can be computed in
 * O(1) at query time. `values` is a bounded ring buffer used to compute the
 * median without storing every value ever seen (bounded memory).
 */
export interface Aggregation {
  key: GroupKey;
  model: string;
  fixture: string;
  count: number;
  sum: number;
  sumSq: number;
  min: number;
  max: number;
  /** Bounded sample of latency values (most recent `capacity` values). */
  values: number[];
  /** Number of values evicted from the ring buffer (for observability). */
  evicted: number;
  successCount: number;
  errorCount: number;
  timeoutCount: number;
  firstSeen: number;
  lastSeen: number;
}

/** Per-run metadata tracked for the /runs/:id endpoint. */
export interface RunMeta {
  id: string;
  model: string;
  fixture: string;
  timestamp: number;
  outcome: RunOutcome;
  latencyMs: number;
  tokens: number;
  /** Number of records observed for this run id (usually 1). */
  recordCount: number;
}

/** A fully computed summary row for one group. */
export interface GroupSummary {
  model: string;
  fixture: string;
  count: number;
  mean: number;
  median: number;
  /** Sample standard deviation (n-1 denominator). NaN when count < 2. */
  sampleStdDev: number;
  min: number;
  max: number;
  successCount: number;
  errorCount: number;
  timeoutCount: number;
  firstSeen: number;
  lastSeen: number;
}

/** The full /summary response body. */
export interface SummaryResponse {
  generatedAt: number;
  totalRecords: number;
  totalRuns: number;
  groups: GroupSummary[];
}

/** The /health response body. */
export interface HealthResponse {
  status: "ok" | "degraded";
  uptimeSec: number;
  totalRecords: number;
  totalRuns: number;
  groupCount: number;
  memory: {
    rssBytes: number;
    heapUsedBytes: number;
    heapTotalBytes: number;
  };
  version: string;
}

/** The /runs/:id response body. */
export interface RunResponse {
  id: string;
  model: string;
  fixture: string;
  timestamp: number;
  outcome: RunOutcome;
  latencyMs: number;
  tokens: number;
  recordCount: number;
}

/** Error thrown for malformed input. Carries a stable `code`. */
export class MalformedInputError extends Error {
  readonly code: string;
  readonly line?: number;
  constructor(message: string, code: string, line?: number) {
    super(message);
    this.name = "MalformedInputError";
    this.code = code;
    this.line = line;
  }
}

/** Error thrown when the in-memory store is full and a new run id arrives. */
export class StoreFullError extends Error {
  readonly code = "STORE_FULL";
  constructor(message: string) {
    super(message);
    this.name = "StoreFullError";
  }
}
```

## 5. `src/config.ts` — configuration & limits

```ts
/**
 * Central configuration. All limits are bounded to keep memory predictable.
 * Values can be overridden via environment variables.
 */

export interface ServiceConfig {
  /** Port to listen on. */
  port: number;
  /** Host to bind. */
  host: string;
  /** Max bytes accepted in a single HTTP request body. */
  maxBodyBytes: number;
  /** Max number of NDJSON lines accepted in a single request. */
  maxLinesPerRequest: number;
  /** Max number of distinct (model, fixture) groups to retain. */
  maxGroups: number;
  /** Max number of distinct run ids to retain (for /runs/:id). */
  maxRuns: number;
  /** Ring-buffer capacity per group for median computation. */
  medianSampleCapacity: number;
  /** Service version string. */
  version: string;
}

function envInt(name: string, fallback: number): number {
  const raw = process.env[name];
  if (raw === undefined || raw === "") return fallback;
  const n = Number.parseInt(raw, 10);
  return Number.isFinite(n) && n > 0 ? n : fallback;
}

function envStr(name: string, fallback: string): string {
  const raw = process.env[name];
  return raw === undefined || raw === "" ? fallback : raw;
}

export function loadConfig(): ServiceConfig {
  return {
    port: envInt("PORT", 8080),
    host: envStr("HOST", "0.0.0.0"),
    maxBodyBytes: envInt("MAX_BODY_BYTES", 1_000_000),
    maxLinesPerRequest: envInt("MAX_LINES_PER_REQUEST", 10_000),
    maxGroups: envInt("MAX_GROUPS", 10_000),
    maxRuns: envInt("MAX_RUNS", 100_000),
    medianSampleCapacity: envInt("MEDIAN_SAMPLE_CAPACITY", 1_000),
    version: envStr("VERSION", "1.0.0"),
  };
}

## 6. `src/validate.ts` — schema validation (no third-party libs)

```ts
/**
 * Hand-rolled validation for a single NDJSON record.
 *
 * `validateRunRecord` takes `unknown` and returns either a `RunRecord`
 * (on success) or throws `MalformedInputError` with a stable `code`.
 *
 * Validation rules:
 *  - Must be a JSON object (not null, not array, not primitive).
 *  - `id`: non-empty string, max 128 chars.
 *  - `model`: non-empty string, max 64 chars, no whitespace.
 *  - `fixture`: non-empty string, max 64 chars, no whitespace.
 *  - `timestamp`: integer, >= 0, <= 10^15 (sanity bound).
 *  - `latencyMs`: finite number, >= 0, <= 10^9.
 *  - `tokens`: integer, >= 0, <= 10^12.
 *  - `outcome`: one of "success" | "error" | "timeout".
 *  - `error`: optional string, max 512 chars; required when outcome === "error".
 *  - No unknown top-level keys (strict schema).
 */

import { MalformedInputError, RunOutcome, RunRecord } from "./types.js";

const ALLOWED_KEYS = new Set([
  "id",
  "model",
  "fixture",
  "timestamp",
  "latencyMs",
  "tokens",
  "outcome",
  "error",
]);

const OUTCOMES: ReadonlySet<RunOutcome> = new Set(["success", "error", "timeout"]);

const MAX_ID_LEN = 128;
const MAX_NAME_LEN = 64;
const MAX_ERROR_LEN = 512;
const MAX_TIMESTAMP = 1_000_000_000_000_000; // 10^15
const MAX_LATENCY = 1_000_000_000; // 10^9 ms
const MAX_TOKENS = 1_000_000_000_000; // 10^12

function isPlainObject(value: unknown): value is Record<string, unknown> {
  if (typeof value !== "object" || value === null) return false;
  const proto = Object.getPrototypeOf(value);
  return proto === Object.prototype || proto === null;
}

function requireString(
  obj: Record<string, unknown>,
  key: string,
  maxLen: number,
  line?: number,
): string {
  const v = obj[key];
  if (typeof v !== "string") {
    throw new MalformedInputError(`Field "${key}" must be a string`, "TYPE_MISMATCH", line);
  }
  if (v.length === 0) {
    throw new MalformedInputError(`Field "${key}" must be non-empty`, "EMPTY_FIELD", line);
  }
  if (v.length > maxLen) {
    throw new MalformedInputError(
      `Field "${key}" exceeds max length ${maxLen}`,
      "FIELD_TOO_LONG",
      line,
    );
  }
  return v;
}

function requireFiniteNumber(
  obj: Record<string, unknown>,
  key: string,
  min: number,
  max: number,
  line?: number,
): number {
  const v = obj[key];
  if (typeof v !== "number" || !Number.isFinite(v)) {
    throw new MalformedInputError(`Field "${key}" must be a finite number`, "TYPE_MISMATCH", line);
  }
  if (v < min || v > max) {
    throw new MalformedInputError(
      `Field "${key}" out of range [${min}, ${max}]`,
      "OUT_OF_RANGE",
      line,
    );
  }
  return v;
}

function requireInteger(
  obj: Record<string, unknown>,
  key: string,
  min: number,
  max: number,
  line?: number,
): number {
  const v = requireFiniteNumber(obj, key, min, max, line);
  if (!Number.isInteger(v)) {
    throw new MalformedInputError(`Field "${key}" must be an integer`, "NOT_INTEGER", line);
  }
  return v;
}

/**
 * Validate and narrow an `unknown` JSON value to a `RunRecord`.
 * Throws `MalformedInputError` on any violation.
 */
export function validateRunRecord(raw: unknown, line?: number): RunRecord {
  if (!isPlainObject(raw)) {
    throw new MalformedInputError("Record must be a JSON object", "NOT_OBJECT", line);
  }

  // Reject unknown keys (strict schema).
  for (const key of Object.keys(raw)) {
    if (!ALLOWED_KEYS.has(key)) {
      throw new MalformedInputError(`Unknown field "${key}"`, "UNKNOWN_FIELD", line);
    }
  }

  const id = requireString(raw, "id", MAX_ID_LEN, line);
  const model = requireString(raw, "model", MAX_NAME_LEN, line);
  const fixture = requireString(raw, "fixture", MAX_NAME_LEN, line);

  // model / fixture must not contain whitespace (they are used as group keys).
  if (/\s/.test(model)) {
    throw new MalformedInputError(`Field "model" must not contain whitespace`, "INVALID_MODEL", line);
  }
  if (/\s/.test(fixture)) {
    throw new MalformedInputError(
      `Field "fixture" must not contain whitespace`,
      "INVALID_FIXTURE",
      line,
    );
  }

  const timestamp = requireInteger(raw, "timestamp", 0, MAX_TIMESTAMP, line);
  const latencyMs = requireFiniteNumber(raw, "latencyMs", 0, MAX_LATENCY, line);
  const tokens = requireInteger(raw, "tokens", 0, MAX_TOKENS, line);

  const outcomeRaw = raw["outcome"];
  if (typeof outcomeRaw !== "string" || !OUTCOMES.has(outcomeRaw as RunOutcome)) {
    throw new MalformedInputError(
      `Field "outcome" must be one of: success, error, timeout`,
      "INVALID_OUTCOME",
      line,
    );
  }
  const outcome = outcomeRaw as RunOutcome;

  let error: string | undefined;
  if (raw["error"] !== undefined) {
    if (typeof raw["error"] !== "string") {
      throw new MalformedInputError(`Field "error" must be a string`, "TYPE_MISMATCH", line);
    }
    if ((raw["error"] as string).length > MAX_ERROR_LEN) {
      throw new MalformedInputError(
        `Field "error" exceeds max length ${MAX_ERROR_LEN}`,
        "FIELD_TOO_LONG",
        line,
      );
    }
    error = raw["error"] as string;
  }

  if (outcome === "error" && error === undefined) {
    throw new MalformedInputError(
      `Field "error" is required when outcome is "error"`,
      "MISSING_ERROR",
      line,
    );
  }

  const record: RunRecord = {
    id,
    model,
    fixture,
    timestamp,
    latencyMs,
    tokens,
    outcome,
  };
  if
