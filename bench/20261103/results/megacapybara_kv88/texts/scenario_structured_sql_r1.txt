-- 1. Schema Definition
DROP VIEW IF EXISTS v_model_performance_summary;
DROP VIEW IF EXISTS v_fixture_latency_breakdown;
DROP FUNCTION IF EXISTS upsert_request_metrics;

DROP TABLE IF EXISTS request_metrics;
DROP TABLE IF EXISTS runs;
DROP TABLE IF EXISTS fixtures;
DROP TABLE IF EXISTS models;

CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    architecture VARCHAR(100) NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    description TEXT,
    token_count INT NOT NULL CHECK (token_count > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    input_tokens INT NOT NULL CHECK (input_tokens > 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DECIMAL(10, 2) NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted BOOLEAN NOT NULL DEFAULT FALSE,
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_accepted);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);

-- 2. Views
CREATE VIEW v_model_performance_summary AS
SELECT 
    m.name AS model_name,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.latency_ms) AS avg_latency_ms,
    MAX(rm.latency_ms) AS max_latency_ms,
    MIN(rm.latency_ms) AS min_latency_ms,
    SUM(CASE WHEN rm.mtp_accepted THEN 1 ELSE 0 END)::FLOAT / COUNT(*) * 100 AS mtp_acceptance_rate_pct
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name;

CREATE VIEW v_fixture_latency_breakdown AS
SELECT 
    f.kind AS fixture_kind,
    f.token_count AS avg_input_tokens,
    COUNT(rm.metric_id) AS total_requests,
    AVG(rm.latency_ms) AS avg_latency_ms,
    STDDEV_POP(rm.latency_ms) AS latency_stddev
FROM fixtures f
JOIN runs r ON f.fixture_id = r.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY f.kind, f.token_count;

-- 3. Transaction-Safe Upsert Function
CREATE OR REPLACE FUNCTION upsert_request_metrics(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DECIMAL(10, 2),
    p_mtp_accepted BOOLEAN
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, input_tokens, output_tokens, latency_ms, mtp_accepted)
    VALUES (p_run_id, p_request_index, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        timestamp = CURRENT_TIMESTAMP;
END;
$$ LANGUAGE plpgsql;
