// RAII wrapper for device memory
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_(0) {}
    
    explicit DeviceBuffer(size_t bytes) : ptr_(nullptr), size_(bytes) {
        if (bytes > 0) {
            CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        }
    }
    
    ~DeviceBuffer() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }
    
    // Disable copy
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;
    
    // Enable move
    DeviceBuffer(DeviceBuffer&& other) noexcept : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }
    
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            if (ptr_) cudaFree(ptr_);
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }
    
    void* get() const { return ptr_; }
    size_t size() const { return size_; }
    
    template<typename T>
    T* as() const { return static_cast<T*>(ptr_); }

private:
    void* ptr_;
    size_t size_;
};

// Deterministic input generation using a simple LCG for reproducibility
// We generate BF16 values in the range [-1, 1] with some structure
std::vector<__nv_bfloat16> generate_input(int rows, int cols, unsigned int seed) {
    std::vector<__nv_bfloat16> data(static_cast<size_t>(rows) * cols);
    
    // Use a deterministic PRNG
    std::mt19937 rng(seed);
    std::uniform_real_distribution<float> dist(-1.0f, 1.0f);
    
    for (size_t i = 0; i < data.size(); ++i) {
        float val = dist(rng);
        // Add some structure: make certain rows have larger magnitudes
        int row = static_cast<int>(i / cols);
        if (row % 100 == 0) {
            val *= 10.0f;  // Every 100th row has larger values
        }
        data[i] = __float2bfloat16(val);
    }
    
    return data;
}

// Convert BF16 to FP32 for reference computation
std::vector<float> bf16_to_fp32(const std::vector<__nv_bfloat16>& bf16_data) {
    std::vector<float> fp32_data(bf16_data.size());
    for (size_t i = 0; i < bf16_data.size(); ++i) {
        fp32_data[i] = __bfloat162float(bf16_data[i]);
    }
    return fp32_data;
}

// Numerically stable FP32 reference: compute row-wise normalization
// For each row, compute norm = sqrt(sum(x^2)), then divide each element by norm
// If norm is zero, leave the row unchanged (or set to zero)
std::vector<float> compute_reference_normalized(const std::vector<float>& input, int rows, int cols) {
    std::vector<float> result(input.size(), 0.0f);
    
    for (int r = 0; r < rows; ++r) {
        const float* row_ptr = &input[static_cast<size_t>(r) * cols];
        
        // Compute sum of squares using Kahan summation for numerical stability
        float sum_sq = 0.0f;
        float c = 0.0f;  // compensation term
        
        for (int c_idx = 0; c_idx < cols; ++c_idx) {
            float x = row_ptr[c_idx];
            float x2 = x * x;
            
            // Kahan summation
            float y = x2 - c;
            float t = sum_sq + y;
            c = (t - sum_sq) - y;
            sum_sq = t;
        }
        
        float norm = sqrtf(sum_sq);
        
        if (norm > 1e-30f) {
            float inv_norm = 1.0f / norm;
            for (int c_idx = 0; c_idx < cols; ++c_idx) {
                result[static_cast<size_t>(r) * cols + c_idx] = row_ptr[c_idx] * inv_norm;
            }
        } else {
            // Row is all zeros or very small; leave as zero
            for (int c_idx = 0; c_idx < cols; ++c_idx) {
                result[static_cast<size_t>(r) * cols + c_idx] = 0.0f;
            }
        }
    }
    
    return result;
}

// Vectorized CUDA kernel for BF16 row normalization on sm_120a
// Each block handles one row. We use vectorized loads/stores for efficiency.
// On sm_120a, we can leverage wider vector types if available, but we'll use
// standard 128-bit (float4 equivalent) vectorization for BF16.
// 
// Memory layout: Row-major, contiguous BF16 elements per row.
// Launch geometry: gridDim.x = rows, blockDim.x = min(cols/vec_size, max_threads_per_block)

// Define vector type for 8 BF16 values (16 bytes = 128 bits)
struct bf16x8 {
    __nv_bfloat16 v[8];
};

// Kernel: normalize each row in-place
// Input/output: d_data points to BF16 matrix (rows x cols), modified in-place
// d_norms: optional output array of computed norms (for verification)
__global__ void bf16_row_normalize_kernel(
    __nv_bfloat16* __restrict__ data,
    float* __restrict__ norms,
    int rows,
    int cols)
{
    int row = blockIdx.x;
    if (row >= rows) return;
    
    __nv_bfloat16* row_ptr = data + static_cast<size_t>(row) * cols;
    
    // Each thread processes multiple elements
    int tid = threadIdx.x;
    int num_threads = blockDim.x;
    
    // Process 8 BF16 elements per iteration (128-bit vector)
    constexpr int VEC_SIZE = 8;
    int vec_cols = cols / VEC_SIZE;
    int remainder = cols % VEC_SIZE;
    
    // Shared memory for partial sums
    extern __shared__ float s_partial[];
    
    // Each thread accumulates its local sum of squares
    float local_sum_sq = 0.0f;
    float local_c = 0.0f;  // Kahan compensation
    
    // Process vectorized chunks
    for (int idx = tid; idx < vec_cols; idx += num_threads) {
        size_t offset = static_cast<size_t>(idx) * VEC_SIZE;
        
        // Load 8 BF16 values as a 128-bit vector
        uint4 raw = reinterpret_cast<const uint4*>(row_ptr)[idx];
        bf16x8 vec;
        memcpy(vec.v, &raw, sizeof(uint4));
        
        // Compute sum of squares for these 8 elements
        for (int j = 0; j < VEC_SIZE; ++j) {
            float x = __bfloat162float(vec.v[j]);
            float x2 = x * x;
            
            // Kahan summation
            float y = x2 - local_c;
            float t = local_sum_sq + y;
            local_c = (t - local_sum_sq) - y;
            local_sum_sq = t;
        }
    }
    
    // Handle remainder elements
    if (remainder > 0 && tid < remainder) {
        size_t offset = static_cast<size_t>(vec_cols) * VEC_SIZE + tid;
        float x = __bfloat162float(row_ptr[offset]);
        float x2 = x * x;
        
        float y = x2 - local_c;
        float t = local_sum_sq + y;
        local_c = (t - local_sum_sq) - y;
        local_sum_sq = t;
    }
    
    // Reduce within block using shared memory
    s_partial[tid] = local_sum_sq;
    __syncthreads();
    
    // Tree reduction
    for (int stride = num_threads / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            float a = s_partial[tid];
            float b = s_partial[tid + stride];
            
            // Kahan-style addition for combining partial sums
            float y = b - local_c;
            float t = a + y;
            local_c = (t - a) - y;
            s_partial[tid] = t;
        }
        __syncthreads();
    }
    
    float total_sum_sq = s_partial[0];
    float norm = sqrtf(total_sum_sq);
    
    // Store norm for verification
    if (tid == 0 && norms != nullptr) {
        norms[row] = norm;
    }
    
    __syncthreads();
    
    // Now normalize: divide each element by norm
    float inv_norm = (norm > 1e-30f) ? (1.0f / norm) : 0.0f;
    
    // Re-process vectorized chunks and store normalized values
    for (int idx = tid; idx < vec_cols; idx += num_threads) {
        size_t offset = static_cast<size_t>(idx) * VEC_SIZE;
        
        uint4 raw = reinterpret_cast<const uint4*>(row_ptr)[idx];
        bf16x8 vec;
        memcpy(vec.v, &raw, sizeof(uint4));
        
        for (int j = 0; j < VEC_SIZE; ++j) {
            float x = __bfloat162float(vec.v[j]);
            float normalized = x * inv_norm;
            vec.v[j] = __float2bfloat16(normalized);
        }
        
        uint4 out_raw;
        memcpy(&out_raw, vec.v, sizeof(uint4));
        reinterpret_cast<uint4*>(row_ptr)[idx] = out_raw;
    }
    
    // Handle remainder
    if (remainder > 0 && tid < remainder) {
        size_t offset = static_cast<size_t>(vec_cols) * VEC_SIZE + tid;
        float x = __bfloat162float(row_ptr[offset]);
        float normalized = x * inv_norm;
        row_ptr[offset] = __float2bfloat16(normalized);
    }
}
