// RAII wrapper for device memory
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_(0) {}
    
    explicit DeviceBuffer(size_t bytes) : ptr_(nullptr), size_(bytes) {
        if (bytes > 0) {
            CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        }
    }
    
    ~DeviceBuffer() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }
    
    // Move constructor
    DeviceBuffer(DeviceBuffer&& other) noexcept : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }
    
    // Move assignment
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            if (ptr_) cudaFree(ptr_);
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }
    
    // Delete copy operations
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;
    
    void* get() const { return ptr_; }
    size_t size() const { return size_; }
    
private:
    void* ptr_;
    size_t size_;
};

// Convert BF16 to FP32
__device__ inline float bf16_to_float(__nv_bfloat16 val) {
    return __bfloat162float(val);
}

// Convert FP32 to BF16
__device__ inline __nv_bfloat16 float_to_bf16(float val) {
    return __float2bfloat16(val);
}

// Vectorized kernel: normalize each row of a BF16 matrix in-place
// Matrix is stored in row-major order with dimensions rows x cols
// Each block handles one row. Threads within the block cooperate to compute
// the L2 norm and then divide each element by it.
template<int VEC_SIZE>
__global__ void normalize_rows_kernel(__nv_bfloat16* data, int rows, int cols) {
    int row = blockIdx.x;
    if (row >= rows) return;
    
    int tid = threadIdx.x;
    int total_threads = blockDim.x;
    
    // Pointer to this row's data
    __nv_bfloat16* row_data = data + static_cast<size_t>(row) * cols;
    
    // Number of vector elements per thread
    int vec_per_thread = cols / (total_threads * VEC_SIZE);
    int remainder = cols % (total_threads * VEC_SIZE);
    
    // Shared memory for partial sums
    extern __shared__ float shared_sum[];
    
    float local_sum_sq = 0.0f;
    
    // Process vectorized chunks
    for (int i = 0; i < vec_per_thread; ++i) {
        int idx = (tid + i * total_threads) * VEC_SIZE;
        
        // Load VEC_SIZE BF16 values as a uint32_t pair (for VEC_SIZE=4, that's 8 bytes = 2 uint32_t)
        // For VEC_SIZE=4: load 4 bf16 = 8 bytes = 2 uint32_t
        // We'll use aligned loads where possible
        
        float vals[VEC_SIZE];
        for (int j = 0; j < VEC_SIZE; ++j) {
            vals[j] = bf16_to_float(row_data[idx + j]);
            local_sum_sq += vals[j] * vals[j];
        }
    }
    
    // Handle remainder elements
    if (remainder > 0) {
        int base_idx = vec_per_thread * total_threads * VEC_SIZE;
        int rem_tid = tid;
        if (rem_tid < remainder) {
            int idx = base_idx + rem_tid;
            float val = bf16_to_float(row_data[idx]);
            local_sum_sq += val * val;
        }
    }
    
    // Reduce within block using shared memory
    shared_sum[tid] = local_sum_sq;
    __syncthreads();
    
    // Tree reduction
    for (int stride = total_threads / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shared_sum[tid] += shared_sum[tid + stride];
        }
        __syncthreads();
    }
    
    float row_norm = shared_sum[0];
    float inv_norm = 1.0f / sqrtf(row_norm);
    
    // Write back normalized values
    for (int i = 0; i < vec_per_thread; ++i) {
        int idx = (tid + i * total_threads) * VEC_SIZE;
        for (int j = 0; j < VEC_SIZE; ++j) {
            float val = bf16_to_float(row_data[idx + j]) * inv_norm;
            row_data[idx + j] = float_to_bf16(val);
        }
    }
    
    if (remainder > 0) {
        int base_idx = vec_per_thread * total_threads * VEC_SIZE;
        int rem_tid = tid;
        if (rem_tid < remainder) {
            int idx = base_idx + rem_tid;
            float val = bf16_to_float(row_data[idx]) * inv_norm;
            row_data[idx] = float_to_bf16(val);
        }
    }
}

// Simpler non-vectorized kernel for fallback or comparison
__global__ void normalize_rows_simple_kernel(__nv_bfloat16* data, int rows, int cols) {
    int row = blockIdx.x;
    if (row >= rows) return;
    
    int tid = threadIdx.x;
    int total_threads = blockDim.x;
    
    __nv_bfloat16* row_data = data + static_cast<size_t>(row) * cols;
    
    extern __shared__ float shared_sum[];
    
    float local_sum_sq = 0.0f;
    
    for (int i = tid; i < cols; i += total_threads) {
        float val = bf16_to_float(row_data[i]);
        local_sum_sq += val * val;
    }
    
    shared_sum[tid] = local_sum_sq;
    __syncthreads();
    
    for (int stride = total_threads / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shared_sum[tid] += shared_sum[tid + stride];
        }
        __syncthreads();
    }
    
    float row_norm = shared_sum[0];
    float inv_norm = 1.0f / sqrtf(row_norm);
    
    for (int i = tid; i < cols; i += total_threads) {
        float val = bf16_to_float(row_data[i]) * inv_norm;
        row_data[i] = float_to_bf16(val);
    }
}

// Host-side reference implementation using FP32 for numerical stability
void reference_normalize_fp32(std::vector<float>& data, int rows, int cols) {
    for (int r = 0; r < rows; ++r) {
        float* row = data.data() + static_cast<size_t>(r) * cols;
        
        // Compute L2 norm in FP32
        float sum_sq = 0.0f;
        for (int c = 0; c < cols; ++c) {
            sum_sq += row[c] * row[c];
        }
        float norm = std::sqrt(sum_sq);
        float inv_norm = 1.0f / norm;
        
        // Normalize
        for (int c = 0; c < cols; ++c) {
            row[c] *= inv_norm;
        }
    }
}

// Generate deterministic input data
std::vector<__nv_bfloat16> generate_input(int rows, int cols, unsigned seed) {
    std::mt19937 rng(seed);
    std::uniform_real_distribution<float> dist(-1.0f, 1.0f);
    
    std::vector<__nv_bfloat16> data(static_cast<size_t>(rows) * cols);
    for (size_t i = 0; i < data.size(); ++i) {
        float val = dist(rng);
        data[i] = float_to_bf16(val);
    }
    return data;
}

// Parse command-line arguments
struct Args {
    int rows = 1024;
    int cols = 1024;
    unsigned seed = 42;
    int warmup_iters = 5;
    int timing_iters = 20;
    bool verbose = false;
    bool help = false;
};

Args parse_args(int argc, char** argv) {
    Args args;
    
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        
        if (arg == "--help" || arg == "-h") {
            args.help = true;
        } else if (arg == "--rows" && i + 1 < argc) {
            args.rows = std::stoi(argv[++i]);
        } else if (arg == "--cols" && i + 1 < argc) {
            args.cols = std::stoi(argv[++i]);
        } else if (arg == "--seed" && i + 1 < argc) {
            args.seed = static_cast<unsigned>(std::stoul(argv[++i]));
        } else if (arg == "--warmup" && i + 1 < argc) {
            args.warmup_iters = std::stoi(argv[++i]);
        } else if (arg == "--iters" && i + 1 < argc) {
            args.timing_iters = std::stoi(argv[++i]);
        } else if (arg == "--verbose" || arg == "-v") {
            args.verbose = true;
        } else {
            fprintf(stderr, "Unknown argument: %s\n", arg.c_str());
            args.help = true;
        }
    }
    
    return args;
}

void print_usage() {
    printf("Usage: bf16_row_normalize [options]\n");
    printf("Options:\n");
    printf("  --rows N       Number of rows (default: 1024)\n");
    printf("  --cols N       Number of columns (default: 1024)\n");
    printf("  --seed N       Random seed for deterministic input (default: 42)\n");
    printf("  --warmup N     Number of warm-up iterations (default: 5)\n");
    printf("  --iters N      Number of timing iterations (default: 20)\n");
    printf("  --verbose      Enable verbose output\n");
    printf("  --help         Show this help message\n");
}

int main(int argc, char** argv) {
    Args args = parse_args(argc, argv);
    
    if (args.help) {
        print_usage();
        return 0;
    }
    
    if (args.rows <= 0 || args.cols <= 0) {
        fprintf(stderr, "Error: rows and cols must be positive.\n");
        return EXIT_FAILURE;
    }
    
    printf("BF16 Row Normalization on sm_120a GPU\n");
    printf("======================================\n");
    printf("Matrix dimensions: %d x %d\n", args.rows, args.cols);
    printf("Seed: %u\n", args.seed);
    printf("Warm-up iterations: %d\n", args.warmup_iters);
    printf("Timing iterations: %d\n", args.timing_iters);
    printf("\n");
    
    // Check GPU properties
    int device_count = 0;
    CUDA_CHECK(cudaGetDeviceCount(&device_count));
    if (device_count == 0) {
        fprintf(stderr, "No CUDA devices found.\n");
        return EXIT_FAILURE;
    }
    
    cudaDeviceProp prop;
    CUDA_CHECK(cudaGetDeviceProperties(&prop, 0));
    printf("GPU: %s\n", prop.name);
    printf("Compute capability: %d.%d\n", prop.major, prop.minor);
    printf("Global memory: %.2f GB\n", prop.totalGlobalMem / (1024.0 * 1024.0 * 1024.0));
    printf("SM count: %d\n", prop.multiProcessorCount);
    printf("\n");
    
    size_t num_elements = static_cast<size_t>(args.rows) * args.cols;
    size_t byte_size = num_elements * sizeof(__nv_bfloat16);
    
    printf("Total elements: %zu\n", num_elements);
    printf("Data size: %.2f MB\n", byte_size / (1024.0 * 1024.0));
    printf("\n");
    
    // Generate deterministic input
    auto host_data = generate_input(args.rows, args.cols, args.seed);
    
    // Create device buffers
    DeviceBuffer d_input(byte_size);
    DeviceBuffer d_output(byte_size);
    
    // Copy input to device
    CUDA_CHECK(cudaMemcpy(d_input.get(), host_data.data(), byte_size, cudaMemcpyHostToDevice));
    
    // Also create an FP32 copy for reference verification
    std::vector<float> fp32_data(num_elements);
    for (size_t i = 0; i < num_elements; ++i) {
        fp32_data[i] = bf16_to_float(host_data[i]);
    }
    
    // Determine launch geometry
    // Use 256 threads per block
    const int BLOCK_SIZE = 256;
    int grid_size = args.rows;
    size_t shared_mem_size = BLOCK_SIZE * sizeof(float);
    
    // Choose vector size based on column count
    // For good performance, we want each thread to handle multiple elements
    int vec_size = 4;  // 4 BF16 = 8 bytes per vector load
    
    if (args.verbose) {
        printf("Launch config: grid=%d, block=%d, shared_mem=%zu bytes, vec_size=%d\n", 
               grid_size, BLOCK_SIZE, shared_mem_size, vec_size);
    }
    
    // Warm-up runs
    if (args.verbose) {
        printf("Running %d warm-up iterations...\n", args.warmup_iters);
    }
    
    for (int i = 0; i < args.warmup_iters; ++i) {
        // Reset output to input before each run
        CUDA_CHECK(cudaMemcpy(d_output.get(), d_input.get(), byte_size, cudaMemcpyDeviceToDevice));
        
        normalize_rows_simple_kernel<<<grid_size, BLOCK_SIZE, shared_mem_size>>>(
            static_cast<__nv_bfloat16*>(d_output.get()), args.rows, args.cols);
        CUDA_CHECK(cudaDeviceSynchronize());
    }
    
    // Timing runs
    cudaEvent_t start_event, stop_event;
    CUDA_CHECK(cudaEventCreate(&start_event));
    CUDA_CHECK(cudaEventCreate(&stop_event));
    
    float total_time_ms = 0.0f;
    
    if (args.verbose) {
        printf("Running %d timing iterations...\n", args.timing_iters);
    }
    
    for (int i = 0; i < args.timing_iters; ++i) {
        // Reset output to input
        CUDA_CHECK(cudaMemcpy(d_output.get(), d_input.get(), byte_size, cudaMemcpyDeviceToDevice));
        
        CUDA_CHECK(cudaEventRecord(start_event));
        
        normalize_rows_simple_kernel<<<grid_size, BLOCK_SIZE, shared_mem_size>>>(
            static_cast<__nv_bfloat16*>(d_output.get()), args.rows, args.cols);
        
        CUDA_CHECK(cudaEventRecord(stop_event));
        CUDA_CHECK(cudaEventSynchronize(stop_event));
        
        float elapsed_ms = 0.0f;
        CUDA_CHECK(cudaEventElapsedTime(&elapsed_ms, start_event, stop_event));
        total_time_ms += elapsed_ms;
    }
    
    float avg_time_ms = total_time_ms / args.timing_iters;
    double throughput_gbps = (byte_size * 2.0) / (avg_time_ms * 1e-3) / 1e9;  // Read + write
    
    printf("Average kernel time: %.4f ms\n", avg_time_ms);
    printf("Throughput: %.2f GB/s (read+write)\n", throughput_gbps);
    printf("\n");
    
    // Verify correctness against FP32 reference
    if (args.verbose) {
        printf("Verifying correctness against FP32 reference...\n");
    }
    
    // Get the last result from device
    std::vector<__nv_bfloat16> device_result(num_elements);
    CUDA_CHECK(cudaMemcpy(device_result.data(), d_output.get(), byte_size, cudaMemcpyDeviceToHost));
    
    // Compute reference in FP32
    std::vector<float> ref_data = fp32_data;
    reference_normalize_fp32(ref_data, args.rows, args.cols);
    
    // Compare
    float max_rel_error = 0.0f;
    float avg_rel_error = 0.0f;
    int mismatches = 0;
    const float tolerance = 0.01f;  //
