BEGIN;

-- 1. Create Tables
CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    architecture VARCHAR(100),
    context_length INT CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    description TEXT,
    token_count INT CHECK (token_count >= 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) CHECK (status IN ('running', 'completed', 'failed')),
    total_tokens_generated INT CHECK (total_tokens_generated >= 0),
    total_time_ms INT CHECK (total_time_ms >= 0),
    mtp_acceptance_rate FLOAT CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    long_context_retention_score FLOAT CHECK (long_context_retention_score BETWEEN 0 AND 1),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tokens_per_second FLOAT CHECK (tokens_per_second >= 0),
    latency_ms INT CHECK (latency_ms >= 0),
    memory_usage_mb INT CHECK (memory_usage_mb >= 0),
    batch_size INT CHECK (batch_size >= 1)
);

-- 2. Create Indexes
CREATE INDEX IF NOT EXISTS idx_runs_model_id ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture_id ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_started_at ON runs(started_at);
CREATE INDEX IF NOT EXISTS idx_request_metrics_run_id ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_request_metrics_timestamp ON request_metrics(timestamp);
