BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    model_name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0)
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_kind VARCHAR(50) NOT NULL,
    description TEXT,
    UNIQUE (fixture_kind)
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    start_time TIMESTAMP NOT NULL DEFAULT NOW(),
    end_time TIMESTAMP NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    batch_size INT NOT NULL CHECK (batch_size > 0),
    UNIQUE (model_id, fixture_id, start_time)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_id INT NOT NULL,
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms FLOAT NOT NULL CHECK (latency_ms > 0),
    mtp_accepted INT NOT NULL DEFAULT 0 CHECK (mtp_accepted >= 0),
    UNIQUE (run_id, request_id)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);

CREATE OR REPLACE VIEW vw_run_summary AS
SELECT 
    r.run_id,
    m.model_name,
    f.fixture_kind,
    r.start_time,
    r.context_length,
    r.batch_size,
    COUNT(met.metric_id) as total_requests,
    AVG(met.latency_ms) as avg_latency_ms,
    SUM(met.input_tokens) as total_input_tokens,
    SUM(met.output_tokens) as total_output_tokens,
    AVG(met.mtp_accepted) as avg_mtp_accepted
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
GROUP BY r.run_id, m.model_name, f.fixture_kind, r.start_time, r.context_length, r.batch_size;

CREATE OR REPLACE VIEW vw_model_performance AS
SELECT 
    m.model_name,
    f.fixture_kind,
    COUNT(DISTINCT r.run_id) as run_count,
    AVG(rm.latency_ms) as avg_latency,
    STDDEV(rm.latency_ms) as latency_stddev,
    SUM(rm.output_tokens) * 1.0 / SUM(EXTRACT(EPOCH FROM (r.end_time - r.start_time))) as throughput_tokens_per_sec
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.model_name, f.fixture_kind;

CREATE OR REPLACE FUNCTION upsert_request_metrics(
    p_run_id INT,
    p_request_id INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms FLOAT,
    p_mtp_accepted INT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_id, input_tokens, output_tokens, latency_ms, mtp_accepted)
    VALUES (p_run_id, p_request_id, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted)
    ON CONFLICT (run_id, request_id) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted;
END;
$$ LANGUAGE plpgsql;
