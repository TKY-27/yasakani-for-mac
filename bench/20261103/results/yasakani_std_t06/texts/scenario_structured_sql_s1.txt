BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0)
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL,
    description TEXT,
    UNIQUE (kind)
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models (model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures (fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT NOW(),
    duration_ms INT NOT NULL CHECK (duration_ms > 0),
    status VARCHAR(20) NOT NULL CHECK (status IN ('completed', 'failed', 'aborted')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs (run_id) ON DELETE CASCADE,
    tokens_generated INT NOT NULL CHECK (tokens_generated >= 0),
    tokens_per_second DECIMAL(10, 2) NOT NULL CHECK (tokens_per_second >= 0),
    first_token_latency_ms INT NOT NULL CHECK (first_token_latency_ms > 0),
    mtp_accepted_count INT NOT NULL DEFAULT 0 CHECK (mtp_accepted_count >= 0),
    mtp_total_count INT NOT NULL DEFAULT 0 CHECK (mtp_total_count >= 0),
    long_context_retention_score DECIMAL(5, 2) CHECK (long_context_retention_score BETWEEN 0 AND 1)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs (model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs (fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_started ON runs (started_at);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics (run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_tps ON request_metrics (tokens_per_second);

CREATE VIEW IF NOT EXISTS v_model_fixture_summary AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(r.run_id) AS total_runs,
    AVG(rm.tokens_per_second) AS avg_tps,
    MIN(rm.first_token_latency_ms) AS min_ttft,
    MAX(rm.first_token_latency_ms) AS max_ttft
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE VIEW IF NOT EXISTS v_mtp_acceptance_stats AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    SUM(rm.mtp_accepted_count) AS total_accepted,
    SUM(rm.mtp_total_count) AS total_attempts,
    CASE 
        WHEN SUM(rm.mtp_total_count) > 0 THEN ROUND(100.0 * SUM(rm.mtp_accepted_count) / SUM(rm.mtp_total_count), 2)
        ELSE 0 
    END AS acceptance_pct
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_tokens_gen INT,
    p_tps DECIMAL(10, 2),
    p_ttft INT,
    p_mtp_acc INT,
    p_mtp_tot INT,
    p_retention DECIMAL(5, 2)
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, tokens_generated, tokens_per_second, first_token_latency_ms, mtp_accepted_count, mtp_total_count, long_context_retention_score)
    VALUES (p_run_id, p_tokens_gen, p_tps, p_ttft, p_mtp_acc, p_mtp_tot, p_retention)
    ON CONFLICT (run_id) DO UPDATE SET
        tokens_generated = EXCLUDED.tokens_generated,
        tokens_per_second = EXCLUDED.tokens_per_second,
        first_token_latency_ms = EXCLUDED.first_token_latency_ms,
        mtp_accepted_count = EXCLUDED.mtp_accepted_count,
        mtp_total_count = EXCLUDED.mtp_total_count,
        long_context_retention_score = EXCLUDED.long_context_retention_score;
END;
$$ LANGUAGE plpgsql;
