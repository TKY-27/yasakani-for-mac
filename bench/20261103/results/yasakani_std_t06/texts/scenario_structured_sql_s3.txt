BEGIN;

CREATE SCHEMA IF NOT EXISTS llm_bench;

CREATE TABLE IF NOT EXISTS llm_bench.models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS llm_bench.fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL,
    description VARCHAR(255) NOT NULL,
    avg_token_length INT NOT NULL CHECK (avg_token_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_fixture_kind CHECK (kind IN ('code', 'math', 'long_context', 'conversation'))
);

CREATE TABLE IF NOT EXISTS llm_bench.runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES llm_bench.models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES llm_bench.fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'running')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS llm_bench.request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES llm_bench.runs(run_id) ON DELETE CASCADE,
    request_id INT NOT NULL,
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms FLOAT NOT NULL CHECK (latency_ms > 0),
    mtp_accepted_tokens INT DEFAULT 0 CHECK (mtp_accepted_tokens >= 0),
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_id)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON llm_bench.runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON llm_bench.runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON llm_bench.request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_timestamp ON llm_bench.request_metrics(timestamp);

CREATE VIEW IF NOT EXISTS llm_bench.v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(rm.metric_id) AS total_requests,
    AVG(rm.latency_ms) AS avg_latency_ms,
    SUM(rm.input_tokens) AS total_input_tokens,
    SUM(rm.output_tokens) AS total_output_tokens,
    (SUM(rm.output_tokens) / NULLIF(COUNT(rm.metric_id), 0)) AS avg_output_tokens_per_request
FROM llm_bench.runs r
JOIN llm_bench.models m ON r.model_id = m.model_id
JOIN llm_bench.fixtures f ON r.fixture_id = f.fixture_id
JOIN llm_bench.request_metrics rm ON r.run_id = rm.run_id
GROUP BY r.run_id, m.name, f.kind;

CREATE VIEW IF NOT EXISTS llm_bench.v_mtp_analysis AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    AVG(rm.mtp_accepted_tokens) AS avg_mtp_accepted,
    MAX(rm.mtp_accepted_tokens) AS max_mtp_accepted,
    (SUM(rm.mtp_accepted_tokens) / NULLIF(SUM(rm.output_tokens), 0)) * 100 AS mtp_acceptance_rate_pct
FROM llm_bench.request_metrics rm
JOIN llm_bench.runs r ON rm.run_id = r.run_id
JOIN llm_bench.models m ON r.model_id = m.model_id
JOIN llm_bench.fixtures f ON r.fixture_id = f.fixture_id
GROUP BY r.run_id, m.name, f.kind;

CREATE OR REPLACE FUNCTION llm_bench.upsert_request_metric(
    p_run_id INT,
    p_request_id INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms FLOAT,
    p_mtp_accepted_tokens INT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO llm_bench.request_metrics (run_id, request_id, input_tokens, output_tokens, latency_ms, mtp_accepted_tokens)
    VALUES (p_run_id, p_request_id, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted_tokens)
    ON CONFLICT (run_id, request_id) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens;
END;
$$ LANGUAGE plpgsql;
