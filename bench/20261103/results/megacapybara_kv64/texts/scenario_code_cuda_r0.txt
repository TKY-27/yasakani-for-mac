#define CUDA_CHECK(call) \

// RAII wrapper for device memory
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_(0) {}
    
    explicit DeviceBuffer(size_t bytes) : ptr_(nullptr), size_(bytes) {
        CUDA_CHECK(cudaMalloc(&ptr_, bytes));
    }
    
    ~DeviceBuffer() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }
    
    // Disable copy
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;
    
    // Enable move
    DeviceBuffer(DeviceBuffer&& other) noexcept : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }
    
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            if (ptr_) cudaFree(ptr_);
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }
    
    void* get() const { return ptr_; }
    size_t size() const { return size_; }
    
private:
    void* ptr_;
    size_t size_;
};

// Convert BF16 to FP32
__device__ inline float bf16_to_float(__nv_bfloat16 b) {
    return __bfloat162float(b);
}

// Convert FP32 to BF16
__device__ inline __nv_bfloat16 float_to_bf16(float f) {
    return __float2bfloat16(f);
}

// Vectorized kernel: normalize each row of a BF16 matrix in-place.
// Each block handles one row. Threads within the block cooperatively compute
// the L2 norm of the row using shared memory reduction, then divide each element.
// We use vectorized loads/stores with uint4 (16 bytes = 8 BF16 values per thread).
template<int VEC_SIZE>
__global__ void row_normalize_kernel(__nv_bfloat16* data, int rows, int cols) {
    // Each block processes one row
    int row = blockIdx.x;
    if (row >= rows) return;
    
    __nv_bfloat16* row_ptr = data + static_cast<size_t>(row) * cols;
    
    // Number of vector elements per row
    int vec_per_row = cols / VEC_SIZE;
    
    // Shared memory for partial sums (one per thread)
    extern __shared__ float s_partial[];
    
    int tid = threadIdx.x;
    int num_threads = blockDim.x;
    
    // Each thread loads VEC_SIZE consecutive BF16 values as a uint4
    // uint4 is 16 bytes = 8 BF16 values
    uint4 val;
    float sum_sq = 0.0f;
    
    // Load and accumulate squared values
    for (int i = tid; i < vec_per_row; i += num_threads) {
        // reinterpret as uint4 for vectorized load
        val = reinterpret_cast<uint4*>(row_ptr)[i];
        
        // Extract 8 BF16 values from the uint4
        __nv_bfloat16* elems = reinterpret_cast<__nv_bfloat16*>(&val);
        for (int j = 0; j < VEC_SIZE; ++j) {
            float f = bf16_to_float(elems[j]);
            sum_sq += f * f;
        }
    }
    
    // Store partial sum in shared memory
    s_partial[tid] = sum_sq;
    __syncthreads();
    
    // Tree reduction in shared memory
    for (int stride = num_threads / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_partial[tid] += s_partial[tid + stride];
        }
        __syncthreads();
    }
    
    // All threads read the final sum
    float total_sum_sq = s_partial[0];
    float inv_norm = rsqrtf(total_sum_sq);
    
    // Normalize: multiply each element by inv_norm
    for (int i = tid; i < vec_per_row; i += num_threads) {
        val = reinterpret_cast<uint4*>(row_ptr)[i];
        __nv_bfloat16* elems = reinterpret_cast<__nv_bfloat16*>(&val);
        for (int j = 0; j < VEC_SIZE; ++j) {
            float f = bf16_to_float(elems[j]);
            elems[j] = float_to_bf16(f * inv_norm);
        }
        reinterpret_cast<uint4*>(row_ptr)[i] = val;
    }
}

// Host-side reference implementation using FP32 for numerical stability
void reference_normalize_fp32(std::vector<float>& mat, int rows, int cols) {
    for (int r = 0; r < rows; ++r) {
        float sum_sq = 0.0f;
        for (int c = 0; c < cols; ++c) {
            float v = mat[r * cols + c];
            sum_sq += v * v;
        }
        float inv_norm = 1.0f / std::sqrt(sum_sq);
        for (int c = 0; c < cols; ++c) {
            mat[r * cols + c] *= inv_norm;
        }
    }
}

// Generate deterministic input: fill BF16 matrix with reproducible pseudo-random values
void generate_input(std::vector<__nv_bfloat16>& mat, int rows, int cols, unsigned seed) {
    std::mt19937 rng(seed);
    std::uniform_real_distribution<float> dist(-1.0f, 1.0f);
    
    for (size_t i = 0; i < mat.size(); ++i) {
        float v = dist(rng);
        mat[i] = float_to_bf16(v);
    }
}

// Parse command-line arguments
struct Args {
    int rows = 1024;
    int cols = 4096;
    int warmup_iters = 5;
    int timing_iters = 20;
    double tolerance = 1e-2; // relative tolerance for verification
    bool verbose = true;
};

bool parse_args(int argc, char** argv, Args& args) {
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        if (arg == "--rows" && i + 1 < argc) {
            args.rows = std::stoi(argv[++i]);
        } else if (arg == "--cols" && i + 1 < argc) {
            args.cols = std::stoi(argv[++i]);
        } else if (arg == "--warmup" && i + 1 < argc) {
            args.warmup_iters = std::stoi(argv[++i]);
        } else if (arg == "--iters" && i + 1 < argc) {
            args.timing_iters = std::stoi(argv[++i]);
        } else if (arg == "--tolerance" && i + 1 < argc) {
            args.tolerance = std::stod(argv[++i]);
        } else if (arg == "--quiet") {
            args.verbose = false;
        } else if (arg == "--help" || arg == "-h") {
            printf("Usage: %s [options]\n", argv[0]);
            printf("  --rows N       Number of rows (default: 1024)\n");
            printf("  --cols N       Number of columns (default: 4096)\n");
            printf("  --warmup N     Warm-up iterations (default: 5)\n");
            printf("  --iters N      Timing iterations (default: 20)\n");
            printf("  --tolerance T  Relative tolerance for verification (default: 1e-2)\n");
            printf("  --quiet        Suppress verbose output\n");
            return false;
        } else {
            fprintf(stderr, "Unknown argument: %s\n", arg.c_str());
            return false;
        }
    }
    return true;
}

int main(int argc, char** argv) {
    Args args;
    if (!parse_args(argc, argv, args)) {
        return EXIT_FAILURE;
    }
    
    if (args.verbose) {
        printf("BF16 Row Normalization on sm_120a\n");
        printf("Rows: %d, Cols: %d\n", args.rows, args.cols);
        printf("Warmup iters: %d, Timing iters: %d\n", args.warmup_iters, args.timing_iters);
        printf("Tolerance: %.6e\n", args.tolerance);
    }
    
    int rows = args.rows;
    int cols = args.cols;
    size_t total_elements = static_cast<size_t>(rows) * cols;
    size_t byte_size = total_elements * sizeof(__nv_bfloat16);
    
    if (args.verbose) {
        printf("Total elements: %zu, Byte size: %zu bytes (%.2f MB)\n", 
               total_elements, byte_size, byte_size / (1024.0 * 1024.0));
    }
    
    // Check that cols is divisible by VEC_SIZE (8)
    constexpr int VEC_SIZE = 8;
    if (cols % VEC_SIZE != 0) {
        fprintf(stderr, "Error: cols must be divisible by %d\n", VEC_SIZE);
        return EXIT_FAILURE;
    }
    
    // Generate deterministic input on host
    std::vector<__nv_bfloat16> h_input(total_elements);
    generate_input(h_input, rows, cols, 42);
    
    // Create FP32 copy for reference computation
    std::vector<float> h_ref(total_elements);
    for (size_t i = 0; i < total_elements; ++i) {
        h_ref[i] = bf16_to_float(h_input[i]);
    }
    
    // Compute reference normalization in FP32
    reference_normalize_fp32(h_ref, rows, cols);
    
    // Allocate device buffers
    DeviceBuffer d_input(byte_size);
    DeviceBuffer d_output(byte_size);
    
    // Copy input to device
    CUDA_CHECK(cudaMemcpy(d_input.get(), h_input.data(), byte_size, cudaMemcpyHostToDevice));
    
    // Configure kernel launch parameters
    // One block per row
    dim3 grid(rows);
    // Choose block size: enough threads to cover vec_per_row
    int vec_per_row = cols / VEC_SIZE;
    int block_size = std::min(vec_per_row, 256); // Cap at 256 threads per block
    
    // Shared memory size: block_size floats
    size_t shared_mem_size = block_size * sizeof(float);
    
    if (args.verbose) {
        printf("Grid: (%d, 1, 1), Block: (%d, 1, 1)\n", rows, block_size);
        printf("Shared memory per block: %zu bytes\n", shared_mem_size);
    }
    
    // Warm-up runs
    for (int i = 0; i < args.warmup_iters; ++i) {
        row_normalize_kernel<VEC_SIZE><<<grid, block_size, shared_mem_size>>>(
            static_cast<__nv_bfloat16*>(d_input.get()), rows, cols);
    }
    CUDA_CHECK(cudaDeviceSynchronize());
    
    // Reset input for timing (since warm-up modified it)
    CUDA_CHECK(cudaMemcpy(d_input.get(), h_input.data(), byte_size, cudaMemcpyHostToDevice));
    
    // Timing runs with CUDA events
    cudaEvent_t start_event, stop_event;
    CUDA_CHECK(cudaEventCreate(&start_event));
    CUDA_CHECK(cudaEventCreate(&stop_event));
    
    float total_time_ms = 0.0f;
    
    for (int i = 0; i < args.timing_iters; ++i) {
        // Reset input before each timed run
        CUDA_CHECK(cudaMemcpy(d_input.get(), h_input.data(), byte_size, cudaMemcpyHostToDevice));
        
        CUDA_CHECK(cudaEventRecord(start_event));
        row_normalize_kernel<VEC_SIZE><<<grid, block_size, shared_mem_size>>>(
            static_cast<__nv_bfloat16*>(d_input.get()), rows, cols);
        CUDA_CHECK(cudaEventRecord(stop_event));
        CUDA_CHECK(cudaEventSynchronize(stop_event));
        
        float elapsed_ms;
        CUDA_CHECK(cudaEventElapsedTime(&elapsed_ms, start_event, stop_event));
        total_time_ms += elapsed_ms;
    }
    
    float avg_time_ms = total_time_ms / args.timing_iters;
    
    // Copy result back to host for verification
    std::vector<__nv_bfloat16> h_output(total_elements);
    CUDA_CHECK(cudaMemcpy(h_output.data(), d_input.get(), byte_size, cudaMemcpyDeviceToHost));
    
    // Verify against reference
    double max_rel_error = 0.0;
    int mismatches = 0;
    for (size_t i = 0; i < total_elements; ++i) {
        float ref_val = h_ref[i];
        float out_val = bf16_to_float(h_output[i]);
        
        float denom = std::max(std::fabs(ref_val), 1e-8f);
        float rel_err = std::fabs(out_val - ref_val) / denom;
        
        if (rel_err > max_rel_error) {
            max_rel_error = rel_err;
        }
        
        if (rel_err > args.tolerance) {
            mismatches++;
        }
    }
    
    // Cleanup events
    CUDA_CHECK(cudaEventDestroy(start_event));
    CUDA_CHECK(cudaEventDestroy(stop_event));
    
    if (args.verbose) {
        printf("\nResults:\n");
        printf("  Average kernel time: %.4f ms\n", avg_time_ms);
        printf("  Max relative error: %.6e\n", max_rel_error);
        printf("  Mismatches (> tolerance): %d / %zu\n", mismatches, total_elements);
        
        if (mismatches == 0) {
            printf("  Verification: PASSED\n");
        } else {
            printf("  Verification: FAILED\n");
        }
        
        // Compute throughput
        double gb_read = byte_size / (1024.0 * 1024.0);
        double gb_written = byte_size / (1024.0 * 1024.0);
        double total_gb = gb_read + gb_written;
        double gbps = total_gb / (avg_time_ms / 1000.0);
        printf("  Throughput: %.2f GB/s (read+write)\n", gbps);
    }
    
    return (mismatches == 0) ? EXIT_SUCCESS : EXIT_FAILURE;
}

## Explanation of Memory Layout and Launch Geometry

### Memory Layout

The BF16 matrix is stored in **row-major order** in both host and device memory. For a matrix with `R` rows and `C` columns, element `(r, c)` resides at linear offset `r * C + c`. Each element is 2 bytes (`__nv_bfloat16`), so the total allocation is `R * C * 2` bytes.

Within each row, elements are laid out contiguously. The kernel exploits this by loading **8 consecutive BF16 values** (16 bytes) per thread via a single `uint4` load instruction. This achieves full coalescing because adjacent threads access adjacent 16-byte chunks, and the hardware can service these as wide, aligned transactions. The requirement that `cols` be divisible by 8 ensures every row's worth of vector loads aligns cleanly without tail handling.

### Launch Geometry

- **Grid**: One block per row → `gridDim.x = R`, `gridDim.y = gridDim.z = 1`. This maps naturally to the problem structure: each block independently normalizes one complete row.
- **Block**: Up to 256 threads per block (`blockDim.x = min(cols/8, 256)`). If a row has more than 256 vector elements, each thread iterates over multiple vectors in a strided loop. Fewer threads would underutilize the SM; more would waste registers and reduce occupancy.
