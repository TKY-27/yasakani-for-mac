    do { \

// RAII wrapper for device memory
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_(0) {}
    
    explicit DeviceBuffer(size_t bytes) : ptr_(nullptr), size_(bytes) {
        if (bytes > 0) {
            CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        }
    }
    
    ~DeviceBuffer() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }
    
    // Move constructor
    DeviceBuffer(DeviceBuffer&& other) noexcept : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }
    
    // Move assignment
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            if (ptr_) cudaFree(ptr_);
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }
    
    // Delete copy operations
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;
    
    void* get() const { return ptr_; }
    size_t size() const { return size_; }
    
    template<typename T>
    T* as() const { return static_cast<T*>(ptr_); }

private:
    void* ptr_;
    size_t size_;
};

// Convert BF16 to FP32
__device__ __forceinline__ float bf16_to_float(__nv_bfloat16 val) {
    return __bfloat162float(val);
}

// Convert FP32 to BF16
__device__ __forceinline__ __nv_bfloat16 float_to_bf16(float val) {
    return __float2bfloat16(val);
}

// Vectorized kernel: each thread block processes one row.
// Each thread handles multiple elements using vector loads/stores.
// The kernel computes the L2 norm in FP32 for numerical stability,
// then divides each element by the norm and stores back as BF16.
template<int VEC_SIZE>
__global__ void normalize_rows_kernel(
    __nv_bfloat16* __restrict__ data,
    int rows,
    int cols)
{
    // Each block handles one row
    int row = blockIdx.x;
    if (row >= rows) return;
    
    // Pointer to the start of this row
    __nv_bfloat16* row_ptr = data + static_cast<size_t>(row) * cols;
    
    // Total number of vector elements per row
    int vec_count = cols / VEC_SIZE;
    
    // Shared memory for reduction
    extern __shared__ float sdata[];
    
    // Each thread accumulates sum of squares for its assigned elements
    float local_sum_sq = 0.0f;
    
    // Process elements in chunks of VEC_SIZE
    for (int idx = threadIdx.x; idx < vec_count; idx += blockDim.x) {
        // Load VEC_SIZE BF16 values as a single uint32_t (for VEC_SIZE=2) or larger
        // For VEC_SIZE=2: two BF16 fit in one uint16_t pair -> use uint32_t load
        // We'll handle VEC_SIZE=2 specifically for simplicity and efficiency
        
        if constexpr (VEC_SIZE == 2) {
            // Load two BF16 values at once
            uint32_t packed = reinterpret_cast<const uint32_t*>(row_ptr)[idx];
            __nv_bfloat16 v0 = __ushort_as_bfloat16(static_cast<uint16_t>(packed & 0xFFFF));
            __nv_bfloat16 v1 = __ushort_as_bfloat16(static_cast<uint16_t>((packed >> 16) & 0xFFFF));
            
            float f0 = bf16_to_float(v0);
            float f1 = bf16_to_float(v1);
            
            local_sum_sq += f0 * f0 + f1 * f1;
        } else if constexpr (VEC_SIZE == 4) {
            // Load four BF16 values (8 bytes)
            uint2 packed = reinterpret_cast<const uint2*>(row_ptr)[idx];
            __nv_bfloat16 v0 = __ushort_as_bfloat16(static_cast<uint16_t>(packed.x & 0xFFFF));
            __nv_bfloat16 v1 = __ushort_as_bfloat16(static_cast<uint16_t>((packed.x >> 16) & 0xFFFF));
            __nv_bfloat16 v2 = __ushort_as_bfloat16(static_cast<uint16_t>(packed.y & 0xFFFF));
            __nv_bfloat16 v3 = __ushort_as_bfloat16(static_cast<uint16_t>((packed.y >> 16) & 0xFFFF));
            
            float f0 = bf16_to_float(v0);
            float f1 = bf16_to_float(v1);
            float f2 = bf16_to_float(v2);
            float f3 = bf16_to_float(v3);
            
            local_sum_sq += f0 * f0 + f1 * f1 + f2 * f2 + f3 * f3;
        } else {
            // Fallback: scalar processing
            for (int j = 0; j < VEC_SIZE; ++j) {
                int elem_idx = idx * VEC_SIZE + j;
                if (elem_idx < cols) {
                    float f = bf16_to_float(row_ptr[elem_idx]);
                    local_sum_sq += f * f;
                }
            }
        }
    }
    
    // Block-level reduction to compute total sum of squares
    sdata[threadIdx.x] = local_sum_sq;
    __syncthreads();
    
    // Tree reduction
    for (int stride = blockDim.x / 2; stride > 0; stride >>= 1) {
        if (threadIdx.x < stride) {
            sdata[threadIdx.x] += sdata[threadIdx.x + stride];
        }
        __syncthreads();
    }
    
    // All threads read the final result
    float total_sum_sq = sdata[0];
    float norm = sqrtf(total_sum_sq);
    
    // Handle zero-norm case
    if (norm == 0.0f) {
        norm = 1.0f; // Avoid division by zero; leave values unchanged effectively
    }
    
    // Second pass: divide each element by norm and store back
    for (int idx = threadIdx.x; idx < vec_count; idx += blockDim.x) {
        if constexpr (VEC_SIZE == 2) {
            uint32_t packed = reinterpret_cast<const uint32_t*>(row_ptr)[idx];
            __nv_bfloat16 v0 = __ushort_as_bfloat16(static_cast<uint16_t>(packed & 0xFFFF));
            __nv_bfloat16 v1 = __ushort_as_bfloat16(static_cast<uint16_t>((packed >> 16) & 0xFFFF));
            
            float f0 = bf16_to_float(v0) / norm;
            float f1 = bf16_to_float(v1) / norm;
            
            uint32_t out_packed = static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f0)))) |
                                  (static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f1)))) << 16);
            reinterpret_cast<uint32_t*>(row_ptr)[idx] = out_packed;
        } else if constexpr (VEC_SIZE == 4) {
            uint2 packed = reinterpret_cast<const uint2*>(row_ptr)[idx];
            __nv_bfloat16 v0 = __ushort_as_bfloat16(static_cast<uint16_t>(packed.x & 0xFFFF));
            __nv_bfloat16 v1 = __ushort_as_bfloat16(static_cast<uint16_t>((packed.x >> 16) & 0xFFFF));
            __nv_bfloat16 v2 = __ushort_as_bfloat16(static_cast<uint16_t>(packed.y & 0xFFFF));
            __nv_bfloat16 v3 = __ushort_as_bfloat16(static_cast<uint16_t>((packed.y >> 16) & 0xFFFF));
            
            float f0 = bf16_to_float(v0) / norm;
            float f1 = bf16_to_float(v1) / norm;
            float f2 = bf16_to_float(v2) / norm;
            float f3 = bf16_to_float(v3) / norm;
            
            uint32_t x_out = static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f0)))) |
                             (static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f1)))) << 16);
            uint32_t y_out = static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f2)))) |
                             (static_cast<uint32_t>(static_cast<uint16_t>(__bfloat16_as_ushort(float_to_bf16(f3)))) << 16);
            uint2 out_packed = make_uint2(x_out, y_out);
            reinterpret_cast<uint2*>(row_ptr)[idx] = out_packed;
        } else {
            for (int j = 0; j < VEC_SIZE; ++j) {
                int elem_idx = idx * VEC_SIZE + j;
                if (elem_idx < cols) {
                    float f = bf16_to_float(row_ptr[elem_idx]) / norm;
                    row_ptr[elem_idx] = float_to_bf16(f);
                }
            }
        }
    }
}

// Host-side reference implementation using FP32 for numerical stability
void reference_normalize_fp32(
    const std::vector<float>& input,
    std::vector<float>& output,
    int rows,
    int cols)
{
    output.resize(static_cast<size_t>(rows) * cols);
    
    for (int r = 0; r < rows; ++r) {
        const float* row_in = input.data() + static_cast<size_t>(r) * cols;
        float* row_out = output.data() + static_cast<size_t>(r) * cols;
        
        // Compute L2 norm in FP32
        float sum_sq = 0.0f;
        for (int c = 0; c < cols; ++c) {
            sum_sq += row_in[c] * row_in[c];
        }
        float norm = std::sqrt(sum_sq);
        if (norm == 0.0f) norm = 1.0f;
        
        // Normalize
        for (int c = 0; c < cols; ++c) {
            row_out[c] = row_in[c] / norm;
        }
    }
}

// Generate deterministic input data
void generate_input(
    std::vector<__nv_bfloat16>& bf16_data,
    std::vector<float>& fp32_data,
    int rows,
    int cols,
    unsigned seed)
{
    size_t total = static_cast<size_t>(rows) * cols;
    bf16_data.resize(total);
    fp32_data.resize(total);
    
    std::mt19937 rng(seed);
    std::uniform_real_distribution<float> dist(-1.0f, 1.0f);
    
    for (size_t i = 0; i < total; ++i) {
        float val = dist(rng);
        fp32_data[i] = val;
        bf16_data[i] = float_to_bf16(val);
    }
}

// Verify results with tolerance-based comparison
bool verify_results(
    const std::vector<__nv_bfloat16>& gpu_result,
    const std::vector<float>& ref_result,
    int rows,
    int cols,
    float tolerance,
    bool verbose)
{
    size_t total = static_cast<size_t>(rows) * cols;
    int max_err_idx = -1;
    float max_err = 0.0f;
    int mismatch_count = 0;
    
    for (size_t i = 0; i < total; ++i) {
        float gpu_val = bf16_to_float(gpu_result[i]);
        float ref_val = ref_result[i];
        float diff = std::fabs(gpu_val - ref_val);
        
        if (diff > max_err) {
            max_err = diff;
            max_err_idx = static_cast<int>(i);
        }
        
        // Relative error check
        float denom = std::max(std::fabs(ref_val), 1e-8f);
        float rel_err = diff / denom;
        
        if (rel_err > tolerance) {
            mismatch_count++;
            if (verbose && mismatch_count <= 10) {
                int r = static_cast<int>(i / cols);
                int c = static_cast<int>(i % cols);
                printf("  Mismatch at [%d,%d]: GPU=%.6f, Ref=%.6f, diff=%.6e, rel_err=%.6e\n",
                       r, c, gpu_val, ref_val, diff
