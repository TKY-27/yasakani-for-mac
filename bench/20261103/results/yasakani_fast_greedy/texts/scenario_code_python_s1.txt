Below is a complete, runnable Python 3.11 package. Every file is in its own fenced block with its path.

**Design notes (so the code is self-explanatory):**
- **Determinism:** the runner is driven by a single `asyncio` event loop; the "GPU" is a simulated device whose output is a pure function of the job's `seed` (no wall-clock, no randomness).
- **One active GPU job:** a single `asyncio.Semaphore(1)` plus a `gpu_busy` flag persisted in the state file.
- **Atomic state:** state is written to a temp file in the same directory and `os.replace`d (atomic on POSIX).
- **Resume:** on start, jobs already `completed`/`failed` are skipped; a `running` job is re-queued (crash recovery).
- **Structured JSONL events:** every lifecycle transition is emitted as one JSON object per line.
- **SIGINT:** a signal handler sets a cancellation flag; the active job is cancelled and the state is flushed.
- **Secret redaction:** configured secret values are scrubbed from every emitted event.

```python
"""Configuration loading and validation."""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


@dataclass
class Config:
    """Runtime configuration for the runner.

    ``secrets`` is a mapping of a label to the literal secret value that must
    be redacted from every emitted event.
    """

    state_path: Path
    events_path: Path
    manifest_path: Path
    secrets: dict[str, str] = field(default_factory=dict)
    gpu_device: str = "gpu0"
    gpu_concurrency: int = 1
    max_retries: int = 0
    retry_backoff: float = 0.0
    tick: float = 0.01
    flush_every: int = 1

    def __post_init__(self) -> None:
        if self.gpu_concurrency < 1:
            raise ValueError("gpu_concurrency must be >= 1")
        if self.max_retries < 0:
            raise ValueError("max_retries must be >= 0")
        if self.tick <= 0:
            raise ValueError("tick must be > 0")
        if self.flush_every < 1:
            raise ValueError("flush_every must be >= 1")


def load_config(path: str | Path) -> Config:
    """Load a JSON configuration file into a :class:`Config`.

    The file may contain a flat object or an object with a ``"config"`` key.
    """
    p = Path(path)
    raw: dict[str, Any] = json.loads(p.read_text(encoding="utf-8"))
    if isinstance(raw, dict) and "config" in raw and isinstance(raw["config"], dict):
        raw = raw["config"]

    secrets = raw.get("secrets", {})
    if not isinstance(secrets, dict):
        raise ValueError("config 'secrets' must be an object")
    secrets = {str(k): str(v) for k, v in secrets.items()}

    return Config(
        state_path=Path(raw["state_path"]),
        events_path=Path(raw["events_path"]),
        manifest_path=Path(raw["manifest_path"]),
        secrets=secrets,
        gpu_device=str(raw.get("gpu_device", "gpu0")),
        gpu_concurrency=int(raw.get("gpu_concurrency", 1)),
        max_retries=int(raw.get("max_retries", 0)),
        retry_backoff=float(raw.get("retry_backoff", 0.0)),
        tick=float(raw.get("tick", 0.01)),
        flush_every=int(raw.get("flush_every", 1)),
    )
```

### `jobrunner/manifest.py`

```python
"""Manifest (job list) parsing and validation."""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


@dataclass
class Job:
    """A single unit of GPU work.

    ``seed`` drives the deterministic output; ``duration`` is the simulated
    number of ticks the job occupies the GPU for; ``should_fail`` forces a
    failure (used by tests); ``metadata`` is free-form and is redacted on
    emission.
    """

    id: str
    seed: int
    duration: float = 1.0
    should_fail: bool = False
    metadata: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "seed": self.seed,
            "duration": self.duration,
            "should_fail": self.should_fail,
            "metadata": self.metadata,
        }


@dataclass
class Manifest:
    """An ordered collection of jobs."""

    jobs: list[Job] = field(default_factory=list)

    def by_id(self, job_id: str) -> Job:
        for job in self.jobs:
            if job.id == job_id:
                return job
        raise KeyError(f"unknown job id: {job_id!r}")


def _parse_job(raw: dict[str, Any], index: int) -> Job:
    if "id" not in raw:
        raise ValueError(f"job at index {index} is missing required field 'id'")
    job_id = str(raw["id"])
    seed = raw.get("seed", 0)
    if not isinstance(seed, int):
        raise ValueError(f"job {job_id!r}: 'seed' must be an integer")
    duration = raw.get("duration", 1.0)
    if not isinstance(duration, (int, float)) or duration <= 0:
        raise ValueError(f"job {job_id!r}: 'duration' must be a positive number")
    should_fail = bool(raw.get("should_fail", False))
    metadata = raw.get("metadata", {})
    if not isinstance(metadata, dict):
        raise ValueError(f"job {job_id!r}: 'metadata' must be an object")
    return Job(id=job_id, seed=seed, duration=float(duration), should_fail=should_fail, metadata=metadata)


def load_manifest(path: str | Path) -> Manifest:
    """Load a JSON manifest file into a :class:`Manifest`.

    Accepts either a bare list of job objects or an object with a ``"jobs"``
    list. Duplicate job ids are rejected.
    """
    p = Path(path)
    raw = json.loads(p.read_text(encoding="utf-8"))
    if isinstance(raw, dict):
        raw_jobs = raw.get("jobs")
        if not isinstance(raw_jobs, list):
            raise ValueError("manifest object must contain a 'jobs' list")
    elif isinstance(raw, list):
        raw_jobs = raw
    else:
        raise ValueError("manifest must be a JSON list or object with a 'jobs' list")

    jobs: list[Job] = []
    seen: set[str] = set()
    for index, raw_job in enumerate(raw_jobs):
        if not isinstance(raw_job, dict):
            raise ValueError(f"job at index {index} must be an object")
        job = _parse_job(raw_job, index)
        if job.id in seen:
            raise ValueError(f"duplicate job id: {job.id!r}")
        seen.add(job.id)
        jobs.append(job)

    return Manifest(jobs=jobs)
```

### `jobrunner/gpu.py`

```python
"""Simulated deterministic GPU device."""

from __future__ import annotations

import asyncio
from dataclasses import dataclass
from typing import Any


@dataclass
class GpuResult:
    """Deterministic output of a job."""

    job_id: str
    seed: int
    output: dict[str, Any]


class GpuDevice:
    """A simulated GPU that produces deterministic results.

    The result is a pure function of the job's ``seed`` and ``duration``; no
    wall-clock time or randomness is involved, so repeated runs are identical.
    """

    def __init__(self, name: str = "gpu0", tick: float = 0.01) -> None:
        self.name = name
        self.tick = tick

    def compute(self, job_id: str, seed: int, duration: float) -> GpuResult:
        # Deterministic pseudo-output derived from the seed.
        checksum = (seed * 2654435761) % (2**32)
        output = {
            "checksum": checksum,
            "seed": seed,
            "duration": duration,
            "device": self.name,
            "tokens": [seed % 256, (seed * 7) % 256, (seed * 13) % 256],
        }
        return GpuResult(job_id=job_id, seed=seed, output=output)

    async def run(self, job_id: str, seed: int, duration: float) -> GpuResult:
        """Simulate occupying the GPU for ``duration`` seconds.

        The wait is interruptible (cancellation-safe) and is split into small
        ticks so that progress events can be emitted and cancellation is
        responsive.
        """
        steps = max(1, int(round(duration / self.tick)))
        for _ in range(steps):
            await asyncio.sleep(self.tick)
        return self.compute(job_id, seed, duration)
```

### `jobrunner/events.py`

```python
"""Structured JSONL event emission and secret redaction."""

from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Protocol


class EventSink(Protocol):
    """Anything that can accept a single event."""

    def emit(self, event: "Event") -> None: ...


@dataclass
class Event:
    """A single structured event.

    ``data`` is the payload; ``seq`` is a monotonically increasing sequence
    number assigned by the sink.
    """

    type: str
    data: dict[str, Any]
    seq: int = 0

    def to_dict(self) -> dict[str, Any]:
        return {"seq": self.seq, "type": self.type, **self.data}


class Redactor:
    """Recursively redacts configured secret values from any structure."""

    def __init__(self, secrets: dict[str, str]) -> None:
        self._secrets = {str(v) for v in secrets.values() if str(v)}

    def redact(self, value: Any) -> Any:
        if isinstance(value, str):
            return self._redact_str(value)
        if isinstance(value, dict):
            return {k: self.redact(v) for k, v in value.items()}
        if isinstance(value, (list, tuple)):
            return [self.redact(v) for v in value]
        return value

    def _redact_str(self, s: str) -> str:
        for secret in self._secrets:
            if secret and secret in s:
                s = s.replace(secret, "***")
        return s


class JsonlEventSink:
    """Writes one JSON object per line to a file, redacting secrets.

    Each line is flushed immediately so events are durable even if the process
    is killed.
    """

    def __init__(self, path: str | Path, redactor: Redactor) -> None:
        self._path = Path(path)
        self._redactor = redactor
        self._seq = 0
        self._file = open(self._path, "a", encoding="utf-8")

    def emit(self, event: Event) -> None:
        self._seq += 1
        event.seq = self._seq
        payload = self._redactor.redact(event.to_dict())
        line = json.dumps(payload, sort_keys=True, separators=(",", ":"))
        self._file.write(line + "\n")
        self._file.flush()

    def close(self) -> None:
        if not self._file.closed:
            self._file.close()

    def __enter__(self) -> "JsonlEventSink":
        return self

    def __exit__(self, exc_type, exc, tb) -> None:
        self.close()
```

### `jobrunner/state.py`

```python
"""Atomic, resumable state persistence."""

from __future__ import annotations

import json
import os
import tempfile
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


@dataclass
class State:
    """Snapshot of runner state.

    ``jobs`` maps job id -> status (``pending``/``running``/``completed``/
    ``failed``/``cancelled``). ``results`` maps job id -> deterministic output.
    ``gpu_busy`` records whether a job was mid-flight (crash recovery).
    """

    version: int = 1
    gpu_busy: bool = False
    jobs: dict[str, str] = field(default_factory=dict)
    results: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return {
            "version": self.version,
            "gpu_busy": self.gpu_busy,
            "jobs": self.jobs,
            "results": self.results,
        }

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "State":
        return cls(
            version=int(data.get("version", 1)),
            gpu_busy=bool(data.get("gpu_busy", False)),
            jobs=dict(data.get("jobs", {})),
            results=dict(data.get("results", {})),
        )


class StateStore:
    """Persists :class:`State` atomically to a JSON file.

    Writes go to a temporary file in the same directory and are moved into
    place with :func:`os.replace`, which is atomic on POSIX filesystems.
    """

    def __init__(self, path: str | Path) -> None:
        self._path = Path(path)
        self._path.parent.mkdir(parents=True, exist_ok=True)

    @property
    def path(self) -> Path:
        return self._path

    def load(self) -> State:
        if not self._path.exists():
            return State()
        raw = json.loads(self._path.read_text(encoding="utf-8"))
        return State.from_dict(raw)

    def save(self, state: State) -> None:
        data = state.to_dict()
        payload = json.dumps(data, sort_keys=True, indent=2)
        fd, tmp = tempfile.mkstemp(dir=str(self._path.parent), prefix=".state-", suffix=".tmp")
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                fh.write(payload)
                fh.flush()
                os.fsync(fh.fileno())
            os.replace(tmp, self._path)
        except BaseException:
            if os.path.exists(tmp):
                os.unlink(tmp)
            raise
```
