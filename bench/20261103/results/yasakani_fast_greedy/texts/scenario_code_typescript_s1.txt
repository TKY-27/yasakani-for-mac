# Request Metrics Service

Below is a complete, self-contained TypeScript 5 service. It uses **only Node built-ins** (`node:http`, `node:fs`, `node:stream`) — no third-party runtime dependencies. The test suite uses Node's built-in `node:test` runner, so `npm test` works with zero installs.

## `src/types.ts`

Strict domain types, the wire record, and the API response shapes.

```ts
/**
 * Domain types for the request-metrics service.
 *
 * A "record" is one newline-delimited request-metric line. A "run" is a
 * collection of records sharing the same runId. Aggregation is grouped by
 * (model, fixture) within a run.
 */

/** A single validated request-metric record. */
export interface RequestMetricRecord {
  /** Unique identifier for the run this record belongs to. */
  runId: string;
  /** Model identifier (e.g. "gpt-4", "llama-3-70b"). */
  model: string;
  /** Fixture / dataset identifier (e.g. "sft-v2", "eval-holdout"). */
  fixture: string;
  /** Latency in milliseconds (non-negative finite number). */
  latencyMs: number;
  /** Token count (non-negative integer). */
  tokens: number;
  /** HTTP status code (integer 100..599). */
  statusCode: number;
  /** Unix epoch milliseconds (integer). */
  timestampMs: number;
  /** Optional free-form error message (present when the request failed). */
  error?: string;
}

/** A single (model, fixture) aggregation group within a run. */
export interface GroupStats {
  model: string;
  fixture: string;
  count: number;
  mean: number;
  median: number;
  /** Sample standard deviation (n-1). 0 when count < 2. */
  sampleStdDev: number;
  min: number;
  max: number;
  /** Sum of tokens across the group. */
  totalTokens: number;
  /** Count of records whose statusCode >= 400. */
  errorCount: number;
}

/** Aggregated statistics for a single run. */
export interface RunSummary {
  runId: string;
  /** Total number of records in the run. */
  recordCount: number;
  /** Distinct (model, fixture) groups. */
  groups: GroupStats[];
  /** Sum of tokens across all records. */
  totalTokens: number;
  /** Count of records whose statusCode >= 400. */
  errorCount: number;
  /** Earliest record timestamp (ms) or null if empty. */
  firstTimestampMs: number | null;
  /** Latest record timestamp (ms) or null if empty. */
  lastTimestampMs: number | null;
}

/** The /summary response body. */
export interface SummaryResponse {
  service: string;
  version: string;
  /** Total records ingested across all runs. */
  totalRecords: number;
  /** Distinct run ids observed. */
  runCount: number;
  /** Distinct models observed. */
  modelCount: number;
  /** Distinct fixtures observed. */
  fixtureCount: number;
  /** Per-run summaries, sorted by runId. */
  runs: RunSummary[];
}

/** The /health response body. */
export interface HealthResponse {
  status: 'ok';
  service: string;
  version: string;
  uptimeSec: number;
  totalRecords: number;
  runCount: number;
  memory: {
    rssBytes: number;
    heapUsedBytes: number;
    heapTotalBytes: number;
  };
}

/** The /runs/:id response body. */
export interface RunResponse {
  runId: string;
  recordCount: number;
  groups: GroupStats[];
  totalTokens: number;
  errorCount: number;
  firstTimestampMs: number | null;
  lastTimestampMs: number | null;
}

/** A malformed-input error carries a machine-readable code and line context. */
export interface MalformedInputError {
  code: 'MALFORMED_INPUT';
  message: string;
  line: number;
  field?: string;
  value?: string;
}

/** A run that has not been observed yet. */
export interface NotFoundError {
  code: 'NOT_FOUND';
  message: string;
  runId: string;
}

/** A request body that exceeds the configured byte limit. */
export interface PayloadTooLargeError {
  code: 'PAYLOAD_TOO_LARGE';
  message: string;
  limitBytes: number;
}

## `src/validate.ts`

Hand-rolled validation (no schema libraries). Produces a `RequestMetricRecord` or a `MalformedInputError`.

```ts
import type { MalformedInputError, RequestMetricRecord } from './types.js';

/** Maximum length for string fields (runId, model, fixture, error). */
const MAX_STRING_LEN = 256;
/** Maximum length for the error message field. */
const MAX_ERROR_LEN = 1024;
/** Maximum magnitude for numeric fields (guards against absurd values). */
const MAX_NUMERIC = 1e15;

/**
 * Validate a single newline-delimited record (a JSON object).
 *
 * Returns the parsed record on success, or a MalformedInputError-shaped
 * object on failure. This is intentionally synchronous and pure so it can
 * be unit-tested without I/O.
 */
export function validateRecord(
  raw: string,
  line: number
): RequestMetricRecord | MalformedInputError {
  const fail = (field: string, message: string, value?: string): MalformedInputError => ({
    code: 'MALFORMED_INPUT',
    message,
    line,
    field,
    value,
  });

  // 1) Must be valid JSON.
  let obj: unknown;
  try {
    obj = JSON.parse(raw);
  } catch {
    return fail('json', `line ${line}: not valid JSON`);
  }

  // 2) Must be a plain object (not null, array, or primitive).
  if (typeof obj !== 'object' || obj === null || Array.isArray(obj)) {
    return fail('json', `line ${line}: expected a JSON object`);
  }
  const o = obj as Record<string, unknown>;

  // 3) Required string fields.
  const runId = o['runId'];
  if (typeof runId !== 'string' || runId.length === 0) {
    return fail('runId', `line ${line}: "runId" must be a non-empty string`, typeof runId);
  }
  if (runId.length > MAX_STRING_LEN) {
    return fail('runId', `line ${line}: "runId" exceeds ${MAX_STRING_LEN} characters`);
  }

  const model = o['model'];
  if (typeof model !== 'string' || model.length === 0) {
    return fail('model', `line ${line}: "model" must be a non-empty string`, typeof model);
  }
  if (model.length > MAX_STRING_LEN) {
    return fail('model', `line ${line}: "model" exceeds ${MAX_STRING_LEN} characters`);
  }

  const fixture = o['fixture'];
  if (typeof fixture !== 'string' || fixture.length === 0) {
    return fail('fixture', `line ${line}: "fixture" must be a non-empty string`, typeof fixture);
  }
  if (fixture.length > MAX_STRING_LEN) {
    return fail('fixture', `line ${line}: "fixture" exceeds ${MAX_STRING_LEN} characters`);
  }

  // 4) Numeric fields.
  const latencyMs = o['latencyMs'];
  if (typeof latencyMs !== 'number' || !Number.isFinite(latencyMs) || latencyMs < 0) {
    return fail('latencyMs', `line ${line}: "latencyMs" must be a non-negative finite number`, String(latencyMs));
  }
  if (latencyMs > MAX_NUMERIC) {
    return fail('latencyMs', `line ${line}: "latencyMs" exceeds ${MAX_NUMERIC}`);
  }

  const tokens = o['tokens'];
  if (typeof tokens !== 'number' || !Number.isInteger(tokens) || tokens < 0) {
    return fail('tokens', `line ${line}: "tokens" must be a non-negative integer`, String(tokens));
  }
  if (tokens > MAX_NUMERIC) {
    return fail('tokens', `line ${line}: "tokens" exceeds ${MAX_NUMERIC}`);
  }

  const statusCode = o['statusCode'];
  if (typeof statusCode !== 'number' || !Number.isInteger(statusCode) || statusCode < 100 || statusCode > 599) {
    return fail('statusCode', `line ${line}: "statusCode" must be an integer in [100, 599]`, String(statusCode));
  }

  const timestampMs = o['timestampMs'];
  if (typeof timestampMs !== 'number' || !Number.isInteger(timestampMs) || timestampMs < 0) {
    return fail('timestampMs', `line ${line}: "timestampMs" must be a non-negative integer (epoch ms)`, String(timestampMs));
  }
  if (timestampMs > MAX_NUMERIC) {
    return fail('timestampMs', `line ${line}: "timestampMs" exceeds ${MAX_NUMERIC}`);
  }

  // 5) Optional error field (string, bounded).
  let error: string | undefined;
  if (o['error'] !== undefined) {
    if (typeof o['error'] !== 'string') {
      return fail('error', `line ${line}: "error" must be a string when present`, typeof o['error']);
    }
    if ((o['error'] as string).length > MAX_ERROR_LEN) {
      return fail('error', `line ${line}: "error" exceeds ${MAX_ERROR_LEN} characters`);
    }
    error = o['error'] as string;
  }

  return {
    runId,
    model,
    fixture,
    latencyMs,
    tokens,
    statusCode,
    timestampMs,
    ...(error !== undefined ? { error } : {}),
  };
}
```

## `src/aggregate.ts`

Incremental aggregation with bounded memory. Each `(run, model, fixture)` group keeps only running sums (mean, sample variance via Welford), a bounded reservoir for the median, and min/max.

```ts
import type { GroupStats, RequestMetricRecord, RunSummary } from './types.js';

/**
 * Bounded-memory incremental aggregator.
 *
 * For each (runId, model, fixture) group we keep:
 *   - count, sum, sumSq  -> mean and sample variance (Welford)
 *   - min, max
 *   - totalTokens, errorCount
 *   - a reservoir sample (capacity `reservoirSize`) for the median
 *
 * The reservoir is a fixed-size array, so per-group memory is O(1) with a
 * constant bound. The number of groups is bounded by the number of distinct
 * (model, fixture) pairs per run, which is naturally small.
 */
export class Aggregator {
  private readonly groups = new Map<string, GroupState>();
  private readonly runMeta = new Map<string, RunMeta>();
  private readonly reservoirSize: number;
  private totalRecords = 0;

  constructor(reservoirSize = 1024) {
    if (!Number.isInteger(reservoirSize) || reservoirSize < 1) {
      throw new Error('reservoirSize must be a positive integer');
    }
    this.reservoirSize = reservoirSize;
  }

  /** Ingest one validated record. Returns the group key for testing. */
  ingest(rec: RequestMetricRecord): string {
    const key = groupKey(rec.runId, rec.model, rec.fixture);
    let g = this.groups.get(key);
    if (g === undefined) {
      g = {
        runId: rec.runId,
        model: rec.model,
        fixture: rec.fixture,
        count: 0,
        sum: 0,
        sumSq: 0,
        min: Number.POSITIVE_INFINITY,
        max: Number.NEGATIVE_INFINITY,
        totalTokens: 0,
        errorCount: 0,
        reservoir: [],
      };
      this.groups.set(key, g);
    }
    this.updateGroup(g, rec);

    let meta = this.runMeta.get(rec.runId);
    if (meta === undefined) {
      meta = { first: rec.timestampMs, last: rec.timestampMs };
      this.runMeta.set(rec.runId, meta);
    } else {
      if (rec.timestampMs < meta.first) meta.first = rec.timestampMs;
      if (rec.timestampMs > meta.last) meta.last = rec.timestampMs;
    }

    this.totalRecords += 1;
    return key;
  }

  private updateGroup(g: GroupState, rec: RequestMetricRecord): void {
    const x = rec.latencyMs;
    g.count += 1;
    g.sum += x;
    g.sumSq += x * x;
    if (x < g.min) g.min = x;
    if (x > g.max) g.max = x;
    g.totalTokens += rec.tokens;
    if (rec.statusCode >= 400) g.errorCount += 1;
    this.reservoirAdd(g, x);
  }

  /**
   * Reservoir sampling (Algorithm R). Keeps a fixed-capacity sample that is
   * statistically representative of the stream, enabling an approximate
   * median without storing every value.
   */
  private reservoirAdd(g: GroupState, value: number): void {
    const r = g.reservoir;
    if (r.length < this.reservoirSize) {
      r.push(value);
      return;
    }
    // Replace a random slot with probability 1/count.
    const j = Math.floor(Math.random() * g.count);
    if (j < this.reservoirSize) {
      r[j] = value;
    }
  }

  /** Number of distinct runs observed. */
  get runCount(): number {
    return this.runMeta.size;
  }

  /** Total records ingested. */
  get totalRecordsCount(): number {
    return this.totalRecords;
  }

  /** Distinct models observed across all runs. */
  distinctModels(): Set<string> {
    const s = new Set<string>();
    for (const g of this.groups.values()) s.add(g.model);
    return s;
  }

  /** Distinct fixtures observed across all runs. */
  distinctFixtures(): Set<string> {
    const s = new Set<string>();
    for (const g of this.groups.values()) s.add(g.fixture);
    return s;
  }

  /** Whether a run has been observed. */
  hasRun(runId: string): boolean {
    return this.runMeta.has(runId);
  }

  /** Compute the summary for a single run. Returns null if unknown. */
  runSummary(runId: string): RunSummary | null {
    const meta = this.runMeta.get(runId);
    if (meta === undefined) return null;

    const groups: GroupStats[] = [];
    let totalTokens = 0;
    let errorCount = 0;
    let recordCount = 0;

    for (const g of this.groups.values()) {
      if (g.runId !== runId) continue;
      recordCount += g.count;
      totalTokens += g.totalTokens;
      errorCount += g.errorCount;
      groups.push(this.toGroupStats(g));
    }

    groups.sort((a, b) =>
      a.model === b.model
        ? a.fixture.localeCompare(b.fixture)
        : a.model.localeCompare(b.model
