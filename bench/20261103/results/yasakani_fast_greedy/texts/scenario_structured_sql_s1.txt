CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('synthetic', 'real_world', 'adversarial', 'multilingual')),
    description TEXT NOT NULL,
    avg_token_count INT NOT NULL CHECK (avg_token_count > 0)
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMPTZ,
    status TEXT NOT NULL DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'running')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL,
    tokens_per_second NUMERIC(10, 2) NOT NULL CHECK (tokens_per_second > 0),
    latency_ms NUMERIC(10, 2) NOT NULL CHECK (latency_ms > 0),
    mtp_acceptance_rate NUMERIC(4, 3) CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    long_context_retention_score NUMERIC(4, 3) CHECK (long_context_retention_score BETWEEN 0 AND 1),
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_tps ON request_metrics(tokens_per_second);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_acceptance_rate);

CREATE VIEW v_model_fixture_summary AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.tokens_per_second) AS avg_tps,
    AVG(rm.latency_ms) AS avg_latency_ms
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name, f.kind;

CREATE VIEW v_mtp_analysis AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance,
    STDDEV(rm.mtp_acceptance_rate) AS mtp_stddev,
    MIN(rm.mtp_acceptance_rate) AS min_mtp,
    MAX(rm.mtp_acceptance_rate) AS max_mtp
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_tps NUMERIC,
    p_latency NUMERIC,
    p_mtp NUMERIC,
    p_retention NUMERIC
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, tokens_per_second, latency_ms, mtp_acceptance_rate, long_context_retention_score)
    VALUES (p_run_id, p_request_index, p_tps, p_latency, p_mtp, p_retention)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        tokens_per_second = EXCLUDED.tokens_per_second,
        latency_ms = EXCLUDED.latency_ms,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        long_context_retention_score = EXCLUDED.long_context_retention_score;
END;
$$ LANGUAGE plpgsql;

    m.name AS model_name,
    f.kind AS fixture_kind,
    AVG(rm.tokens_per_second) AS avg_throughput_tps,
    STDDEV(rm.tokens_per_second) AS throughput_stddev,
    COUNT(*) AS sample_count
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name, f.kind
ORDER BY m.name, f.kind;
