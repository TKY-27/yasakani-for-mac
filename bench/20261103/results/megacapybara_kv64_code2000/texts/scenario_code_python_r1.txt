Below is a complete, runnable Python 3.11 package named `deterministic_job_runner`.

Design notes (so the code makes sense):
- **Determinism**: jobs are processed in manifest order; each job's "work" is a pure function of its inputs (no wall-clock or randomness), so identical manifests produce identical outputs and event streams.
- **One active GPU job**: a single worker coroutine + an `asyncio.Lock` guarantee at most one job runs at a time.
- **Atomic state**: state is written to a temp file then `os.replace()`'d into place.
- **Resume**: on start, completed/failed/cancelled jobs are skipped and their results restored from persisted state.
- **JSONL events**: structured events streamed to stdout (and optionally a file).
- **SIGINT**: first signal cancels the active job (graceful); second signal exits immediately.
- **Secret redaction**: configured secret values are scrubbed from all emitted text.

### README.md

```markdown
# Deterministic Job Runner

A deterministic asynchronous job runner for Python 3.11.

## Features
- Reads a JSON manifest describing jobs.
- Enforces exactly one active GPU job at a time.
- Persists atomic state (temp file + rename) so crashes don't corrupt progress.
- Resumes previously completed work.
- Streams structured JSONL events to stdout (and optionally a file).
- Handles SIGINT: first press cancels the active job gracefully; second exits.
- Redacts configured secrets from all emitted output.

## Usage
    python -m deterministic_job_runner.cli --manifest manifest.json \
        --state-dir ./state --events-file events.jsonl

## Manifest format
See `examples/manifest.json`.
```

"""Data models for the job runner."""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from enum import Enum
from typing import Any


class JobStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"


@dataclass
class Job:
    id: str
    type: str = "gpu"
    params: dict[str, Any] = field(default_factory=dict)
    status: JobStatus = JobStatus.PENDING
    result: Any = None
    error: str | None = None
    attempts: int = 0

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "type": self.type,
            "params": self.params,
            "status": self.status.value,
            "result": self.result,
            "error": self.error,
            "attempts": self.attempts,
        }

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "Job":
        return cls(
            id=data["id"],
            type=data.get("type", "gpu"),
            params=data.get("params", {}),
            status=JobStatus(data.get("status", "pending")),
            result=data.get("result"),
            error=data.get("error"),
            attempts=data.get("attempts", 0),
        )


@dataclass
class Manifest:
    jobs: list[Job]

    @classmethod
    def from_json(cls, path: str) -> "Manifest":
        with open(path, "r", encoding="utf-8") as f:
            raw = json.load(f)
        jobs = [Job.from_dict(j) for j in raw.get("jobs", [])]
        return cls(jobs=jobs)

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "Manifest":
        jobs = [Job.from_dict(j) for j in data.get("jobs", [])]
        return cls(jobs=jobs)

"""Secret redaction utilities."""

from __future__ import annotations

import re
from typing import Iterable


class Redactor:
    """Redacts configured secret values from strings and JSON-serializable objects."""

    def __init__(self, secrets: Iterable[str] | None = None) -> None:
        # Store secrets sorted by length descending so longer matches win.
        self._secrets: list[str] = sorted(
            (s for s in (secrets or []) if s), key=len, reverse=True
        )
        # Build a combined regex for fast bulk replacement when possible.
        escaped = [re.escape(s) for s in self._secrets]
        self._pattern = re.compile("|".join(escaped)) if escaped else None

    def redact_string(self, text: str) -> str:
        if not self._secrets:
            return text
        if self._pattern is None:
            return text
        return self._pattern.sub("[REDACTED]", text)

    def redact_object(self, obj: object) -> object:
        """Recursively redact secrets in a JSON-serializable structure."""
        if isinstance(obj, str):
            return self.redact_string(obj)
        if isinstance(obj, dict):
            return {k: self.redact_object(v) for k, v in obj.items()}
        if isinstance(obj, (list, tuple)):
            return [self.redact_object(item) for item in obj]
        return obj

    def has_secrets(self) -> bool:
        return len(self._secrets) > 0
