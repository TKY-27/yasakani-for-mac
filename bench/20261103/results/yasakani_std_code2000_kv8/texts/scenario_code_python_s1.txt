Below is a complete, runnable Python 3.11 package. It is designed so that the "GPU" is simulated (a deterministic, time‑based workload) so the whole thing runs and is testable on any machine, while still enforcing a single active GPU job, atomic state persistence, resume, JSONL event streaming, SIGINT handling, and secret redaction.

**Layout**

```
gpu_runner/
├── pyproject.toml
├── gpu_runner/
│   ├── __init__.py
│   ├── __main__.py
│   ├── manifest.py
│   ├── state.py
│   ├── events.py
│   ├── redact.py
│   ├── runner.py
│   └── cli.py
└── tests/
    ├── conftest.py
    ├── test_success.py
    ├── test_failure.py
    ├── test_cancellation.py
    └── test_resume.py
```

### `gpu_runner/manifest.py`

```python
"""Manifest loading and validation.

The manifest is a JSON document describing a set of GPU jobs. Each job has a
deterministic workload (a fixed number of steps) so that repeated runs produce
identical results.
"""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


class ManifestError(ValueError):
    """Raised when a manifest is malformed or invalid."""


@dataclass(frozen=True)
class Job:
    """A single GPU job.

    Attributes:
        id: Unique identifier for the job.
        name: Human readable name.
        steps: Number of deterministic steps to execute.
        gpu: Logical GPU index to run on (informational; the runner enforces
            a single active GPU job regardless).
        params: Arbitrary parameters for the job.
        secrets: Names of parameters that must be redacted in logs/events.
    """

    id: str
    name: str
    steps: int
    gpu: int = 0
    params: dict[str, Any] = field(default_factory=dict)
    secrets: list[str] = field(default_factory=list)

    def redact_params(self, redactor: "Redactor") -> dict[str, Any]:
        """Return a copy of params with configured secrets redacted."""
        out: dict[str, Any] = {}
        for key, value in self.params.items():
            if key in self.secrets:
                out[key] = redactor.redact(key, value)
            else:
                out[key] = value
        return out


@dataclass(frozen=True)
class Manifest:
    """A collection of jobs plus global settings."""

    jobs: list[Job]
    max_concurrency: int = 1
    seed: int = 0

    def job_ids(self) -> list[str]:
        return [j.id for j in self.jobs]

    def get(self, job_id: str) -> Job:
        for job in self.jobs:
            if job.id == job_id:
                return job
        raise KeyError(job_id)


def _require(obj: Any, key: str, kind: type, where: str) -> Any:
    if key not in obj:
        raise ManifestError(f"Missing required field '{key}' in {where}")
    value = obj[key]
    if not isinstance(value, kind):
        raise ManifestError(
            f"Field '{key}' in {where} must be {kind.__name__}, got {type(value).__name__}"
        )
    return value


def _parse_job(raw: dict[str, Any], index: int) -> Job:
    where = f"jobs[{index}]"
    job_id = _require(raw, "id", str, where)
    name = raw.get("name", job_id)
    if not isinstance(name, str):
        raise ManifestError(f"Field 'name' in {where} must be str")
    steps = _require(raw, "steps", int, where)
    if steps < 0:
        raise ManifestError(f"Field 'steps' in {where} must be >= 0")
    gpu = raw.get("gpu", 0)
    if not isinstance(gpu, int):
        raise ManifestError(f"Field 'gpu' in {where} must be int")
    params = raw.get("params", {})
    if not isinstance(params, dict):
        raise ManifestError(f"Field 'params' in {where} must be object")
    secrets = raw.get("secrets", [])
    if not isinstance(secrets, list) or not all(isinstance(s, str) for s in secrets):
        raise ManifestError(f"Field 'secrets' in {where} must be list of str")
    return Job(
        id=job_id,
        name=name,
        steps=steps,
        gpu=gpu,
        params=params,
        secrets=list(secrets),
    )


def load_manifest(path: str | Path) -> Manifest:
    """Load and validate a JSON manifest from disk."""
    path = Path(path)
    if not path.exists():
        raise ManifestError(f"Manifest not found: {path}")
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ManifestError(f"Invalid JSON in manifest {path}: {exc}") from exc

    if not isinstance(data, dict):
        raise ManifestError("Manifest root must be a JSON object")

    jobs_raw = _require(data, "jobs", list, "manifest")
    jobs: list[Job] = []
    seen: set[str] = set()
    for i, raw in enumerate(jobs_raw):
        if not isinstance(raw, dict):
            raise ManifestError(f"jobs[{i}] must be an object")
        job = _parse_job(raw, i)
        if job.id in seen:
            raise ManifestError(f"Duplicate job id '{job.id}'")
        seen.add(job.id)
        jobs.append(job)

    max_concurrency = data.get("max_concurrency", 1)
    if not isinstance(max_concurrency, int) or max_concurrency < 1:
        raise ManifestError("Field 'max_concurrency' must be int >= 1")

    seed = data.get("seed", 0)
    if not isinstance(seed, int):
        raise ManifestError("Field 'seed' must be int")

    return Manifest(jobs=jobs, max_concurrency=max_concurrency, seed=seed)
```
