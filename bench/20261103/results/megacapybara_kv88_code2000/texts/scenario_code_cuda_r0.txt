// RAII wrapper for device memory
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_(0) {}
    
    explicit DeviceBuffer(size_t bytes) : ptr_(nullptr), size_(bytes) {
        if (bytes > 0) {
            CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        }
    }
    
    ~DeviceBuffer() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }
    
    // Disable copy
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;
    
    // Enable move
    DeviceBuffer(DeviceBuffer&& other) noexcept : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }
    
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            if (ptr_) cudaFree(ptr_);
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }
    
    void* get() const { return ptr_; }
    size_t size() const { return size_; }
    
    template<typename T>
    T* as() const { return static_cast<T*>(ptr_); }

private:
    void* ptr_;
    size_t size_;
};

bool parse_args(int argc, char** argv, Args& args) {
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        if (arg == "-h" || arg == "--help") {
            print_usage(argv[0]);
            return false;
        } else if (arg == "-r") {
            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -r\n"); return false; }
            args.rows = atoi(argv[++i]);
            if (args.rows <= 0) { fprintf(stderr, "Rows must be positive\n"); return false; }
        } else if (arg == "-c") {
            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -c\n"); return false; }
            args.cols = atoi(argv[++i]);
            if (args.cols <= 0) { fprintf(stderr, "Cols must be positive\n"); return false; }
        } else if (arg == "-w") {
            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -w\n"); return false; }
            args.warmup_iters = atoi(argv[++i]);
            if (args.warmup_iters < 0) { fprintf(stderr, "Warmup iters must be non-negative\n"); return false; }
        } else if (arg == "-t") {
            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -t\n"); return false; }
            args.timing_iters = atoi(argv[++i]);
            if (args.timing_iters <= 0) { fprintf(stderr, "Timing iters must be positive\n"); return false; }
        } else if (arg == "-T") {
            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -T\n"); return false; }
            args.tolerance = atof(argv[++i]);

            if (i + 1 >= argc) { fprintf(stderr, "Missing value for -s\n"); return false; }
            args.seed = static_cast<unsigned int>(atoi(argv[++i]));
        } else if (arg == "-v") {
            args.verbose = true;
        } else {
            fprintf(stderr, "Unknown option: %s\n", arg.c_str());
            return false;

    }

// Convert BF16 to FP32
__device__ inline float bf16_to_float(__nv_bfloat16 b) {
    return __bfloat162float(b);
}

// Convert FP32 to BF16
__device__ inline __nv_bfloat16 float_to_bf16(float f) {
    return __float2bfloat16(f);
}
