BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    model_name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_kind VARCHAR(50) NOT NULL,
    fixture_name VARCHAR(100) NOT NULL,
    description TEXT,
    UNIQUE (fixture_kind, fixture_name)
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ended_at TIMESTAMP,
    status VARCHAR(20) NOT NULL CHECK (status IN ('completed', 'failed', 'running')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL,
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted_tokens INT NOT NULL DEFAULT 0 CHECK (mtp_accepted_tokens >= 0),
    mtp_total_proposed INT NOT NULL DEFAULT 0 CHECK (mtp_total_proposed >= 0),
    UNIQUE (run_id, request_index)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX IF NOT EXISTS idx_metrics_mtp ON request_metrics(mtp_total_proposed, mtp_accepted_tokens);

CREATE VIEW IF NOT EXISTS v_run_summary AS
SELECT 
    r.run_id,
    m.model_name,
    f.fixture_kind,
    f.fixture_name,
    COUNT(mr.metric_id) AS total_requests,
    AVG(mr.latency_ms) AS avg_latency_ms,
    SUM(mr.input_tokens) AS total_input_tokens,
    SUM(mr.output_tokens) AS total_output_tokens,
    SUM(mr.output_tokens) / NULLIF(COUNT(mr.metric_id), 0) AS avg_output_tokens_per_request,
    (SUM(mr.output_tokens) * 1000.0) / NULLIF(SUM(mr.latency_ms), 0) AS throughput_tps
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics mr ON r.run_id = mr.run_id
GROUP BY r.run_id, m.model_name, f.fixture_kind, f.fixture_name;

CREATE VIEW IF NOT EXISTS v_mtp_analysis AS
SELECT 
    r.run_id,
    m.model_name,
    f.fixture_kind,
    AVG(mr.mtp_accepted_tokens) AS avg_accepted,
    AVG(mr.mtp_total_proposed) AS avg_proposed,
    CASE WHEN SUM(mr.mtp_total_proposed) > 0 
         THEN (SUM(mr.mtp_accepted_tokens) * 100.0) / SUM(mr.mtp_total_proposed) 
         ELSE 0 END AS acceptance_rate_pct
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics mr ON r.run_id = mr.run_id
GROUP BY r.run_id, m.model_name, f.fixture_kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted INT,
    p_mtp_proposed INT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (
        run_id, request_index, input_tokens, output_tokens, 
        latency_ms, mtp_accepted_tokens, mtp_total_proposed
    ) VALUES (
        p_run_id, p_request_index, p_input_tokens, p_output_tokens,
        p_latency_ms, p_mtp_accepted, p_mtp_proposed
    )
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens,
        mtp_total_proposed = EXCLUDED.mtp_total_proposed;
END;
$$ LANGUAGE plpgsql;
