CREATE TABLE IF NOT EXISTS models (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INTEGER NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS fixtures (
    id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('synthetic', 'real_world', 'adversarial', 'multilingual')),
    description TEXT,
    avg_tokens_per_request INTEGER NOT NULL CHECK (avg_tokens_per_request > 0),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS runs (
    id SERIAL PRIMARY KEY,
    model_id INTEGER NOT NULL REFERENCES models(id) ON DELETE CASCADE,
    fixture_id INTEGER NOT NULL REFERENCES fixtures(id) ON DELETE CASCADE,
    run_date DATE NOT NULL,
    hardware TEXT NOT NULL,
    batch_size INTEGER NOT NULL CHECK (batch_size > 0),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (model_id, fixture_id, run_date, hardware, batch_size)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    id SERIAL PRIMARY KEY,
    run_id INTEGER NOT NULL REFERENCES runs(id) ON DELETE CASCADE,
    request_index INTEGER NOT NULL CHECK (request_index >= 0),
    latency_ms REAL NOT NULL CHECK (latency_ms >= 0),
    tokens_per_second REAL NOT NULL CHECK (tokens_per_second >= 0),
    mtp_acceptance_rate REAL NOT NULL CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    long_context_retention_score REAL NOT NULL CHECK (long_context_retention_score BETWEEN 0 AND 1),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX idx_metrics_tps ON request_metrics(tokens_per_second);

CREATE VIEW v_run_summary AS
SELECT 
    r.id AS run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.run_date,
    r.hardware,
    r.batch_size,
    COUNT(*) AS total_requests,
    AVG(rm.latency_ms) AS avg_latency_ms,
    AVG(rm.tokens_per_second) AS avg_tps,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance,
    AVG(rm.long_context_retention_score) AS avg_retention
FROM runs r
JOIN models m ON r.model_id = m.id
JOIN fixtures f ON r.fixture_id = f.id
JOIN request_metrics rm ON rm.run_id = r.id
GROUP BY r.id, m.name, f.kind, r.run_date, r.hardware, r.batch_size;

CREATE VIEW v_model_performance AS
SELECT 
    m.id AS model_id,
    m.name AS model_name,
    m.context_length,
    COUNT(DISTINCT r.id) AS total_runs,
    AVG(rm.tokens_per_second) AS overall_avg_tps,
    STDDEV(rm.tokens_per_second) AS tps_stddev,
    AVG(rm.mtp_acceptance_rate) AS overall_avg_mtp,
    AVG(rm.long_context_retention_score) AS overall_avg_retention
FROM models m
JOIN runs r ON m.id = r.model_id
JOIN request_metrics rm ON r.id = rm.run_id
GROUP BY m.id, m.name, m.context_length;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INTEGER,
    p_request_index INTEGER,
    p_latency_ms REAL,
    p_tps REAL,
    p_mtp_rate REAL,
    p_retention REAL
) RETURNS INTEGER AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, latency_ms, tokens_per_second, mtp_acceptance_rate, long_context_retention_score)
    VALUES (p_run_id, p_request_index, p_latency_ms, p_tps, p_mtp_rate, p_retention)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        latency_ms = EXCLUDED.latency_ms,
        tokens_per_second = EXCLUDED.tokens_per_second,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        long_context_retention_score = EXCLUDED.long_context_retention_score,
        created_at = NOW();
    RETURN (SELECT id FROM request_metrics WHERE run_id = p_run_id AND request_index = p_request_index);
END;
$$ LANGUAGE plpgsql;

SELECT 
    m.name AS model,
    f.kind AS fixture,
    AVG(rm.tokens_per_second) AS avg_throughput,
    STDDEV(rm.tokens_per_second) AS throughput_stddev
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.id
JOIN models m ON r.model_id = m.id
JOIN fixtures f ON r.fixture_id = f.id
GROUP BY m.name, f.kind
ORDER BY avg_throughput DESC;

SELECT 
    m.name AS model,
    AVG(rm.tokens_per_second) AS mean_tps,
    STDDEV(rm.tokens_per_second) AS stddev_tps,
    STDDEV(rm.tokens_per_second) / AVG(rm.tokens_per_second) AS coefficient_of_variation
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.id
JOIN models m ON r.model_id = m.id
GROUP BY m.name
ORDER BY coefficient_of_variation DESC;

SELECT 
    m.name AS model,
    f.kind AS fixture,
    AVG(rm.long_context_retention_score) AS avg_retention,
    MIN(rm.long_context_retention_score) AS min_retention,
    MAX(rm.long_context_retention_score) AS max_retention
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.id
JOIN models m ON r.model_id = m.id
JOIN fixtures f ON r.fixture_id = f.id
WHERE f.kind IN ('real_world', 'multilingual')
GROUP BY m.name, f.kind
ORDER BY avg_retention DESC;

SELECT 
    m.name AS model,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance,
    COUNT(CASE WHEN rm.mtp_acceptance_rate > 0.8 THEN 1 END) AS high_acceptance_requests,
    COUNT(CASE WHEN rm.mtp_acceptance_rate < 0.7 THEN 1 END) AS low_acceptance_requests
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.id
JOIN models m ON r.model_id = m.id
GROUP BY m.name
ORDER BY avg_mtp_acceptance DESC;
