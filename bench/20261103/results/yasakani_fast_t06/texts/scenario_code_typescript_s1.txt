Below is a complete, self-contained TypeScript 5 service. Everything is runnable — no placeholders, no third‑party schema libraries. The service streams NDJSON, validates with a hand‑rolled validator, groups by model + fixture, computes mean / median / sample standard deviation, and exposes `/health`, `/summary`, and `/runs/:id`.

/**
 * Strict domain types.
 *
 * NOTE: every numeric field in the raw NDJSON payload is typed as `string`
 * (the wire format). The service parses them to `number` after validation.
 * This keeps the "raw" and "parsed" representations distinct and makes the
 * validator's job explicit.
 */

/** A single request-metrics record as it appears on the wire (one NDJSON line). */
export interface RawMetricRecord {
  /** Unique identifier for the request/run. */
  runId: string;
  /** Model identifier, e.g. "gpt-4" or "llama-3-70b". */
  model: string;
  /** Fixture / dataset identifier, e.g. "sft-001" or "eval-needle". */
  fixture: string;
  /** Latency in milliseconds (string on the wire). */
  latencyMs: string;
  /** Tokens per second (string on the wire). */
  tokensPerSec: string;
  /** Whether the request succeeded (string on the wire: "true"/"false"). */
  success: string;
  /** ISO-8601 timestamp of the request. */
  timestamp: string;
}

/** The validated, numeric form of a record. */
export interface MetricRecord {
  runId: string;
  model: string;
  fixture: string;
  latencyMs: number;
  tokensPerSec: number;
  success: boolean;
  timestamp: string;
}

/** A (model, fixture) group key. */
export interface GroupKey {
  model: string;
  fixture: string;
}

/** A single run's statistics. */
export interface RunStats {
  runId: string;
  model: string;
  fixture: string;
  count: number;
  meanLatencyMs: number;
  medianLatencyMs: number;
  sampleStdDevLatencyMs: number | null;
  meanTokensPerSec: number;
  medianTokensPerSec: number;
  sampleStdDevTokensPerSec: number | null;
  successRate: number;
  firstTimestamp: string;
  lastTimestamp: string;
}

/** A (model, fixture) group's aggregate statistics. */
export interface GroupStats {
  model: string;
  fixture: number;
  runCount: number;
  recordCount: number;
  meanLatencyMs: number;
  medianLatencyMs: number;
  sampleStdDevLatencyMs: number | null;
  meanTokensPerSec: number;
  medianTokensPerSec: number;
  sampleStdDevTokensPerSec: number | null;
  successRate: number;
}

/** The full /summary payload. */
export interface SummaryPayload {
  totalRecords: number;
  totalRuns: number;
  totalModels: number;
  totalFixtures: number;
  groups: GroupStats[];
}

/** Per-run payload for /runs/:id. */
export interface RunPayload {
  runId: string;
  stats: RunStats;
  records: MetricRecord[];
}

/**
 * Application error types.
 *
 * `ValidationError` carries a machine-readable `code` and a human `message`
 * so the HTTP layer can map it to a 400 with a stable error contract.
 */

export type ValidationErrorCode =
  | "NOT_OBJECT"
  | "MISSING_FIELD"
  | "WRONG_TYPE"
  | "INVALID_NUMBER"
  | "INVALID_BOOLEAN"
  | "INVALID_TIMESTAMP"
  | "INVALID_RUNID"
  | "EMPTY_PAYLOAD";

export class ValidationError extends Error {
  readonly code: ValidationErrorCode;
  readonly field: string | null;

  constructor(code: ValidationErrorCode, field: string | null, message: string) {
    super(message);
    this.name = "ValidationError";
    this.code = code;
    this.field = field;
  }

  /** Serialize to a stable JSON error body. */
  toJSON(): { code: string; field: string | null; message: string } {
    return { code: this.code, field: this.field, message: this.message };
  }
}

/** Thrown when the input stream is exhausted or a fatal stream error occurs. */
export class StreamError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "StreamError";
  }
}

## `src/validator.ts`

```ts
import { RawMetricRecord, MetricRecord } from "./types.js";
import { ValidationError } from "./errors.js";

/**
 * Hand-rolled validator. No third-party schema libraries.
 *
 * The validator is intentionally strict:
 *  - the payload must be a plain object (not an array, null, or primitive);
 *  - every required field must be present with the correct wire type;
 *  - numeric fields must parse to a finite number;
 *  - the boolean field must be the string "true" or "false";
 *  - the timestamp must be a valid ISO-8601 date;
 *  - the runId must be non-empty and reasonably bounded.
 */

const MAX_RUNID_LENGTH = 128;
const MAX_IDENTIFIER_LENGTH = 256;

const REQUIRED_STRING_FIELDS: (keyof RawMetricRecord)[] = [
  "runId",
  "model",
  "fixture",
  "latencyMs",
  "tokensPerSec",
  "success",
  "timestamp",
];

function isPlainObject(value: unknown): value is Record<string, unknown> {
  if (typeof value !== "object" || value === null) return false;
  // Reject arrays, Dates, Maps, etc.
  const proto = Object.getPrototypeOf(value);
  return proto === Object.prototype || proto === null;
}

function isFiniteNumberString(value: string): boolean {
  // Reject empty, whitespace-only, and non-numeric strings.
  const trimmed = value.trim();
  if (trimmed.length === 0) return false;
  const num = Number(trimmed);
  return Number.isFinite(num);
}

function isIsoTimestamp(value: string): boolean {
  // Accept a superset: full ISO-8601 with optional offset.
  // We validate by round-tripping through Date and checking the string
  // matches a strict ISO pattern to avoid accepting "1700000000000".
  const isoPattern =
    /^\d{4}-\d{2}-\d{2}[Tt]\d{2}:\d{2}:\d{2}(\.\d+)?([Zz]|[+-]\d{2}:?\d{2})?$/;
  if (!isoPattern.test(value)) return false;
  const ms = Date.parse(value);
  return !Number.isNaN(ms);
}

/**
 * Validate a raw wire payload and return a numeric MetricRecord.
 * Throws `ValidationError` on the first problem found.
 */
export function validateRawRecord(payload: unknown): MetricRecord {
  if (!isPlainObject(payload)) {
    throw new ValidationError(
      "NOT_OBJECT",
      null,
      "Payload must be a JSON object."
    );
  }

  const obj = payload as Record<string, unknown>;

  for (const field of REQUIRED_STRING_FIELDS) {
    if (!(field in obj) || obj[field] === null) {
      throw new ValidationError(
        "MISSING_FIELD",
        field,
        `Missing required field "${field}".`
      );
    }
    if (typeof obj[field] !== "string") {
      throw new ValidationError(
        "WRONG_TYPE",
        field,
        `Field "${field}" must be a string, got ${typeof obj[field]}.`
      );
    }
  }

  const runId = obj.runId as string;
  const model = obj.model as string;
  const fixture = obj.fixture as string;
  const latencyMsRaw = obj.latencyMs as string;
  const tokensPerSecRaw = obj.tokensPerSec as string;
  const successRaw = obj.success as string;
  const timestamp = obj.timestamp as string;

  if (runId.trim().length === 0 || runId.length > MAX_RUNID_LENGTH) {
    throw new ValidationError(
      "INVALID_RUNID",
      "runId",
      `runId must be non-empty and at most ${MAX_RUNID_LENGTH} characters.`
    );
  }
  if (model.trim().length === 0 || model.length > MAX_IDENTIFIER_LENGTH) {
    throw new ValidationError(
      "INVALID_RUNID",
      "model",
      `model must be non-empty and at most ${MAX_IDENTIFIER_LENGTH} characters.`
    );
  }
  if (fixture.trim().length === 0 || fixture.length > MAX_IDENTIFIER_LENGTH) {
    throw new ValidationError(
      "INVALID_RUNID",
      "fixture",
      `fixture must be non-empty and at most ${MAX_IDENTIFIER_LENGTH} characters.`
    );
  }

  if (!isFiniteNumberString(latencyMsRaw)) {
    throw new ValidationError(
      "INVALID_NUMBER",
      "latencyMs",
      `latencyMs must be a finite number, got "${latencyMsRaw}".`
    );
  }
  if (!isFiniteNumberString(tokensPerSecRaw)) {
    throw new ValidationError(
      "INVALID_NUMBER",
      "tokensPerSec",
      `tokensPerSec must be a finite number, got "${tokensPerSecRaw}".`
    );
  }

  if (successRaw !== "true" && successRaw !== "false") {
    throw new ValidationError(
      "INVALID_BOOLEAN",
      "success",
      `success must be the string "true" or "false", got "${successRaw}".`
    );
  }

  if (!isIsoTimestamp(timestamp)) {
    throw new ValidationError(
      "INVALID_TIMESTAMP",
      "timestamp",
      `timestamp must be a valid ISO-8601 date, got "${timestamp}".`
    );
  }

  return {
    runId,
    model,
    fixture,
    latencyMs: Number(latencyMsRaw),
    tokensPerSec: Number(tokensPerSecRaw),
    success: successRaw === "true",
    timestamp,
  };
}

/**
 * Parse a single NDJSON line (a JSON string) into a validated MetricRecord.
 * Throws `ValidationError` with code "NOT_OBJECT"/"EMPTY_PAYLOAD"/"INVALID_NUMBER"
 * as appropriate.
 */
export function parseAndValidateLine(line: string): MetricRecord {
  const trimmed = line.trim();
  if (trimmed.length === 0) {
    throw new ValidationError("EMPTY_PAYLOAD", null, "Empty line.");
  }
  let payload: unknown;
  try {
    payload = JSON.parse(trimmed);
  } catch {
    throw new ValidationError(
      "NOT_OBJECT",
      null,
      "Line is not valid JSON."
    );
  }
  return validateRawRecord(payload);
}

## `src/stats.ts`

```ts
import { MetricRecord } from "./types.js";

/**
 * Pure statistics helpers. No state, no I/O.
 */

/** Arithmetic mean of a non-empty array. */
export function mean(values: number[]): number {
  if (values.length === 0) return 0;
  let sum = 0;
  for (const v of values) sum += v;
  return sum / values.length;
}

/**
 * Median of a non-empty array.
 * Works on a copy so the input is not mutated.
 */
export function median(values: number[]): number {
  if (values.length === 0) return 0;
  const sorted = [...values].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  if (sorted.length % 2 === 1) {
    return sorted[mid];
  }
  return (sorted[mid - 1] + sorted[mid]) / 2;
}

/**
 * Sample standard deviation (n - 1 denominator, Bessel's correction).
 * Returns null when there are fewer than 2 observations (undefined sample variance).
 */
export function sampleStdDev(values: number[]): number | null {
  const n = values.length;
  if (n < 2) return null;
  const m = mean(values);
  let ss = 0;
  for (const v of values) {
    const d = v - m;
    ss += d * d;
  }
  const variance = ss / (n - 1);
  return Math.sqrt(variance);
}

/**
 * Compute run-level statistics for a single run (a set of records).
 * `records` must be non-empty.
 */
export function computeRunStats(
  runId: string,
  model: string,
  fixture: string,
  records: MetricRecord[]
): {
  count: number;
  meanLatencyMs: number;
  medianLatencyMs: number;
  sampleStdDevLatencyMs: number | null;
  meanTokensPerSec: number;
  medianTokensPerSec: number;
  sampleStdDevTokensPerSec: number | null;
  successRate: number;
  firstTimestamp: string;
  lastTimestamp: string;
} {
  const latencies = records.map((r) => r.latencyMs);
  const tps = records.map((r) => r.tokensPerSec);
  const successes = records.filter((r) => r.success).length;

  // First / last timestamps: use lexicographic comparison, which is valid
  // for ISO-8601 strings in the same timezone format. We normalize to UTC
  // for a stable ordering by parsing and comparing epoch ms.
  let first = records[0].timestamp;
  let last = records[0].timestamp;
  let firstMs = Date.parse(first);
  let lastMs = Date.parse(last);
  for (const r of records) {
    const ms = Date.parse(r.timestamp);
    if (ms < firstMs) {
      firstMs = ms;
      first = r.timestamp;
    }
    if (ms > lastMs) {
      lastMs = ms;
      last = r.timestamp;
    }
  }

  return {
    count: records.length,
    meanLatencyMs: mean(latencies),
    medianLatencyMs: median(latencies),
    sampleStdDevLatencyMs: sampleStdDev(latencies),
    meanTokensPerSec: mean(tps),
    medianTokensPerSec: median(tps),
    sampleStdDevTokensPerSec: sampleStdDev(tps),
    successRate: successes / records.length
