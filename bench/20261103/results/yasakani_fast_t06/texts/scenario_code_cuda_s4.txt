// ---------------------------------------------------------------------------
// RAII device buffer
// ---------------------------------------------------------------------------
struct DeviceBuffer {
    void*   d_ptr = nullptr;
    size_t  size_bytes = 0;

    explicit DeviceBuffer(size_t bytes) : size_bytes(bytes) {
        if (bytes > 0) {
            CUDA_CHECK(cudaMalloc(&d_ptr, bytes));
            CUDA_CHECK(cudaMemset(d_ptr, 0, bytes));
        }
    }

    DeviceBuffer() = default;

    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    DeviceBuffer(DeviceBuffer&& other) noexcept
        : d_ptr(other.d_ptr), size_bytes(other.size_bytes)
    {
        other.d_ptr = nullptr;
        other.size_bytes = 0;
    }

    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            free();
            d_ptr = other.d_ptr;
            size_bytes = other.size_bytes;
            other.d_ptr = nullptr;
            other.size_bytes = 0;
        }
        return *this;
    }

    ~DeviceBuffer() { free(); }

    void free() {
        if (d_ptr) {
            cudaFree(d_ptr);
            d_ptr = nullptr;
        }
        size_bytes = 0;
    }

    void* ptr() const { return d_ptr; }
    size_t size() const { return size_bytes; }
};

// ---------------------------------------------------------------------------
// Vectorized BF16 row-normalization kernel
//
// Each thread block handles one row.
// Each thread processes VEC=8 consecutive bf16 elements (16 bytes = one
// uint4 load).  The block cooperatively computes the per-row L2 norm in
// FP32, then scales every element by 1/norm.
//
// Launch geometry:
//   gridDim.x = rows
//   blockDim.x = min(cols / VEC, 1024)   (capped at 1024 threads/block)
//
// Memory layout:
//   The matrix is stored in row-major order: element (r, c) lives at
//   offset  r * cols + c  in the bf16 array.  Each row occupies
//   cols * 2 bytes.  A single uint4 (16 bytes) load fetches 8 bf16
//   values.  The grid is one block per row so that the entire row's
//   data is owned by a single block, enabling a simple shared-memory
//   reduction for the norm without any inter-block synchronization.
// ---------------------------------------------------------------------------
constexpr int VEC = 8;   // bf16 elements per uint4 load (16 bytes / 2 bytes)

__global__ void normalize_row_kernel(
    const __nv_bfloat16* __restrict__ input,
    __nv_bfloat16*       __restrict__ output,
    int rows,
    int cols)
{
    // Each block owns one row
    const int row = blockIdx.x;
    if (row >= rows) return;

    const __nv_bfloat16* row_ptr = input + static_cast<size_t>(row) * cols;

    // Each thread handles VEC consecutive elements.
    // Number of vector groups per row:
    const int vec_per_row = (cols + VEC - 1) / VEC;

    // Shared memory for per-thread partial sums of squares (FP32)
    __shared__ float s_partial[1024];

    const int tid = threadIdx.x;

    // Each thread may process multiple vector groups if vec_per_row > blockDim.x
    float local_sum_sq = 0.0f;

    for (int v = tid; v < vec_per_row; v += blockDim.x) {
        // Load 8 bf16 values as one uint4
        const uint4* vec_ptr = reinterpret_cast<const uint4*>(row_ptr + v * VEC);
        uint4 raw = *vec_ptr;

        // Unpack 8 bf16 values
        const __nv_bfloat16* bf = reinterpret_cast<const __nv_bfloat16*>(&raw);
        float f[VEC];
        float sum_sq = 0.0f;
        for (int i = 0; i < VEC; ++i) {
            f[i] = __bfloat162float(bf[i]);
            sum_sq += f[i] * f[i];
        }
        local_sum_sq += sum_sq;

        // Store the unpacked floats for the scaling pass (kept in registers
        // across the reduction; we re-read from global in the scaling pass
        // to keep register pressure low – see below).
    }

    // ---- Block reduction for row L2 norm ----
    s_partial[tid] = local_sum_sq;
    __syncthreads();

    // Intra-warp reduction (32 threads per warp)
    for (int offset = 16; offset > 0; offset >>= 1) {
        if (tid < offset) {
            // Only the first `offset` threads participate in this level
            if (tid + offset < blockDim.x) {
                s_partial[tid] += s_partial[tid + offset];
            }
        }
        __syncwarp();
    }

    // Inter-warp reduction: warp 0 reduces the per-warp sums
    if (tid < 32) {
        int num_warps = (blockDim.x + 31) / 32;
        for (int w = 1; w < num_warps; ++w) {
            if (w * 32 < blockDim.x) {
                s_partial[tid] += s_partial[w * 32 + tid];
            }
        }
        __syncwarp();
    }

    // Thread 0 computes the final norm
    float norm = 0.0f;
    if (tid == 0) {
        norm = sqrtf(s_partial[0]);
        // Guard against zero-norm rows (all-zero row)
        if (norm > 0.0f) {
            norm = 1.0f / norm;
        } else {
            norm = 0.0f;
        }
    }

    // Broadcast norm to all threads via shared memory
    __shared__ float s_norm;
    if (tid == 0) {
        s_norm = norm;
    }
    __syncthreads();
    float inv_norm = s_norm;

    // ---- Scaling pass: re-read input, multiply, write output ----
    for (int v = tid; v < vec_per_row; v += blockDim.x) {
        const uint4* vec_in  = reinterpret_cast<const uint4*>(row_ptr + v * VEC);
        uint4       raw_in   = *vec_in;
        const __nv_bfloat16* bf_in = reinterpret_cast<const __nv_bfloat16*>(&raw_in);

        uint4 raw_out;
        __nv_bfloat16* bf_out = reinterpret_cast<__nv_bfloat16*>(&raw_out);

        int remaining = cols - v * VEC;
        int n = (remaining < VEC) ? remaining : VEC;

        for (int i = 0; i < n; ++i) {
            float f = __bfloat162float(bf_in[i]);
            float scaled = f * inv_norm;
            bf_out[i] = __float2bfloat16(scaled);
        }

        // Pad remaining elements with zero if the row length is not a multiple of VEC
        for (int i = n; i < VEC; ++i) {
            bf_out[i] = __float2bfloat16(0.0f);
        }

        uint4* vec_out = reinterpret_cast<uint4*>(output + static_cast<size_t>(row) * cols + v * VEC);
        *vec_out = raw_out;
    }
}

// ---------------------------------------------------------------------------
// Deterministic input generation (host side, FP32 -> BF16)
// Uses a simple LCG for reproducibility without depending on <random>
// (which can vary across implementations).
// ---------------------------------------------------------------------------
void generate_input_bf16(
    std::vector<__nv_bfloat16>& data,
    size_t rows,
    size_t cols,
    unsigned int seed)
{
    data.resize(rows * cols);
    unsigned int state = seed;
    for (size_t i = 0; i < data.size(); ++i) {
        // 15-bit LCG (Numerical Recipes)
        state = state * 1103515245u + 12345u;
        float f = static_cast<float>((state >> 8) & 0xFFFF) / 65536.0f - 0.5f;
        data[i] = __float2bfloat16(f);
    }
}

// ---------------------------------------------------------------------------
// FP32 reference: numerically stable row normalization
// ---------------------------------------------------------------------------
void reference_normalize_fp32(
    const std::vector<__nv_bfloat16>& input,
    std::vector<float>& output,
    size_t rows,
    size_t cols)
{
    output.resize(rows * cols);
    for (size_t r = 0; r < rows; ++r) {
        const __nv_bfloat16* row_in = input.data() + r * cols;

        // Compute L2 norm in FP32 (stable enough for verification)
        float sum_sq = 0.0f;
        for (size_t c = 0; c < cols; ++c) {
            float val = __bfloat162float(row_in[c]);
            sum_sq += val * val;
        }
        float norm = sqrtf(sum_sq);
        float inv_norm = (norm > 0.0f) ? (1.0f / norm) : 0.0f;

        float* row_out = output.data() + r * cols;
        for (size_t c = 0; c < cols; ++c) {
            float val = __bfloat162float(row_in[c]);
            row_out[c] = val * inv_norm;
        }
    }
}

// ---------------------------------------------------------------------------
// Tolerance-based verification
// ---------------------------------------------------------------------------
bool verify_results(
    const std::vector<__nv_bfloat16>& gpu_result,
    const std::vector<float>&        ref_result,
    size_t rows,
    size_t cols,
    float  rtol,
    float  atol,
    float& max_rel_err,
    float& max_abs_err)
{
    max_rel_err = 0.0f;
    max_abs_err = 0.0f;

    size_t total = rows * cols;
    for (size_t i = 0; i < total; ++i) {
        float gpu_val = __bfloat162float(gpu_result[i]);
        float ref_val = ref_result[i];

        float abs_diff = fabsf(gpu_val - ref_val);
        float denom = fabsf(ref_val);
        if (denom < 1e-8f) denom = 1.0f;
        float rel_diff = abs_diff / denom;

        if (abs_diff > max_abs_err) max_abs_err = abs_diff;
        if (rel_diff > max_rel_err) max_rel_err = rel_diff;

        // Tolerance: |gpu - ref| <= atol + rtol * |ref|
        if (abs_diff > (atol + rtol * fabsf(ref_val))) {
            return false;
        }
    }
    return true;
}

// Argument parsing

struct Args {
    int rows  = 4096;
    int cols  = 8192;
    int iters = 10;
};

void parse_args(int argc, char* argv[], Args& args) {
    if (argc >= 2) {
        args.rows = std::atoi(argv[1]);
        if (args.rows <= 0) {
            fprintf(stderr, "Error: rows must be positive.\n");
            exit(EXIT_FAILURE);
        }
    }
    if (argc >= 3) {
        args.cols = std::atoi(argv[2]);
        if (args.cols <= 0) {
            fprintf(stderr, "Error: cols must be positive.\n");
            exit(EXIT_FAILURE);
        }
    }
    if (argc >= 4) {
        args.iters = std::atoi(argv[3]);
        if (args.iters <= 0) {
            fprintf(stderr, "Error: iters must be positive.\n");
            exit(EXIT_FAILURE);
        }
    }
}

    std::vector<__nv_bfloat16> h_input;

    DeviceBuffer d_input(total_bytes);
    DeviceBuffer d_output(total_bytes);

    // One block per row; each block has up to 1024 threads.

    int block_size  = std::min(vec_per_row, 1024);
    int grid_size   = static_cast<int>(rows);

    printf("  block = %d threads\n", block_size);
    printf("  VEC   = %d bf16 per uint4 load\n", VEC);
