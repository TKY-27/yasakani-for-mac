CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    param_count BIGINT NOT NULL CHECK (param_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_kind VARCHAR(50) NOT NULL,
    source VARCHAR(100) NOT NULL,
    avg_token_count INT NOT NULL CHECK (avg_token_count > 0),
    checksum VARCHAR(64) NOT NULL UNIQUE,
    CHECK (fixture_kind IN ('code', 'prose', 'math', 'structured'))
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    completed_at TIMESTAMPTZ,
    hardware VARCHAR(50) NOT NULL DEFAULT 'A100',
    batch_size INT NOT NULL CHECK (batch_size > 0),
    max_new_tokens INT NOT NULL CHECK (max_new_tokens > 0),
    CHECK (completed_at IS NULL OR completed_at >= started_at)
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL,
    tokens_per_second DECIMAL(10, 4) NOT NULL CHECK (tokens_per_second > 0),
    latency_ms DECIMAL(10, 2) NOT NULL CHECK (latency_ms > 0),
    mtp_acceptance_rate DECIMAL(4, 4) CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    context_retention_score DECIMAL(4, 4) CHECK (context_retention_score BETWEEN 0 AND 1),
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_tps ON request_metrics(tokens_per_second);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_acceptance_rate);

CREATE VIEW v_run_summary AS
SELECT
    r.run_id,
    m.name AS model_name,
    f.fixture_kind,
    COUNT(rm.metric_id) AS total_requests,
    AVG(rm.tokens_per_second) AS avg_tps,
    STDDEV_POP(rm.tokens_per_second) AS tps_stddev,
    AVG(rm.latency_ms) AS avg_latency,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_rate,
    AVG(rm.context_retention_score) AS avg_retention
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY r.run_id, m.name, f.fixture_kind;

CREATE VIEW v_long_context_analysis AS
SELECT
    m.name AS model_name,
    f.fixture_kind,
    COUNT(DISTINCT r.run_id) AS runs,
    AVG(rm.context_retention_score) AS mean_retention,
    MIN(rm.context_retention_score) AS min_retention,
    MAX(rm.context_retention_score) AS max_retention,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY rm.context_retention_score) AS p95_retention
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
WHERE rm.context_retention_score IS NOT NULL
GROUP BY m.name, f.fixture_kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_tps DECIMAL,
    p_latency DECIMAL,
    p_mtp DECIMAL,
    p_retention DECIMAL
) RETURNS request_metrics AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, tokens_per_second, latency_ms, mtp_acceptance_rate, context_retention_score)
    VALUES (p_run_id, p_request_index, p_tps, p_latency, p_mtp, p_retention)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        tokens_per_second = EXCLUDED.tokens_per_second,
        latency_ms = EXCLUDED.latency_ms,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        context_retention_score = EXCLUDED.context_retention_score;
    RETURNING *;
END;
$$ LANGUAGE plpgsql;
