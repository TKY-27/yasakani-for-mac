BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    max_context_length INTEGER NOT NULL CHECK (max_context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_kind VARCHAR(50) NOT NULL CHECK (fixture_kind IN ('synthetic', 'real_world', 'adversarial', 'multilingual')),
    description VARCHAR(255),
    token_count INTEGER NOT NULL CHECK (token_count > 0)
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INTEGER NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INTEGER NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    status VARCHAR(20) DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'interrupted')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INTEGER NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    input_tokens INTEGER NOT NULL CHECK (input_tokens > 0),
    output_tokens INTEGER NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms > 0),
    mtp_acceptance_rate DOUBLE PRECISION CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    long_context_retention_score DOUBLE PRECISION CHECK (long_context_retention_score BETWEEN 0 AND 1)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_timestamp ON request_metrics(timestamp);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);

CREATE VIEW v_throughput_summary AS
SELECT 
    m.name AS model_name,
    f.fixture_kind,
    COUNT(r.run_id) AS total_runs,
    AVG(rm.latency_ms) AS avg_latency_ms,
    SUM(rm.input_tokens + rm.output_tokens) / (SUM(rm.latency_ms) / 1000.0) AS tokens_per_second
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name, f.fixture_kind;

CREATE VIEW v_variability_analysis AS
SELECT 
    m.name AS model_name,
    f.fixture_kind,
    STDDEV(rm.latency_ms) AS latency_stddev,
    STDDEV(rm.mtp_acceptance_rate) AS mtp_stddev,
    STDDEV(rm.long_context_retention_score) AS retention_stddev,
    COUNT(rm.metric_id) AS sample_size
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
GROUP BY m.name, f.fixture_kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INTEGER,
    p_input_tokens INTEGER,
    p_output_tokens INTEGER,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_acceptance_rate DOUBLE PRECISION,
    p_long_context_retention_score DOUBLE PRECISION
) RETURNS INTEGER AS $$
DECLARE
    v_metric_id INTEGER;
BEGIN
    INSERT INTO request_metrics (
        run_id,
        input_tokens,
        output_tokens,
        latency_ms,
        mtp_acceptance_rate,
        long_context_retention_score
    ) VALUES (
        p_run_id,
        p_input_tokens,
        p_output_tokens,
        p_latency_ms,
        p_mtp_acceptance_rate,
        p_long_context_retention_score
    )
    RETURNING metric_id INTO v_metric_id;
    RETURN v_metric_id;
END;
$$ LANGUAGE plpgsql;
