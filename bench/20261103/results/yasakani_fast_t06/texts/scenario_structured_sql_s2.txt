CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    model_name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_window INT NOT NULL CHECK (context_window > 0),
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_kind TEXT NOT NULL CHECK (fixture_kind IN ('code', 'prose', 'math', 'mixed')),
    fixture_name TEXT NOT NULL UNIQUE,
    avg_token_length INT NOT NULL CHECK (avg_token_length > 0),
    complexity_score NUMERIC(5,2) NOT NULL CHECK (complexity_score BETWEEN 0 AND 10)
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    run_timestamp TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    batch_size INT NOT NULL CHECK (batch_size > 0),
    sequence_length INT NOT NULL CHECK (sequence_length > 0),
    hardware_config TEXT NOT NULL,
    UNIQUE (model_id, fixture_id, run_timestamp)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    tokens_per_second NUMERIC(10,4) NOT NULL CHECK (tokens_per_second > 0),
    latency_ms NUMERIC(10,2) NOT NULL CHECK (latency_ms > 0),
    memory_usage_gb NUMERIC(8,2) NOT NULL CHECK (memory_usage_gb > 0),
    mtp_acceptance_rate NUMERIC(4,3) NOT NULL CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    long_context_retention_score NUMERIC(4,3) NOT NULL CHECK (long_context_retention_score BETWEEN 0 AND 1),
    variability_index NUMERIC(4,3) NOT NULL CHECK (variability_index BETWEEN 0 AND 1),
    UNIQUE (run_id)
);

CREATE INDEX idx_runs_model_id ON runs(model_id);
CREATE INDEX idx_runs_fixture_id ON runs(fixture_id);
CREATE INDEX idx_runs_timestamp ON runs(run_timestamp);
CREATE INDEX idx_metrics_run_id ON request_metrics(run_id);
CREATE INDEX idx_metrics_tps ON request_metrics(tokens_per_second);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_acceptance_rate);

CREATE VIEW v_model_performance_summary AS
SELECT 
    m.model_id,
    m.model_name,
    m.architecture,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.tokens_per_second) AS avg_tps,
    AVG(rm.latency_ms) AS avg_latency_ms,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance,
    AVG(rm.long_context_retention_score) AS avg_retention,
    AVG(rm.variability_index) AS avg_variability
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.model_id, m.model_name, m.architecture;

CREATE VIEW v_fixture_analysis AS
SELECT 
    f.fixture_id,
    f.fixture_kind,
    f.fixture_name,
    f.avg_token_length,
    f.complexity_score,
    COUNT(DISTINCT r.run_id) AS run_count,
    AVG(rm.tokens_per_second) AS avg_tps,
    STDDEV(rm.tokens_per_second) AS tps_stddev,
    AVG(rm.variability_index) AS avg_variability,
    AVG(rm.long_context_retention_score) AS avg_retention
FROM fixtures f
JOIN runs r ON f.fixture_id = r.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY f.fixture_id, f.fixture_kind, f.fixture_name, f.avg_token_length, f.complexity_score;

CREATE OR REPLACE FUNCTION upsert_request_metrics(
    p_run_id INT,
    p_tps NUMERIC,
    p_latency NUMERIC,
    p_memory NUMERIC,
    p_mtp NUMERIC,
    p_retention NUMERIC,
    p_variability NUMERIC
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, tokens_per_second, latency_ms, memory_usage_gb, mtp_acceptance_rate, long_context_retention_score, variability_index)
    VALUES (p_run_id, p_tps, p_latency, p_memory, p_mtp, p_retention, p_variability)
    ON CONFLICT (run_id) DO UPDATE SET
        tokens_per_second = EXCLUDED.tokens_per_second,
        latency_ms = EXCLUDED.latency_ms,
        memory_usage_gb = EXCLUDED.memory_usage_gb,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        long_context_retention_score = EXCLUDED.long_context_retention_score,
        variability_index = EXCLUDED.variability_index;
END;
$$ LANGUAGE plpgsql;
