```python
class Redactor:
    def __init__(self, mappings: Sequence[tuple[str, str]]):  # (value, replacement)
        self._pairs = [(v, r) for v, r in mappings if v]
        # sort by length desc to redact longer values first
        self._pairs.sort(key=lambda p: len(p[0]), reverse=True)

    def redact(self, text: str) -> str:
        out = text
        for value, repl in self._pairs:
            if value in out:
                out = out.replace(value, repl)
        return out
```
