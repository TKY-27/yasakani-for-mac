CREATE OR REPLACE VIEW v_run_summary AS
SELECT r.run_id, m.model_id, m.name AS model_name, f.fixture_id, f.kind AS fixture_kind,
 r.started_at, r.finished_at, r.status,
 count(x.metric_id) AS request_count,
 sum(x.input_tokens) AS total_input_tokens,
 sum(x.output_tokens) AS total_output_tokens,
 avg(x.latency_ms) AS avg_latency_ms,
 percentile_cont(0.5) WITHIN GROUP (ORDER BY x.latency_ms) AS p50_latency_ms,
 avg(x.throughput_tps) AS avg_tps,
 sum(x.mtp_proposed) AS mtp_proposed,
 sum(x.mtp_accepted) AS mtp_accepted
FROM runs r JOIN models m ON ... JOIN fixtures f ... LEFT JOIN request_metrics x ...
GROUP BY r.run_id, m.model_id, m.name, f.fixture_id, f.kind, r.started_at, r.finished_at, r.status;

CREATE OR REPLACE VIEW v_model_fixture_performance AS
SELECT m.model_id, m.name, f.fixture_id, f.kind,
 count(DISTINCT r.run_id) AS run_count,
 count(x.metric_id) AS request_count,
 avg(x.throughput_tps) AS avg_tps,
 stddev(x.throughput_tps) AS tps_stddev,
 avg(x.latency_ms) AS avg_latency_ms,
 stddev(x.latency_ms) AS latency_stddev,
 avg(x.retention_score) FILTER (WHERE x.retention_score IS NOT NULL) AS avg_retention_score,
 sum(x.mtp_proposed) AS mtp_proposed,
 sum(x.mtp_accepted) AS mtp_accepted,
 CASE WHEN sum(x.mtp_proposed) > 0 THEN sum(x.mtp_accepted)::numeric / sum(x.mtp_proposed) ELSE NULL END AS mtp_accept_rate
FROM models m JOIN fixtures f ... LEFT JOIN runs r ON ... LEFT JOIN request_metrics x ...
GROUP BY m.model_id, m.name, f.fixture_id, f
