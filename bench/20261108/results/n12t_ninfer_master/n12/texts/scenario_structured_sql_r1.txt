CREATE OR REPLACE FUNCTION upsert_run(
  p_model_name text,
  p_fixture_name text,
  p_run_ts timestamptz,
  p_seed int,
  p_batch_size int,
  p_hardware text,
  p_request_count int,
  p_total_tokens int,
  p_latency_p50_ms numeric,
  p_latency_p95_ms numeric,
  p_throughput_tps numeric,
  p_mtp_accept_rate numeric
) RETURNS int
LANGUAGE plpgsql
AS $$
DECLARE
  v_model_id int;
  v_fixture_id int;
  v_run_id int;
BEGIN
  SELECT model_id INTO v_model_id FROM models WHERE model_name = p_model_name;
  IF v_model_id IS NULL THEN
    RAISE EXCEPTION 'model % not found', p_model_name;
  END IF;
  SELECT fixture_id INTO v_fixture_id FROM fixtures WHERE fixture_name = p_fixture_name;
  IF v_fixture_id IS NULL THEN
    RAISE EXCEPTION 'fixture % not found', p_fixture_name;
  END IF;

  INSERT INTO runs (model_id, fixture_id, run_ts, seed, batch_size, hardware, request_count, total_tokens, latency_p50_ms, latency_p95_ms, throughput_tps, mtp_accept_rate)
  VALUES (v_model_id, v_fixture_id, p_run_ts, p_seed, p_batch_size, p_hardware, p_request_count, p_total_tokens, p_latency_p50_ms, p_latency_p95_ms, p_throughput_tps, p_mtp_accept_rate)
  ON CONFLICT (model_id, fixture_id, run_ts, seed)
  DO UPDATE SET
    batch_size = EXCLUDED.batch_size,
    hardware = EXCLUDED.hardware,
    request_count = EXCLUDED.request_count,
    total_tokens = EXCLUDED.total_tokens,
    latency_p50_ms = EXCLUDED.latency_p50_ms,
    latency_p95_ms = EXCLUDED.latency_p95_ms,
    throughput_tps = EXCLUDED.throughput_tps,
    mtp_accept_rate = EXCLUDED.mtp_accept_rate;

  SELECT run_id INTO v_run_id FROM runs WHERE model_id = v_model_id AND fixture_id = v_fixture_id AND run_ts = p_run_ts AND seed = p_seed;
  RETURN v_run_id;
END;
$$;
