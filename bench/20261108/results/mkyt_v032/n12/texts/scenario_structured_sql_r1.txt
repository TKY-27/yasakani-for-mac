BEGIN;

CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    model_name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    fixture_name VARCHAR(100) NOT NULL UNIQUE,
    fixture_kind VARCHAR(50) NOT NULL CHECK (fixture_kind IN ('short', 'medium', 'long', 'code')),
    token_count INT NOT NULL CHECK (token_count > 0),
    description TEXT
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id),
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id),
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id),
    request_index INT NOT NULL,
    tokens_generated INT NOT NULL CHECK (tokens_generated >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted BOOLEAN NOT NULL DEFAULT FALSE,
    mtp_rejected BOOLEAN NOT NULL DEFAULT FALSE,
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_accepted);

CREATE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.model_name,
    f.fixture_name,
    f.fixture_kind,
    r.started_at,
    r.finished_at,
    r.status,
    COUNT(qm.metric_id) AS total_requests,
    SUM(qm.tokens_generated) AS total_tokens,
    AVG(qm.latency_ms) AS avg_latency_ms,
    SUM(CASE WHEN qm.mtp_accepted THEN 1 ELSE 0 END) AS mtp_accept_count,
    SUM(CASE WHEN qm.mtp_rejected THEN 1 ELSE 0 END) AS mtp_reject_count
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics qm ON r.run_id = qm.run_id
GROUP BY r.run_id, m.model_name, f.fixture_name, f.fixture_kind, r.started_at, r.finished_at, r.status;

CREATE VIEW v_model_performance AS
SELECT 
    m.model_name,
    m.parameter_count,
    m.context_length,
    COUNT(DISTINCT r.run_id) AS total_runs,
    SUM(COALESCE(rs.total_tokens, 0)) AS total_tokens_generated,
    AVG(COALESCE(rs.avg_latency_ms, 0)) AS overall_avg_latency_ms,
    SUM(COALESCE(rs.mtp_accept_count, 0)) AS total_mtp_accepted,
    SUM(COALESCE(rs.mtp_reject_count, 0)) AS total_mtp_rejected,
    CASE 
        WHEN SUM(COALESCE(rs.mtp_accept_count, 0) + COALESCE(rs.mtp_reject_count, 0)) > 0 
        THEN ROUND(100.0 * SUM(COALESCE(rs.mtp_accept_count, 0)) / (SUM(COALESCE(rs.mtp_accept_count, 0)) + SUM(COALESCE(rs.mtp_reject_count, 0))), 2)
        ELSE 0 
    END AS mtp_acceptance_rate_pct
FROM models m
LEFT JOIN runs r ON m.model_id = r.model_id
LEFT JOIN v_run_summary rs ON r.run_id = rs.run_id
GROUP BY m.model_name, m.parameter_count, m.context_length;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_tokens_generated INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted BOOLEAN,
    p_mtp_rejected BOOLEAN
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, tokens_generated, latency_ms, mtp_accepted, mtp_rejected)
    VALUES (p_run_id, p_request_index, p_tokens_generated, p_latency_ms, p_mtp_accepted, p_mtp_rejected)
    ON CONFLICT (run_id, request_index)
    DO UPDATE SET
        tokens_generated = EXCLUDED.tokens_generated,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_rejected = EXCLUDED.mtp_rejected;
END;
$$ LANGUAGE plpgsql;

SELECT 
    m.model_name,
    f.fixture_kind,
    COUNT(qm.metric_id) AS total_requests,
    SUM(qm.tokens_generated) AS total_tokens,
    ROUND(AVG(qm.latency_ms), 2) AS avg_latency_ms,
    ROUND(100.0 * SUM(CASE WHEN qm.mtp_accepted THEN 1 ELSE 0 END) / COUNT(qm.metric_id), 2) AS mtp_acceptance_rate_pct
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics qm ON r.run_id = qm.run_id
GROUP BY m.model_name, f.fixture_kind
ORDER BY m.model_name, f.fixture_kind;

SELECT 
    m.model_name,
    f.fixture_kind,
    STDDEV(qm.latency_ms) AS latency_stddev,
    MIN(qm.latency_ms) AS min_latency_ms,
    MAX(qm.latency_ms) AS max_latency_ms,
    ROUND(100.0 * STDDEV(qm.latency_ms) / AVG(qm.latency_ms), 2) AS coefficient_of_variation_pct
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics qm ON r.run_id = qm.run_id
GROUP BY m.model_name, f.fixture_kind
ORDER BY m.model_name, f.fixture_kind;

SELECT 
    m.model_name,
    f.fixture_name,
    f.fixture_kind,
    f.token_count AS fixture_token_count,
    SUM(qm.tokens_generated) AS total_tokens_generated,
    ROUND(100.0 * SUM(qm.tokens_generated) / f.token_count, 2) AS retention_ratio_pct
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics qm ON r.run_id = qm.run_id
WHERE f.fixture_kind = 'long'
GROUP BY m.model_name, f.fixture_name, f.fixture_kind, f.token_count
ORDER BY m.model_name, f.fixture_name;
