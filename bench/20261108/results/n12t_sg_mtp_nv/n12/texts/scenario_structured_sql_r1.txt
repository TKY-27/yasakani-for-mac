CREATE TABLE request_metrics (
 metric_id bigserial PRIMARY KEY,
 run_id bigint NOT NULL REFERENCES runs(run_id),
 request_number smallint NOT NULL CHECK (request_number > 0),
 prompt_tokens integer NOT NULL CHECK (prompt_tokens >= 0),
 completion_tokens integer NOT NULL CHECK (completion_tokens >= 0),
 latency_ms numeric(12,3) NOT NULL CHECK (latency_ms > 0),
 first_token_latency_ms numeric(12,3) NOT NULL CHECK (first_token_latency_ms >= 0 AND first_token_latency_ms <= latency_ms),
 mtp_proposed_tokens integer NOT NULL DEFAULT 0 CHECK (mtp_proposed_tokens >= 0),
 mtp_accepted_tokens integer NOT NULL DEFAULT 0 CHECK (mtp_accepted_tokens >= 0 AND mtp_accepted_tokens <= mtp_proposed_tokens),
 retention_score numeric(5,4) CHECK (retention_score IS NULL OR (retention_score >= 0 AND retention_score <= 1)),
 UNIQUE (run_id, request_number)
);
