CREATE VIEW v_benchmark_summary AS
SELECT m.model_name, f.fixture_name, f.fixture_kind, r.run_id, r.status,
count(qm.request_id) AS requests,
avg(qm.completion_tokens / nullif(qm.latency_ms,0) * 1000) AS avg_completion_tps,
percentile_cont(0.5) within group (order by qm.latency_ms) AS p50_latency_ms,
percentile_cont(0.99) within group (order by qm.latency_ms) AS p99_latency_ms,
avg(qm.mtp_accepted::numeric / nullif(qm.mtp_candidates,0)) AS avg_mtp_acceptance_rate,
avg(qm.context_retention) AS avg_context_retention
FROM runs r JOIN models m ON ... JOIN fixtures f ... JOIN request_metrics qm ON qm.run_id=r.run_id
GROUP BY ...;
