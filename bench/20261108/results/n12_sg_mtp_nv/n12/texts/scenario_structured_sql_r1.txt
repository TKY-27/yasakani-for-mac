BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    tokens_in INT NOT NULL CHECK (tokens_in >= 0),
    tokens_out INT NOT NULL CHECK (tokens_out >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted BOOLEAN NOT NULL DEFAULT FALSE,
    mtp_rejected BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_started ON runs(started_at);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX IF NOT EXISTS idx_metrics_mtp ON request_metrics(mtp_accepted);

CREATE OR REPLACE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.status,
    r.started_at,
    r.finished_at,
    COUNT(met.metric_id) AS total_requests,
    SUM(met.tokens_in) AS total_tokens_in,
    SUM(met.tokens_out) AS total_tokens_out,
    AVG(met.latency_ms) AS avg_latency_ms,
    MIN(met.latency_ms) AS min_latency_ms,
    MAX(met.latency_ms) AS max_latency_ms
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
GROUP BY r.run_id, m.name, f.kind, r.status, r.started_at, r.finished_at;

CREATE OR REPLACE VIEW v_mtp_stats AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(*) AS total_mtp_attempts,
    SUM(CASE WHEN met.mtp_accepted THEN 1 ELSE 0 END) AS accepted,
    SUM(CASE WHEN met.mtp_rejected THEN 1 ELSE 0 END) AS rejected,
    ROUND(SUM(CASE WHEN met.mtp_accepted THEN 1 ELSE 0 END)::NUMERIC / NULLIF(COUNT(*), 0) * 100, 2) AS acceptance_rate_pct
FROM request_metrics met
JOIN runs r ON met.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
WHERE met.mtp_accepted OR met.mtp_rejected
GROUP BY m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_tokens_in INT,
    p_tokens_out INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted BOOLEAN,
    p_mtp_rejected BOOLEAN
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (
        run_id, request_index, tokens_in, tokens_out, latency_ms, mtp_accepted, mtp_rejected
    ) VALUES (
        p_run_id, p_request_index, p_tokens_in, p_tokens_out, p_latency_ms, p_mtp_accepted, p_mtp_rejected
    )
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        tokens_in = EXCLUDED.tokens_in,
        tokens_out = EXCLUDED.tokens_out,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_rejected = EXCLUDED.mtp_rejected,
        created_at = CURRENT_TIMESTAMP;
END;
$$ LANGUAGE plpgsql;

    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(met.metric_id) AS total_requests,
    SUM(met.tokens_out) AS total_tokens_out,
    ROUND(SUM(met.tokens_out)::NUMERIC / NULLIF(EXTRACT(EPOCH FROM (MAX(r.finished_at) - MIN(r.started_at))), 0) * 60, 2) AS tokens_per_minute
FROM request_metrics met
JOIN runs r ON met.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
GROUP BY m.name, f.kind
ORDER BY m.name, f.kind;
