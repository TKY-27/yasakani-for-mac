BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    architecture VARCHAR(100) NOT NULL,
    parameter_count BIGINT CHECK (parameter_count > 0),
    context_length INT CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('code', 'math', 'long_context', 'chat')),
    description TEXT,
    avg_token_length INT CHECK (avg_token_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    batch_size INT CHECK (batch_size > 0),
    sequence_length INT CHECK (sequence_length > 0),
    start_time TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    end_time TIMESTAMP,
    status VARCHAR(20) CHECK (status IN ('completed', 'failed', 'running')),
    UNIQUE (model_id, fixture_id, batch_size, sequence_length, start_time)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_runs_status ON runs(status);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL,
    input_tokens INT CHECK (input_tokens >= 0),
    output_tokens INT CHECK (output_tokens >= 0),
    latency_ms FLOAT CHECK (latency_ms >= 0),
    mtp_accepted_tokens INT CHECK (mtp_accepted_tokens >= 0),
    mtp_total_proposed INT CHECK (mtp_total_proposed >= 0),
    first_token_latency_ms FLOAT CHECK (first_token_latency_ms >= 0),
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_accepted_tokens, mtp_total_proposed);

CREATE VIEW v_throughput_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.batch_size,
    r.sequence_length,
    COUNT(*) AS total_requests,
    SUM(rm.output_tokens) * 1.0 / (MAX(rm.timestamp) - MIN(rm.timestamp)) AS tokens_per_second
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY r.run_id, m.name, f.kind, r.batch_size, r.sequence_length;

CREATE VIEW v_mtp_acceptance_rate AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    SUM(rm.mtp_accepted_tokens) * 1.0 / NULLIF(SUM(rm.mtp_total_proposed), 0) AS acceptance_rate,
    AVG(rm.mtp_accepted_tokens) AS avg_accepted,
    AVG(rm.mtp_total_proposed) AS avg_proposed
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed' AND rm.mtp_total_proposed > 0
GROUP BY r.run_id, m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms FLOAT,
    p_mtp_accepted INT,
    p_mtp_proposed INT,
    p_first_token_ms FLOAT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (
        run_id, request_index, input_tokens, output_tokens, latency_ms,
        mtp_accepted_tokens, mtp_total_proposed, first_token_latency_ms
    ) VALUES (
        p_run_id, p_request_index, p_input_tokens, p_output_tokens, p_latency_ms,
        p_mtp_accepted, p_mtp_proposed, p_first_token_ms
    )
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens,
        mtp_total_proposed = EXCLUDED.mtp_total_proposed,
        first_token_latency_ms = EXCLUDED.first_token_latency_ms,
        timestamp = CURRENT_TIMESTAMP;
END;
$$ LANGUAGE plpgsql;
