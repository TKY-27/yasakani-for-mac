Below is a complete, self-contained implementation. It compiles with CUDA 13.1 (sm_120a) and C++20, normalizes each row of a large BF16 matrix in place, verifies against an FP32 reference, and reports timing.

```
project/
├── CMakeLists.txt
└── src/
    └── bf16_row_norm.cu
```

---

### `CMakeLists.txt`

```cmake
cmake_minimum_required(VERSION 3.24)
project(bf16_row_norm LANGUAGES CXX CUDA)

# --- Language standards -----------------------------------------------------
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

# --- Target GPU: sm_120a (Blackwell datacenter) -----------------------------
# The "a" suffix enables arch-specific features (e.g. tcgen05 / TMA).
# We compile for the exact architecture so the binary runs only on sm_120a.
set(CMAKE_CUDA_ARCHITECTURES "120a")

# --- Build type -------------------------------------------------------------
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# --- Executable -------------------------------------------------------------
add_executable(bf16_row_norm src/bf16_row_norm.cu)

target_compile_options(bf16_row_norm PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:
        --expt-relaxed-constexpr          # allow constexpr host/device interop
        -O3                              # full optimization
        --use_fast_math                   # fast intrinsics (we verify numerically)
    >
)

# Link math library (cudart is linked automatically by FindCUDA/CMake).
target_link_libraries(bf16_row_norm PRIVATE cudart)

# --- Install rule (optional) ------------------------------------------------
install(TARGETS bf16_row_norm RUNTIME DESTINATION bin)
```

### `src/bf16_row_norm.cu`

```cpp
// ============================================================================
// bf16_row_norm.cu
//
// Row-wise L2 normalization of a large BF16 matrix on an sm_120a GPU.
//
//   out[i][j] = in[i][j] / sqrt( sum_j in[i][j]^2 )
//
// Pipeline:
//   1. Deterministic input generation (host, seeded PRNG -> device).
//   2. Vectorized CUDA kernel computes per-row norm in FP32, then writes
//      normalized BF16 back to the same buffer (in-place).
//   3. Numerically stable FP32 reference computed on host.
//   4. Tolerance-based verification (max abs error + relative error).
//   5. Warm-up + CUDA-event timing over multiple iterations.
//
// Memory layout:
//   Matrix stored row-major: element (i,j) at linear index i*cols + j.
//   Each thread block handles one or more rows; threads within a block
//   cooperate via shared memory to reduce a row's sum-of-squares.
//
// Launch geometry:
//   gridDim.x  = number of blocks (one per row-group)
//   blockDim.x = threads per block (cooperative reduction width)
//   Each block processes ROWS_PER_BLOCK rows sequentially.
// ============================================================================

#include <cuda_runtime.h>
#include <cuda_bf16.h>          // __nv_bfloat16, __bfloat162, etc.
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cmath>
#include <cstdint>
#include <vector>
#include <string>
#include <algorithm>
#include <numeric>
#include <random>
#include <chrono>
#include <stdexcept>
#include <limits>
#include <type_traits>

// ---------------------------------------------------------------------------
// Error-checking macro
// ---------------------------------------------------------------------------
#define CUDA_CHECK(call)                                                     \
    do {                                                                     \
        cudaError_t err__ = (call);                                          \
        if (err__ != cudaSuccess) {                                          \
            fprintf(stderr,                                                  \
                    "CUDA error %s:%d: %s\n",                                \
                    __FILE__, __LINE__,                                       \
                    cudaGetErrorString(err__));                               \
            exit(EXIT_FAILURE);                                              \
        }                                                                    \
    } while (0)

// ---------------------------------------------------------------------------
// RAII device buffer
// ---------------------------------------------------------------------------
class DeviceBuffer {
public:
    DeviceBuffer() : ptr_(nullptr), size_bytes_(0) {}

    explicit DeviceBuffer(size_t size_bytes) {
        allocate(size_bytes);
    }

    ~DeviceBuffer() {
        free();
    }

    // Non-copyable
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    // Movable
    DeviceBuffer(DeviceBuffer&& other) noexcept
        : ptr_(other.ptr_), size_bytes_(other.size_bytes_) {
        other.ptr_ = nullptr;
        other.size_bytes_ = 0;
    }
    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            free();
            ptr_ = other.ptr_;
            size_bytes_ = other.size_bytes_;
            other.ptr_ = nullptr;
            other.size_bytes_ = 0;
        }
        return *this;
    }

    void*       data()       { return ptr_; }
    const void* data() const { return ptr_; }
    size_t      size_bytes() const { return size_bytes_; }
    bool        valid() const { return ptr_ != nullptr; }

    template <typename T>
    T*       as()       { return static_cast<T*>(ptr_); }
    template <typename T>
    const T* as() const { return static_cast<const T*>(ptr_); }

    // Host -> device
    void copy_from_host(const void* src, size_t bytes) {
        if (bytes > size_bytes_) throw std::runtime_error("copy_from_host: size mismatch");
        CUDA_CHECK(cudaMemcpy(ptr_, src, bytes, cudaMemcpyHostToDevice));
    }

    // device -> host
    void copy_to_host(void* dst, size_t bytes) {
        if (bytes > size_bytes_) throw std::runtime_error("copy_to_host: size mismatch");
        CUDA_CHECK(cudaMemcpy(dst, ptr_, bytes, cudaMemcpyDeviceToHost));
    }

private:
    void allocate(size_t size_bytes) {
        free();
        size_bytes_ = size_bytes;
        if (size_bytes_ == 0) return;
        CUDA_CHECK(cudaMalloc(&ptr_, size_bytes_));
    }
    void free() {
        if (ptr_) {
            CUDA_CHECK(cudaFree(ptr_));
            ptr_ = nullptr;
        }
        size_bytes_ = 0;
    }

    void*  ptr_         = nullptr;
    size_t size_bytes_  = 0;
};

// ---------------------------------------------------------------------------
// Argument parsing
// ---------------------------------------------------------------------------
struct Args {
    int    rows           = 8192;
    int    cols           = 8192;
    int    threads_per_block = 256;
    int    rows_per_block     = 1;
    int    warmup_iters     = 5;
    int    timed_iters      = 20;
    uint32_t seed          = 42;
    float  tolerance_abs   = 1e-2f;   // max absolute error allowed
    float  tolerance_rel   = 1e-2f;   // max relative error allowed
    bool   verbose         = true;
    bool   help            = false;
};

static void print_usage(const char* prog) {
    printf("Usage: %s [options]\n", prog);
    printf("\nOptions:\n");
    printf("  --rows N              Number of rows (default: 8192)\n");
    printf("  --cols N              Number of columns (default: 8192)\n");
    printf("  --threads N           Threads per block (default: 256)\n");
    printf("  --rows-per-block N    Rows processed per block (default: 1)\n");
    printf("  --warmup N            Warm-up iterations (default: 5)\n");
    printf("  --iters N             Timed iterations (default: 20)\n");
    printf("  --seed N              RNG seed (default: 42)\n");
    printf("  --tol-abs F           Max absolute error tolerance (default: 1e-2)\n");
    printf("  --tol-rel F           Max relative error tolerance (default: 1e-2)\n");
    printf("  --quiet               Suppress verbose output\n");
    printf("  --help                Show this message\n");
}

static Args parse_args(int argc, char** argv) {
    Args args;
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        auto next = [&]() -> std::string {
            if (i + 1 >= argc) {
                fprintf(stderr, "Error: missing value for %s\n", arg.c_str());
                exit(EXIT_FAILURE);
            }
            return argv[++i];
        };

        if (arg == "--help" || arg == "-h") {
            args.help = true;
        } else if (arg == "--rows") {
            args.rows = std::stoi(next());
        } else if (arg == "--cols") {
            args.cols = std::stoi(next());
        } else if (arg == "--threads") {
            args.threads_per_block = std::stoi(next());
        } else if (arg == "--rows-per-block") {
            args.rows_per_block = std::stoi(next());
        } else if (arg == "--warmup") {
            args.warmup_iters = std::stoi(next());
        } else if (arg == "--iters") {
            args.timed_iters = std::stoi(next());
        } else if (arg == "--seed") {
            args.seed = static_cast<uint32_t>(std::stoul(next()));
        } else if (arg == "--tol-abs") {
            args.tolerance_abs = std::stof(next());
        } else if (arg == "--tol-rel") {
            args.tolerance_rel = std::stof(next());
        } else if (arg == "--quiet") {
            args.verbose = false;
        } else {
            fprintf(stderr, "Unknown option: %s\n", arg.c_str());
            print_usage(argv[0]);
            exit(EXIT_FAILURE);
        }
    }
    return args;
}

// ---------------------------------------------------------------------------
// Deterministic input generation (host-side, seeded)
// ---------------------------------------------------------------------------
// Generates values in [-1, 1] using a splitmix64-based generator for
// reproducibility across runs and platforms.
static void generate_input(std::vector<float>& out, int rows, int cols, uint32_t seed) {
    out.resize(static_cast<size_t>(rows) * cols);
    // SplitMix64 for deterministic, high-quality pseudo-random floats
    uint64_t state = seed;
    auto splitmix64 = [&state]() -> uint64_t {
        state += 0x9E3779B97F4A7C15ULL;
        uint64_t z = state;
        z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
        z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
        return z ^ (z >> 31);
    };
    for (auto& v : out) {
        // Map to [-1, 1]
        uint64_t r = splitmix64();
        double u = static_cast<double>(r) / static_cast<double>(UINT64_MAX);
        v = static_cast<float>(2.0 * u - 1.0);
    }
}

// ---------------------------------------------------------------------------
// Convert float vector to BF16 vector (for device upload)
// ---------------------------------------------------------------------------
static std::vector<__nv_bfloat16> to_bf16(const std::vector<float>& f) {
    std::vector<__nv_bfloat16> b(f.size());
    for (size_t i = 0; i < f.size(); ++i) {
        b[i] = __float2bfloat16(f[i]);
    }
    return b;
}

// ---------------------------------------------------------------------------
// CUDA kernel: row-wise L2 normalization (in-place)
//
// Layout: row-major, element (i,j) at index i*cols + j.
//
// Strategy:
//   - Each block handles ROWS_PER_BLOCK consecutive rows.
//   - For each row, threads cooperatively compute sum-of-squares in FP32
//     using a tree reduction in shared memory.
//   - After obtaining the norm, each thread normalizes its assigned elements.
//   - Elements are accessed in a strided pattern: thread t handles columns
//     t, t+blockDim.x, t+2*blockDim.x, ...
//   - Vectorization: we load/store 4 BF16 values (8 bytes) per thread when
//     possible, using __nv_bfloat162 pairs for 2-wide access.
//
// Grid:  ceil(rows / ROWS_PER_BLOCK)
// Block: THREADS_PER_BLOCK
// ---------------------------------------------------------------------------

template <int THREADS, int ROWS_PER_BLOCK>
__global__ void row_normalize_kernel(
    __nv_bfloat16* __restrict__ mat,   // in/out, row-major
    int rows,
    int cols)
{
    // Shared memory for reduction (one float per thread)
    __shared__ float sdata[THREADS];
    __shared__ float snorm[ROWS_PER_BLOCK];  // final norms for each row in this block

    const int tid   = threadIdx.x;
    const int bid   = blockIdx.x;
    const int nRows = rows;
    const int nCols = cols;

    // This block handles rows: [bid*ROWS_PER_BLOCK, (bid+1)*ROWS_PER_BLOCK)
    const int rowStart = bid * ROWS_PER_BLOCK;
    const int rowEnd   = min(rowStart + ROWS_PER_BLOCK, nRows);

    for (int localRow = 0; localRow < ROWS_PER_BLOCK; ++localRow) {
        const int row = rowStart + localRow;
        if (row >= nRows) break;

        const __nv_bfloat16* rowPtr = mat + static_cast<size_t>(row) * nCols;

        // ---- Phase 1: Compute sum-of-squares via cooperative reduction ----
        float localSum = 0.0f;

        // Strided access: thread t handles columns t, t+THREADS, t+2*THREADS, ...
        // Use 2-wide loads (__nv_bfloat162) for better memory throughput.
        // cols must be even for 2-wide; we handle odd tail separately.
        const int stride = THREADS;
        int col = tid;

        // Process pairs of BF16 (each __nv_bfloat162 = 2 elements)
        // Number of 2-element chunks per thread:
        //   total elements = cols
        //   elements per thread (strided) = ceil(cols / THREADS)
        //   pairs per thread = ceil(elements_per_thread / 2)
        // We just loop while col < cols, stepping by 2*stride.

        while (col < nCols) {
            // Load up to 2 BF16 values
            float a = 0.0f, b = 0.0f;
            if (col + 1 < nCols) {
                // 2-wide load
                __nv_bfloat162 pair = *reinterpret_cast<const __nv_bfloat162*>(rowPtr + col);
                a = __bfloat162float(pair.x);
                b = __bfloat162float(pair.y);
            } else {
                // Odd tail: single element
                a = __bfloat162float(rowPtr[col]);
            }
            localSum += a * a + b * b;
            col += 2 * stride;
        }

        sdata[tid] = localSum;
        __syncthreads();

        // Tree reduction
        for (int s = THREADS / 2; s > 0; s >>= 1) {
            if (tid < s) {
                sdata[tid] += sdata[tid + s];
            }
            __syncthreads();
        }

        // Thread 0 has the total sum-of-squares
        if (tid == 0) {
            float ssq = sdata[0];
            // Guard against zero-norm rows (all zeros)
            float norm = (ssq > 0.0f) ? rsqrtf(ssq) : 0.0f;
            snorm[localRow] = norm;
        }
        __syncthreads();

        // ---- Phase 2: Normalize and write back ----
        float invNorm = snorm[localRow];

        col = tid;
        while (col < nCols) {
            if (col + 1 < nCols) {
                __nv_bfloat162 pair = *reinterpret_cast<const __nv_bfloat162*>(rowPtr + col);
                float fa = __bfloat162float(pair.x) * invNorm;
                float fb = __bfloat162float(pair.y) * invNorm;
                __nv_bfloat162 outPair;
                outPair.x = __float2bfloat16(fa);
                outPair.y = __float2bfloat16(fb);
                *reinterpret_cast<
