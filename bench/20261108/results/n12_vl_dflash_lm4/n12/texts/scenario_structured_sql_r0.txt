BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'streaming')),
    token_count INT NOT NULL CHECK (token_count > 0),
    description TEXT
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE,
    status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    latency_ms NUMERIC(10, 2) NOT NULL CHECK (latency_ms >= 0),
    tokens_per_second NUMERIC(10, 2) NOT NULL CHECK (tokens_per_second >= 0),
    mtp_acceptance_rate NUMERIC(5, 4) CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    context_retention_score NUMERIC(5, 4) CHECK (context_retention_score BETWEEN 0 AND 1),
    recorded_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE (run_id, request_index)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_status ON runs(status);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX IF NOT EXISTS idx_metrics_mtp ON request_metrics(mtp_acceptance_rate);

CREATE OR REPLACE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.status,
    COUNT(met.metric_id) AS total_requests,
    AVG(met.tokens_per_second) AS avg_tps,
    MIN(met.latency_ms) AS min_latency_ms,
    MAX(met.latency_ms) AS max_latency_ms,
    AVG(met.mtp_acceptance_rate) AS avg_mtp_acceptance,
    AVG(met.context_retention_score) AS avg_retention
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
GROUP BY r.run_id, m.name, f.kind, r.status;

CREATE OR REPLACE VIEW v_model_performance AS
SELECT 
    m.model_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(met.tokens_per_second) AS avg_tps,
    STDDEV(met.tokens_per_second) AS tps_stddev,
    AVG(met.mtp_acceptance_rate) AS avg_mtp_acceptance,
    AVG(met.context_retention_score) AS avg_retention
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
WHERE r.status = 'completed'
GROUP BY m.model_id, m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_latency_ms NUMERIC,
    p_tokens_per_second NUMERIC,
    p_mtp_acceptance_rate NUMERIC,
    p_context_retention_score NUMERIC
) RETURNS request_metrics AS $$
DECLARE
    v_metric request_metrics;
BEGIN
    INSERT INTO request_metrics (
        run_id,
        request_index,
        latency_ms,
        tokens_per_second,
        mtp_acceptance_rate,
        context_retention_score
    )
    VALUES (
        p_run_id,
        p_request_index,
        p_latency_ms,
        p_tokens_per_second,
        p_mtp_acceptance_rate,
        p_context_retention_score
    )
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        latency_ms = EXCLUDED.latency_ms,
        tokens_per_second = EXCLUDED.tokens_per_second,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        context_retention_score = EXCLUDED.context_retention_score,
        recorded_at = NOW()
    RETURNING * INTO v_metric;
    
    RETURN v_metric;
END;
$$ LANGUAGE plpgsql;
