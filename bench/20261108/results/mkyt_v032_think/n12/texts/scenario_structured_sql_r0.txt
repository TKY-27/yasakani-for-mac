-- tables

CREATE TABLE models (
    model_id serial primary key,
    model_name text not null,
    provider text not null default 'local',
    architecture text not null default 'decoder',
    base_context_tokens integer not null default 8192 check (base_context_tokens > 0),
    created_at timestamptz not null default now()
);
ALTER TABLE models ADD CONSTRAINT uq_models_name UNIQUE (model_name);

CREATE TABLE fixtures (
 fixture_id serial primary key,
 fixture_name text not null,
 fixture_kind text not null check (fixture_kind in ('short','long','multi_turn','retrieval')),
 description text not null default '',
 created_at timestamptz not null default now()
);
ALTER TABLE fixtures ADD CONSTRAINT uq_fixtures_name UNIQUE (fixture_name);

CREATE TABLE runs (
 run_id serial primary key,
 model_id integer not null references models(model_id) on delete restrict,
 fixture_id integer not null references fixtures(fixture_id) on delete restrict,
 run_ts timestamptz not null default now(),
 environment text not null default 'dev' check (environment in ('dev','staging','prod')),
 batch_size integer not null default 1 check (batch_size >= 1),
 max_output_tokens integer not null default 256 check (max_output_tokens > 0),
 seed integer not null default 42 check (seed >= 0),
 created_at timestamptz not null default now(),
 constraint uq_runs_natural_key unique (model_id, fixture_id, run_ts)
);

CREATE TABLE request_metrics (
 metric_id serial primary key,
 run_id integer not null references runs(run_id) on delete cascade,
 request_seq integer not null default 0 check (request_seq >= 0),
 prompt_tokens integer not null check (prompt_tokens >= 0),
 completion_tokens integer not null check (completion_tokens >= 0),
 latency_ms numeric(12,3) not null check (latency_ms >= 0),
 ttft_ms numeric(12,3) not null check (ttft_ms >= 0),
 throughput_tps numeric(12,3) not null check (throughput_tps >= 0),
 mtp_acceptance numeric(5,4) not null default 0 check (mtp_acceptance between 0 and 1),
 retention_score numeric(5,4) check (retention_score is null or retention_score between 0 and 1),
 cache_hit boolean not null default false,
 created_at timestamptz not null default now(),
 constraint uq_request_metrics_run_seq unique (run_id, request_seq)
);
