namespace {

constexpr int kVecBytes = 16;
constexpr int kElemsPerVec = kVecBytes / sizeof(__nv_bfloat16); // 8

struct alignas(kVecBytes) BF16x8 {
  __nv_bfloat162 data[kElemsPerVec / 2];
};

__device__ __forceinline__ float bf16x8_sumsq(const BF16x8& v) {
  float acc = 0.f;
  #pragma unroll
  for (int i = 0; i < kElemsPerVec/2; ++i) {
    float2 f = __bfloat1622float2(v.data[i]);
    acc = fmaf(f.x, f.x, acc);
    acc = fmaf(f.y, f.y, acc);
  }
  return acc;
}
