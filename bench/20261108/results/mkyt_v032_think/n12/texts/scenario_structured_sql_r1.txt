CREATE TABLE models (
  id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name text NOT NULL UNIQUE,
  architecture text NOT NULL DEFAULT 'transformer',
  max_context_tokens integer NOT NULL CHECK (max_context_tokens > 0),
  supports_mtp boolean NOT NULL DEFAULT false CHECK (supports_mtp IN (true, false)),
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE fixtures (
  id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name text NOT NULL UNIQUE,
  kind text NOT NULL CHECK (kind IN ('short_chat','medium_chat','long_context','stream_stress')),
  prompt_length_tokens integer NOT NULL CHECK (prompt_length_tokens > 0),
  completion_tokens integer NOT NULL CHECK (completion_tokens > 0),
  temperature numeric(4,2) NOT NULL DEFAULT 0.0 CHECK (temperature >= 0 AND temperature <= 2),
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE runs (
  id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  model_id integer NOT NULL REFERENCES models(id) ON DELETE CASCADE,
  fixture_id integer NOT NULL REFERENCES fixtures(id) ON DELETE CASCADE,
  seed integer NOT NULL DEFAULT 17,
  started_at timestamptz NOT NULL DEFAULT now(),
  finished_at timestamptz,
  status text NOT NULL DEFAULT 'completed' CHECK (status IN ('completed','failed','cancelled')),
  notes text,
  CONSTRAINT runs_model_fixture_seed_start_unique UNIQUE (model_id, fixture_id, seed, started_at) -- for upsert? maybe
);

CREATE TABLE request_metrics (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  run_id integer NOT NULL REFERENCES runs(id) ON DELETE CASCADE,
  request_seq integer NOT NULL CHECK (request_seq >= 0),
  request_started_at timestamptz NOT NULL DEFAULT now(),
  prompt_tokens integer NOT NULL CHECK (prompt_tokens > 0),
  completion_tokens integer NOT NULL CHECK (completion_tokens >= 0),
  latency_ms numeric(12,3) NOT NULL CHECK (latency_ms > 0),
  first_token_latency_ms numeric(12,3) CHECK (first_token_latency_ms > 0),
  throughput_tps numeric(12,3) NOT NULL CHECK (throughput_tps >= 0),
  mtp_proposed_tokens integer NOT NULL DEFAULT 0 CHECK (mtp_proposed_tokens >= 0),
  mtp_accepted_tokens integer NOT NULL DEFAULT 0 CHECK (mtp_accepted_tokens <= mtp_proposed_tokens),
  long_context_retention_pct numeric(5,2) CHECK (long_context_retention_pct >= 0 AND long_context_retention_pct <= 100),
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT request_metrics_run_request_unique UNIQUE (run_id, request_seq)
);
