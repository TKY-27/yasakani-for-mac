Function:
CREATE OR REPLACE FUNCTION upsert_request_metric(
  p_run_id integer,
  p_sequence_no integer,
  p_input_tokens integer,
  p_output_tokens integer,
  p_ttft_ms numeric,
  p_e2e_ms numeric,
  p_tokens_per_second numeric,
  p_mtp_accepted_tokens integer DEFAULT 0,
  p_status text DEFAULT 'success'
) RETURNS request_metrics
LANGUAGE plpgsql
AS $$
DECLARE
  rec request_metrics;
BEGIN
  IF p_run_id IS NULL OR p_sequence_no IS NULL THEN
    RAISE EXCEPTION 'run_id and sequence_no are required';
  END IF;
  IF p_sequence_no <= 0 THEN RAISE EXCEPTION 'sequence_no must be positive'; END IF;
  IF NOT EXISTS (SELECT 1 FROM runs WHERE run_id = p_run_id) THEN
    RAISE EXCEPTION 'unknown run: %', p_run_id;
  END IF;
  IF p_input_tokens < 0 OR p_output_tokens < 0 OR p_ttft_ms < 0 OR p_e2e_ms < 0 OR p_tokens_per_second < 0 THEN
    RAISE EXCEPTION 'non-negative metric values required';
  END IF;
  IF p_mtp_accepted_tokens < 0 OR p_mtp_accepted_tokens > p_output_tokens THEN
    RAISE EXCEPTION 'mtp_accepted_tokens must be between 0 and output_tokens';
  END IF;
  IF p_status IS NULL OR p_status NOT IN ('success','error','timeout') THEN
    RAISE EXCEPTION 'invalid status';
  END IF;

  BEGIN
    INSERT INTO request_metrics (
      run_id, sequence_no, input_tokens, output_tokens, ttft_ms, e2e_ms, tokens_per_second,
      mtp_accepted_tokens, status
    )
    VALUES (
      p_run_id, p_sequence_no, p_input_tokens, p_output_tokens, p_ttft_ms, p_e2e_ms, p_tokens_per_second,
      coalesce(p_mtp_accepted_tokens,0), p_status
    )
    ON CONFLICT (run_id, sequence_no) DO UPDATE SET
      input_tokens = EXCLUDED.input_tokens,
      output_tokens = EXCLUDED.output_tokens,
      ttft_ms = EXCLUDED.ttft_ms,
      e2e_ms = EXCLUDED.e2e_ms,
      tokens_per_second = EXCLUDED.tokens_per_second,
      mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens,
      status = EXCLUDED.status,
      created_at = now()
    RETURNING * INTO rec;
  EXCEPTION
    WHEN check_violation THEN
      RAISE EXCEPTION 'metric constraint violation for run %, sequence %', p_run_id, p_sequence_no;
    WHEN unique_violation THEN
      -- not expected because on conflict, but handle
      RAISE EXCEPTION 'unique violation for run %, sequence %', p_run_id, p_sequence_no;
  END;

  RETURN rec;
END;
$$;
