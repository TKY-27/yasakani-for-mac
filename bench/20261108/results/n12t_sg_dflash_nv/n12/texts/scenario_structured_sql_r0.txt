CREATE TABLE models (
  model_id integer PRIMARY KEY,
  model_code text NOT NULL UNIQUE,
  model_name text NOT NULL,
  vendor text NOT NULL,
  context_length integer NOT NULL CHECK (context_length > 0),
  quantization text NOT NULL DEFAULT 'bf16' CHECK (quantization <> ''),
  mtp_default_enabled boolean NOT NULL DEFAULT false
);

CREATE TABLE fixtures (
  fixture_id integer PRIMARY KEY,
  fixture_code text NOT NULL UNIQUE,
  fixture_kind text NOT NULL CHECK (fixture_kind IN ('chat','rag','agent','long_context')),
  description text NOT NULL,
  typical_input_tokens integer NOT NULL CHECK (typical_input_tokens > 0),
  typical_output_tokens integer NOT NULL CHECK (typical_output_tokens > 0)
);

CREATE TABLE runs (
  run_id integer PRIMARY KEY,
  model_id integer NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
  fixture_id integer NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
  run_started_at timestamptz NOT NULL DEFAULT now(),
  run_finished_at timestamptz,
  status text NOT NULL DEFAULT 'completed' CHECK (status IN ('completed','failed','cancelled')),
  server_host text NOT NULL,
  mtp_enabled boolean NOT NULL DEFAULT false,
  mtp_draft_tokens integer NOT NULL DEFAULT 0 CHECK (mtp_draft_tokens >= 0),
  batch_size integer NOT NULL CHECK (batch_size > 0),
  request_count integer NOT NULL DEFAULT 0 CHECK (request_count >= 0),
  CHECK (run_finished_at IS NULL OR run_finished_at >= run_started_at),
  UNIQUE (model_id, fixture_id, run_started_at)
);

CREATE TABLE request_metrics (
  metric_id bigserial PRIMARY KEY,
  run_id integer NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
  seq_no integer NOT NULL CHECK (seq_no > 0),
  request_started_at timestamptz NOT NULL,
  request_finished_at timestamptz,
  client_id text NOT NULL,
  input_tokens integer NOT NULL CHECK (input_tokens >= 0),
  output_tokens integer NOT NULL CHECK (output_tokens >= 0),
  prefill_tokens integer NOT NULL CHECK (prefill_tokens >= 0),
  decode_tokens integer NOT NULL CHECK (decode_tokens >= 0),
  first_token_latency_ms numeric(12,3) CHECK (first_token_latency_ms IS NULL OR first_token_latency_ms >= 0),
  total_latency_ms numeric(12,3) CHECK (total_latency_ms IS NULL OR total_latency_ms >= 0),
  throughput_tps numeric(12,3) CHECK (throughput_tps IS NULL OR throughput_tps >= 0),
  mtp_tokens_generated integer NOT NULL DEFAULT 0 CHECK (mtp_tokens_generated >= 0),
  mtp_tokens_accepted integer NOT NULL DEFAULT 0 CHECK (mtp_tokens_accepted >= 0),
