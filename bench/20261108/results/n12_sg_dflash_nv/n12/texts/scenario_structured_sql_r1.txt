BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    description TEXT,
    token_count INT NOT NULL CHECK (token_count > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'cancelled')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    input_tokens INT NOT NULL CHECK (input_tokens > 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DECIMAL(10, 2) NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted BOOLEAN NOT NULL DEFAULT FALSE,
    mtp_rejected BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_started ON runs(started_at);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX IF NOT EXISTS idx_metrics_mtp ON request_metrics(mtp_accepted);

CREATE OR REPLACE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.started_at,
    r.status,
    COUNT(rm.metric_id) AS total_requests,
    SUM(rm.input_tokens) AS total_input_tokens,
    SUM(rm.output_tokens) AS total_output_tokens,
    AVG(rm.latency_ms) AS avg_latency_ms,
    MAX(rm.latency_ms) AS max_latency_ms,
    MIN(rm.latency_ms) AS min_latency_ms,
    COUNT(CASE WHEN rm.mtp_accepted THEN 1 END) AS mtp_accept_count,
    COUNT(CASE WHEN rm.mtp_rejected THEN 1 END) AS mtp_reject_count
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY r.run_id, m.name, f.kind, r.started_at, r.status;

CREATE OR REPLACE VIEW v_model_performance AS
SELECT 
    m.model_id,
    m.name,
    m.parameter_count,
    m.context_length,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.latency_ms) AS avg_latency_ms,
    STDDEV(rm.latency_ms) AS latency_stddev,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY rm.latency_ms) AS p95_latency_ms,
    SUM(rm.output_tokens) * 1.0 / NULLIF(SUM(rm.latency_ms) / 1000.0, 0) AS throughput_tokens_per_sec,
    COUNT(CASE WHEN rm.mtp_accepted THEN 1 END) * 1.0 / NULLIF(COUNT(rm.metric_id), 0) AS mtp_accept_rate
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.model_id, m.name, m.parameter_count, m.context_length;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DECIMAL(10, 2),
    p_mtp_accepted BOOLEAN,
    p_mtp_rejected BOOLEAN
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (
        run_id,
        request_index,
        input_tokens,
        output_tokens,
        latency_ms,
        mtp_accepted,
        mtp_rejected
    ) VALUES (
        p_run_id,
        p_request_index,
        p_input_tokens,
        p_output_tokens,
        p_latency_ms,
        p_mtp_accepted,
        p_mtp_rejected
    )
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_rejected = EXCLUDED.mtp_rejected,
        created_at = CURRENT_TIMESTAMP;
END;
$$ LANGUAGE plpgsql;
