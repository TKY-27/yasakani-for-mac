class BoundedQuantile {

  private cap: number;
  private full = false;
  push(v: number) {
    if (this.buf.length < this.cap) this.buf.push(v);
    else {
      // decimate: keep even indices of existing + v appended?
      // strategy: when full, replace buf with buf.filter((_, i) => i % 2 === 0) then push v

      this.buf.push(v);
    }
  }
}
