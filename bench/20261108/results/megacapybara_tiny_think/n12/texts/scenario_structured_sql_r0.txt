v_run_summary:
SELECT r.run_id, m.model_name, f.fixture_name, f.fixture_kind, r.status, r.started_at, r.ended_at,
COUNT(mq.metric_id) AS request_count,
COUNT(*) FILTER (WHERE mq.status='success') AS successful_requests,
AVG(mq.latency_ms) AS avg_latency_ms,
PERCENTILE_CONT(0.50) WITHIN GROUP (ORDER BY mq.latency_ms) AS p50_latency_ms,
PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY mq.latency_ms) AS p95_latency_ms,
SUM(mq.tokens_generated) FILTER (WHERE mq.status='success') AS total_generated_tokens,
SUM(mq.tokens_generated) FILTER (WHERE mq.status='success') / NULLIF(SUM(mq.latency_ms) FILTER (WHERE mq.status='success'), 0) * 1000 AS tokens_per_sec,
SUM(mq.mtp_accepted_tokens) FILTER (WHERE mq.status='success' AND mq.mtp_proposed_tokens > 0) AS mtp_accepted_tokens,
SUM(mq.mtp_proposed_tokens) FILTER (WHERE mq.status='success' AND mq.mtp_proposed_tokens > 0) AS mtp_proposed_tokens,
CASE WHEN SUM(mq.mtp_proposed_tokens) FILTER ... >0 THEN SUM(mq.mtp_accepted_tokens) FILTER ... / SUM(...) ELSE NULL END AS mtp_accept_rate
FROM runs r JOIN models m ... JOIN fixtures f ... LEFT JOIN request_metrics mq ...
