CREATE TABLE IF NOT EXISTS models (
  model_id integer PRIMARY KEY,
  model_name text NOT NULL,
  model_version text NOT NULL,
  architecture text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (model_name, model_version)
);

CREATE TABLE IF NOT EXISTS fixtures (
  fixture_id integer PRIMARY KEY,
  fixture_name text NOT NULL,
  fixture_kind text NOT NULL CHECK (fixture_kind IN ('long_context','code_completion','summarization','chat')),
  context_length integer NOT NULL CHECK (context_length > 0),
  source text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (fixture_name)
);

CREATE TABLE IF NOT EXISTS runs (
  run_id integer PRIMARY KEY,
  model_id integer NOT NULL REFERENCES models(model_id) ON DELETE RESTRICT,
  fixture_id integer NOT NULL REFERENCES fixtures(fixture_id) ON DELETE RESTRICT,
  started_at timestamptz NOT NULL,
  mtp_depth smallint NOT NULL DEFAULT 0 CHECK (mtp_depth IN (0,1,2,3)),
  batch_size smallint NOT NULL DEFAULT 1 CHECK (batch_size > 0),
  status text NOT NULL DEFAULT 'completed' CHECK (status IN ('completed','failed')),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (model_id, fixture_id, started_at, mtp_depth, batch_size) maybe too strict but okay if timestamps distinct. Could omit unique.
);

CREATE TABLE IF NOT EXISTS request_metrics (
  request_metric_id integer PRIMARY KEY,
  run_id integer NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
  request_id text NOT NULL,
  prompt_tokens integer NOT NULL CHECK (prompt_tokens >= 0),
  completion_tokens integer NOT NULL CHECK (completion_tokens >= 0),
  latency_ms numeric(12,3) NOT NULL CHECK (latency_ms >= 0),
  mtp_proposed smallint NOT NULL DEFAULT 0 CHECK (mtp_proposed >= 0),
  mtp_accepted smallint NOT NULL DEFAULT 0 CHECK (mtp_accepted >= 0 AND mtp_accepted <= mtp_proposed),
  mtp_accept_rate numeric(6,4) NOT NULL DEFAULT 0 CHECK (mtp_accept_rate >= 0 AND mtp_accept_rate <= 1),
  status text NOT NULL DEFAULT 'ok' CHECK (status IN ('ok','error','timeout')),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (run_id, request_id)
);

Views:
CREATE OR REPLACE VIEW v_run_throughput AS
SELECT r.run_id, m.model_name, m.model_version, f.fixture_name, f.fixture_kind,
 r.started_at, r.mtp_depth, r.batch_size, r.status,
 count(*) as request_count,
 sum(m.prompt_tokens) as total_prompt_tokens,
 sum(m.completion_tokens) as total_completion_tokens,
 sum(m.latency_ms) as total_latency_ms,
 (sum(m.completion_tokens) * 1000.0) / NULLIF(sum(m.latency_ms), 0) as completion_tokens_per_sec,
 avg(m.latency_ms) as avg_latency_ms,
 stddev_samp(m.latency_ms) as latency_stddev_ms,
 (stddev_samp(m.latency_ms) / NULLIF(avg(m.latency_ms), 0)) as latency_cv,
 avg(m.mtp_accept_rate) as avg_mtp_accept_rate,
 sum(m.mtp_accepted)::numeric / NULLIF(sum(m.mtp_proposed),0) as total_mtp_accept_rate
FROM runs r
JOIN models m ON m.model_id = r.model_id
JOIN fixtures f ON f.fixture_id = r.fixture_id
JOIN request_metrics mtr ON mtr.run_id = r.run_id
GROUP BY r.run_id, m.model_name, m.model_version, f.fixture_name, f.fixture_kind, r.started_at, r.mtp_depth, r.batch_size, r.status;

 percentile_cont(0.5) WITHIN GROUP (ORDER BY mtr.mtp_accept_rate) as p50_mtp_accept_rate,
 percentile_cont(0.95) WITHIN GROUP (ORDER BY mtr.latency_ms) as p95_latency_ms,
 sum(mtr.completion_tokens) as total_completion_tokens,
 (sum(mtr.completion_tokens)*1000.0)/NULLIF(sum(mtr.latency_ms),0) as completion_tokens_per_sec,
 (sum(mtr.mtp_accepted)*1.0)/NULLIF(sum(mtr.mtp_proposed),0) as total_mtp_accept_rate
FROM models m JOIN runs r ... JOIN fixtures f ... JOIN request_metrics mtr ...
WHERE mtr.status = 'ok'
GROUP BY m.model_name, m.model_version, f.fixture_kind, f.context_length;

  END IF;
  IF NOT EXISTS (SELECT 1 FROM runs WHERE run_id = p_run_id) THEN
     RAISE EXCEPTION 'run_id % does not exist', p_run_id;
