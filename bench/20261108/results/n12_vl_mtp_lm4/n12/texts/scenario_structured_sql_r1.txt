BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'stress')),
    description TEXT,
    token_count INT NOT NULL CHECK (token_count > 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP WITH TIME ZONE,
    status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    input_tokens INT NOT NULL CHECK (input_tokens > 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms > 0),
    mtp_accepted INT NOT NULL DEFAULT 0 CHECK (mtp_accepted >= 0),
    mtp_total INT NOT NULL DEFAULT 0 CHECK (mtp_total >= 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_runs_started ON runs(started_at);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_total, mtp_accepted);

CREATE OR REPLACE VIEW v_run_summary AS
SELECT
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.status,
    r.started_at,
    r.completed_at,
    COUNT(met.metric_id) AS total_requests,
    SUM(met.input_tokens) AS total_input_tokens,
    SUM(met.output_tokens) AS total_output_tokens,
    AVG(met.latency_ms) AS avg_latency_ms,
    CASE WHEN SUM(met.mtp_total) > 0 THEN SUM(met.mtp_accepted)::DOUBLE PRECISION / SUM(met.mtp_total) ELSE 0 END AS mtp_acceptance_rate
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
GROUP BY r.run_id, m.name, f.kind, r.status, r.started_at, r.completed_at;

CREATE OR REPLACE VIEW v_model_performance AS
SELECT
    m.model_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS run_count,
    COUNT(met.metric_id) AS total_requests,
    SUM(met.output_tokens)::DOUBLE PRECISION / GREATEST(EXTRACT(EPOCH FROM MAX(r.completed_at) - MIN(r.started_at)), 1) AS throughput_tokens_per_sec,
    STDDEV(met.latency_ms) AS latency_stddev,
    AVG(met.latency_ms) AS avg_latency_ms,
    CASE WHEN SUM(met.mtp_total) > 0 THEN SUM(met.mtp_accepted)::DOUBLE PRECISION / SUM(met.mtp_total) ELSE 0 END AS mtp_acceptance_rate
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics met ON r.run_id = met.run_id
WHERE r.status = 'completed'
GROUP BY m.model_id, m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted INT,
    p_mtp_total INT
) RETURNS request_metrics AS $$
DECLARE
    v_metric request_metrics;
BEGIN
    IF p_run_id IS NULL OR p_request_index IS NULL THEN
        RAISE EXCEPTION 'run_id and request_index cannot be NULL';
    END IF;
    
    IF p_input_tokens <= 0 THEN
        RAISE EXCEPTION 'input_tokens must be positive';
    END IF;
    
    IF p_output_tokens < 0 THEN
        RAISE EXCEPTION 'output_tokens cannot be negative';
    END IF;
    
    IF p_latency_ms <= 0 THEN
        RAISE EXCEPTION 'latency_ms must be positive';
    END IF;
    
    IF p_mtp_accepted < 0 OR p_mtp_total < 0 THEN
        RAISE EXCEPTION 'mtp values cannot be negative';
    END IF;
    
    IF p_mtp_accepted > p_mtp_total THEN
        RAISE EXCEPTION 'mtp_accepted cannot exceed mtp_total';
    END IF;
    
    INSERT INTO request_metrics (run_id, request_index, input_tokens, output_tokens, latency_ms, mtp_accepted, mtp_total)
    VALUES (p_run_id, p_request_index, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted, p_mtp_total)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_total = EXCLUDED.mtp_total,
        created_at = CURRENT_TIMESTAMP
    RETURNING * INTO v_metric;
    
    RETURN v_metric;
END;
$$ LANGUAGE plpgsql;
