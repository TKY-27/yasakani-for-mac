CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(kind)
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE(model_id, fixture_id, started_at)
);

CREATE TABLE request_metrics (
    metric_id BIGSERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_id INT NOT NULL,
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted_tokens INT NOT NULL DEFAULT 0 CHECK (mtp_accepted_tokens >= 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(run_id, request_id)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_accepted_tokens);

CREATE VIEW v_model_performance AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(r.run_id) AS total_runs,
    AVG(rm.latency_ms) AS avg_latency_ms,
    AVG(rm.output_tokens) AS avg_output_tokens,
    SUM(rm.output_tokens) AS total_output_tokens,
    AVG(rm.mtp_accepted_tokens) AS avg_mtp_accepted
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE VIEW v_long_context_analysis AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    rm.input_tokens,
    rm.output_tokens,
    rm.latency_ms,
    (rm.output_tokens::FLOAT / NULLIF(rm.input_tokens, 0)) AS output_input_ratio,
    rm.mtp_accepted_tokens
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE rm.input_tokens > 1000
ORDER BY rm.input_tokens DESC;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_id INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted_tokens INT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (
        run_id,
        request_id,
        input_tokens,
        output_tokens,
        latency_ms,
        mtp_accepted_tokens
    ) VALUES (
        p_run_id,
        p_request_id,
        p_input_tokens,
        p_output_tokens,
        p_latency_ms,
        p_mtp_accepted_tokens
    )
    ON CONFLICT (run_id, request_id) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens,
        created_at = CURRENT_TIMESTAMP;
END;
$$ LANGUAGE plpgsql;

SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(*) AS total_requests,
    SUM(rm.output_tokens) AS total_output_tokens,
    AVG(rm.latency_ms) AS avg_latency_ms,
    (SUM(rm.output_tokens)::FLOAT / SUM(rm.latency_ms / 1000.0)) AS tokens_per_second
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind
ORDER BY m.name, f.kind;

GROUP BY m.name, f.kind
ORDER BY m.name, f.kind;
