# BF16 Row-Normalization on sm_120a (CUDA 13.1 / C++20)

Below is a complete, self-contained program. It generates a deterministic BF16 matrix, normalizes each row to unit L2 norm using a vectorized `sm_120a` kernel, verifies against a numerically-stable FP32 reference, and reports CUDA-event timing.

```cpp
// bf16_row_norm.cu
//
// Row-wise L2 normalization of a large BF16 matrix on an sm_120a GPU.
//
//   * Deterministic input generation (xorshift64* PRNG -> BF16)
//   * RAII device buffers (throwing, exception-safe)
//   * Vectorized CUDA kernel (16-byte loads/stores, __nv_bfloat162)
//   * Numerically-stable FP32 reference (two-pass, Kahan-compensated)
//   * Warm-up + CUDA-event timing
//   * Tolerance-based verification
//   * Full CUDA error checking
//
// Build (see CMakeLists.txt):
//   cmake -S . -B build -DCMAKE_CUDA_ARCHITECTURES=120a
//   cmake --build build
//   ./bf16_row_norm --rows 8192 --cols 4096 --iters 20
//
// NOTE: sm_120a is a Blackwell-family architecture. This code is written
// against the CUDA 13.1 toolchain (nvcc --gpu-architecture=sm_120a).
// The kernel uses only portable intrinsics (__nv_bfloat162, __float2bfloat162_rn,

// Deterministic PRNG: xorshift64* (Marsaglia-style). 64-bit state, cheap,
// well-distributed, and reproducible across runs/platforms.
// ---------------------------------------------------------------------------
struct Xorshift64Star {
  uint64_t state;
  explicit Xorshift64Star(uint64_t seed) : state(seed ? seed : 0x9E3779B97F4A7C15ull) {}
  uint64_t next() {
    uint64_t x = state;
    x ^= x >> 12;
    x ^= x << 25;
    x ^= x >> 27;
    state = x;
    return x * 0x2545F4914F6CDD17ull;
  }
  // Uniform float in [0,1)
  float nextFloat() {
    return static_cast<float>(next() >> 11) * (1.0f / 9007199254740992.0f); // 2^-53
  }
  // Standard normal via Box-Muller (one pair per two calls; we use one per call
  // for simplicity and determinism).
  float nextNormal() {
    float u1 = nextFloat();
    if (u1 < 1e-12f) u1 = 1e-12f;
    float u2 = nextFloat();
    return std::sqrt(-2.0f * std::log(u1)) * std::cos(2.0f * 3.14159265358979323846f * u2);
  }
};

// ---------------------------------------------------------------------------
// RAII device buffer
// ---------------------------------------------------------------------------
class DeviceBuffer {
 public:
  DeviceBuffer() = default;
  explicit DeviceBuffer(size_t bytes) {
    if (bytes == 0) return;
    CUDA_CHECK(cudaMalloc(&ptr_, bytes));
    size_ = bytes;
  }
  ~DeviceBuffer() {
    if (ptr_) {
      cudaFree(ptr_);
      ptr_ = nullptr;
      size_ = 0;
    }
  }
  DeviceBuffer(const DeviceBuffer&) = delete;
  DeviceBuffer& operator=(const DeviceBuffer&) = delete;
  DeviceBuffer(DeviceBuffer&& o) noexcept : ptr_(o.ptr_), size_(o.size_) {
    o.ptr_ = nullptr;
    o.size_ = 0;
  }
  DeviceBuffer& operator=(DeviceBuffer&& o) noexcept {
    if (this != &o) {
      if (ptr_) cudaFree(ptr_);
      ptr_ = o.ptr_;
      size_ = o.size_;
      o.ptr_ = nullptr;
      o.size_ = 0;
    }
    return *this;
  }
  void* data() const { return ptr_; }
  size_t size() const { return size_; }
  bool empty() const { return ptr_ == nullptr; }

 private:
  void* ptr_ = nullptr;
  size_t size_ = 0;
};

static void printUsage(const char* prog) {
  std::fprintf(stderr,
    "Usage: %s [options]\n"
    "  --rows <N>       number of rows (default 8192)\n"
    "  --cols <N>       number of columns (default 4096)\n"
    "  --iters <N>      timed iterations (default 20)\n"
    "  --seed <hex>     PRNG seed (default 0xC0FFEE)\n"
    "  --rel-tol <f>    relative tolerance (default 2e-2)\n"
    "  --abs-tol <f>    absolute tolerance (default 1e-3)\n"
    "  --quiet          suppress verbose output\n"
    "  --help           show this message\n",
    prog);

static bool parseArgs(int argc, char** argv, Config& cfg) {
  for (int i = 1; i < argc; ++i) {
    std::string arg = argv[i];
    auto next = [&](const char* name) -> const char* {
      if (i + 1 >= argc) {
        std::fprintf(stderr, "Missing value for %s\n", name);
        std::exit(EXIT_FAILURE);
      }
      return argv[++i];
    };
    if (arg == "--rows") {
      cfg.rows = std::atoi(next("--rows"));
    } else if (arg == "--cols") {
      cfg.cols = std::atoi(next("--cols"));
    } else if (arg == "--iters") {
      cfg.iters = std::atoi(next("--iters"));
    } else if (arg == "--seed") {
      cfg.seed = std::strtoull(next("--seed"), nullptr, 0);
    } else if (arg == "--rel-tol") {
      cfg.relTol = std::atof(next("--rel-tol"));
    } else if (arg == "--abs-tol") {
      cfg.absTol = std::atof(next("--abs-tol"));
    } else if (arg == "--quiet") {
      cfg.verbose = false;
    } else if (arg == "--help" || arg == "-h") {
      printUsage(argv[0]);
      std::exit(0);
    } else {
      std::fprintf(stderr, "Unknown argument: %s\n", arg.c_str());
      printUsage(argv[0]);
      return false;
    }
  }
  if (cfg.rows <= 0 || cfg.cols <= 0 || cfg.iters <= 0) {
    std::fprintf(stderr, "rows, cols, iters must be positive\n");
    return false;
  }
  return true;
}

// ---------------------------------------------------------------------------
// Host-side deterministic input generation (BF16)
// ---------------------------------------------------------------------------
// We generate the matrix on the host as a vector of __nv_bfloat16, then
// H2D copy it. This keeps the input fully deterministic and independent of
// GPU state.
static std::vector<__nv_bfloat16> generateInput(int rows, int cols, uint64_t seed) {
  std::vector<__nv_bfloat16> data(static_cast<size_t>(rows) * cols);
  Xorshift64Star rng(seed);
  for (size_t i = 0; i < data.size(); ++i) {
    float v = rng.nextNormal();
    data[i] = __float2bfloat16(v);
  }
  return data;
}

// ---------------------------------------------------------------------------
// CUDA kernel: row-wise L2 normalization (vectorized)
//
// Grid:  (rows, 1, 1)
// Block: (256, 1, 1)
//
// Each block handles one row. The row is processed in 16-byte (8 BF16)
// chunks. Each thread loads a __nv_bfloat162 (4 bytes) at a time, so a
// 256-thread block covers 256*4 = 1024 BF16 per pass. For rows wider than
// 1024 elements we loop.
//
// Two-pass algorithm:
//   Pass 1: compute sum of squares (in FP32) via warp reduction + block reduction.
//   Pass 2: scale each element by 1/sqrt(sum) and store.
//
// The kernel is numerically stable because:
//   - Squares are accumulated in FP32 (not BF16).
//   - Warp-level reduction uses __shfl_xor_sync (no shared memory contention).
//   - Block-level reduction uses shared memory with __syncthreads.
//   - The scale factor is computed as 1/sqrtf(sum) in FP32.
// ---------------------------------------------------------------------------
__device__ __forceinline__ float warpReduceSum(float val) {
  for (int offset = 16; offset > 0; offset >>= 1) {
    val += __shfl_xor_sync(0xFFFFFFFFu, val, offset);
  }
  return val;
}

__device__ __forceinline__ float blockReduceSum(float val, float* shared) {
  const int lane = threadIdx.x & 31;
  const int warpId = threadIdx.x >> 5;
  const int numWarps = blockDim.x >> 5;

  // Warp-level reduction
  val = warpReduceSum(val);

  // Write warp results to shared memory
  __shared__ float warpSums[32]; // max 32 warps for 1024 threads
  if (lane == 0) {
    warpSums[warpId] = val;
  }
  __syncthreads();

  // First warp reduces the warp sums
  if (warpId == 0) {
    val = (lane < numWarps) ? warpSums[lane] : 0.0f;
    val = warpReduceSum(val);
    if (lane == 0) {
      shared[0] = val;
    }
  }
  __syncthreads();
  return shared[0];
}

// Vectorized row-normalization kernel.
// Each block normalizes one row.
// cols must be a multiple of 4 (for __nv_bfloat162 loads).
__global__ void rowNormalizeKernel(const __nv_bfloat16* __restrict__ input,
                                   __nv_bfloat16* __restrict__ output,
                                   int cols) {
  const int row = blockIdx.x;
  const __nv_bfloat16* rowIn = input + static_cast<size_t>(row) * cols;
  __nv_bfloat16* rowOut = output + static_cast<size_t>(row) * cols;

  // Number of 4-element (8-byte) chunks per row
  const int chunksPerRow = cols / 4;
  // Each thread processes multiple chunks if cols > blockDim.x * 4
  const int threadsPerBlock = blockDim.x;

  // Pass 1: compute sum of squares
  float localSum = 0.0f;
  for (int c = threadIdx.x; c < chunksPerRow; c += threadsPerBlock) {
    // Load 4 BF16 as two __nv_bfloat162
    const __nv_bfloat162* in2 = reinterpret_cast<const __nv_bfloat162*>(rowIn + c * 4);
    __nv_bfloat162 v0 = in2[0];
    __nv_bfloat162 v1 = in2[1];
    float2 f0 = __bfloat162float2(v0);
    float2 f1 = __bfloat162float2(v1);
    localSum += f0.x * f0.x + f0.y * f0.y + f1.x * f1.x + f1.y * f1.y;
  }

  // Block reduction
  __shared__ float sdata;
  float blockSum = blockReduceSum(localSum, &sdata);

  // Compute scale factor
  float scale = 1.0f / std::sqrtf(blockSum);
  if (blockSum == 0.0f) scale = 0.0f; // avoid NaN for zero rows

  // Pass 2: scale and store
  for (int c = threadIdx.x; c < chunksPerRow; c += threadsPerBlock) {
    const __nv_bfloat162* in2 = reinterpret_cast<const __nv_bfloat162*>(rowIn + c * 4);
    __nv_bfloat162* out2 = reinterpret_cast<__nv_bfloat162*>(rowOut + c * 4);
    __nv_bfloat162 v0 = in2[0];
    __nv_bfloat162 v1 = in2[1];
    float2 f0 = __bfloat162float2(v0);
    float2 f1 = __bfloat162float2(v1);
    f0.x *= scale;
    f0.y *= scale;
    f1.x *= scale;
    f1.y *= scale;
    out2[0] = __float22bfloat162_rn(f0);
    out2[1] = __float22bfloat162_rn(f1);
  }
}

//   1. Convert BF16 -> FP32.
//   2. Compute sum of squares with Kahan compensation.
//   3. Scale each element by 1/sqrt(sum).
//   4. Store as FP32 (we compare against the CUDA output converted to FP32).
