    """A single executable job from the manifest."""

    id: str
    name: str
    command: tuple[str, ...]
    requires_gpu: bool = False
    timeout_seconds: float = _DEFAULT_TIMEOUT
    cwd: str | None = None

    @property

        """Deterministic key used for CPU slot accounting."""
        return f"cpu::{self.id}"

    def fingerprint(self) -> str:
        """Stable fingerprint for this job definition."""
        payload = {
            "id": self.id,
            "name": self.name,
            "command": list(self.command),
            "requires_gpu": self.requires_gpu,
            "timeout_seconds": self.timeout_seconds,
            "cwd": self.cwd,
            "env": dict(sorted(self.env.items())),
        }
        blob = json.dumps(payload, sort_keys=True, separators=(",", ":"))
        return hashlib.sha256(blob.encode("utf-8")).hexdigest()

def _require_str(value: Any, key: str, *, allow_empty: bool = False) -> str:
    if not isinstance(value, str):
        raise ManifestError(f"Field {key!r} must be a string")
    if not allow_empty and not value.strip():
        raise ManifestError(f"Field {key!r} must not be empty")
    return value.strip()


def _require_command(value: Any, job_id: str) -> tuple[str, ...]:
    if not isinstance(value, (list, tuple)):
        raise ManifestError(
            f"Job {job_id!r} field 'command' must be a list of strings"
        )
    if not value:
        raise ManifestError(f"Job {job_id!r} field 'command' must not be empty")

    command: list[str] = []
    for part in value:
        if not isinstance(part, str) or not part.strip():
            raise ManifestError(
                f"Job {job_id!r} command entries must be non-empty strings"
            )
        command.append(part)
    return tuple(command)


def _require_timeout(value: Any, job_id: str) -> float:
    if value is None:
        return _DEFAULT_TIMEOUT
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise ManifestError(
            f"Job {job_id!r} field 'timeout_seconds' must be a positive number"
        )
    timeout = float(value)
    if timeout <= 0:
        raise ManifestError(
            f"Job {job_id!r} field 'timeout_seconds' must be greater than zero"
        )
    return timeout


def _job_slug(job_id: str, name: str | None, index: int) -> str:
    if name:

def _load_json(path: Path) -> dict[str, Any]:
    if not path.exists():
        raise ManifestError(f"Manifest file not found: {path}")
    try:
        with path.open("r", encoding="utf-8") as fh:
            data = json.load(fh)
    except json.JSONDecodeError as exc:
        raise ManifestError(f"Invalid JSON manifest: {exc}") from exc
    if not isinstance(data, dict):
        raise ManifestError("Manifest root must be a JSON object")
    return data
