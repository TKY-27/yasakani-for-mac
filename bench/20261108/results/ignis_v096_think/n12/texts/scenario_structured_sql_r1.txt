-- PostgreSQL script: model benchmark / request metrics
-- Dependency order:
--   1) tables
--   2) indexes
--   3) views

--   6) analytical SELECT queries

BEGIN;

-- ---------------------------------------------------------------------
-- Tables
-- ---------------------------------------------------------------------

DROP VIEW IF EXISTS v_run_summary;
DROP VIEW IF EXISTS v_model_fixture_performance;
DROP FUNCTION IF EXISTS upsert_request_metric(
    p_model_id          int,
    p_fixture_id        int,
    p_run_started_at    timestamptz,
    p_request_seq       int,
    p_input_tokens      int,
    p_output_tokens     int,
    p_latency_ms        numeric,
    p_queue_ms          numeric,
    p_prefill_ms        numeric,
    p_decode_ms         numeric,
    p_mtp_proposed      int,
    p_mtp_accepted      int
);

DROP TABLE IF EXISTS request_metrics;
DROP TABLE IF EXISTS runs;
DROP TABLE IF EXISTS fixtures;
DROP TABLE IF EXISTS models;

CREATE TABLE models (
    model_id      smallint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name          text NOT NULL,
    version       text NOT NULL,
    family        text NOT NULL,
    context_limit int NOT NULL,
    notes         text,
    created_at    timestamptz NOT NULL DEFAULT now(),
    updated_at    timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT models_name_version_unique UNIQUE (name, version),
    CONSTRAINT models_context_limit_positive CHECK (context_limit > 0),
    CONSTRAINT models_name_not_empty CHECK (btrim(name) <> '')
);

CREATE TABLE fixtures (
    fixture_id    smallint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    kind          text NOT NULL,
    name          text NOT NULL,
    description   text,
    target_len    int NOT NULL,
    created_at    timestamptz NOT NULL DEFAULT now(),
    updated_at    timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT fixtures_kind_allowed CHECK (
        kind IN ('synthetic', 'long_context', 'mtp', 'adversarial')
    ),
    CONSTRAINT fixtures_target_len_positive CHECK (target_len > 0),
    CONSTRAINT fixtures_name_not_empty CHECK (btrim(name) <> '')
);

CREATE TABLE runs (
    run_id           serial PRIMARY KEY,
    model_id         smallint NOT NULL REFERENCES models (model_id),
    fixture_id       smallint NOT NULL REFERENCES fixtures (fixture_id),
    run_name         text NOT NULL,
    started_at       timestamptz NOT NULL,
    completed_at     timestamptz,
    status           text NOT NULL DEFAULT 'completed',
    concurrency      int NOT NULL DEFAULT 1,
    temperature      numeric NOT NULL DEFAULT 0.0,
    top_p            numeric NOT NULL DEFAULT 1.0,
    notes            text,
    created_at       timestamptz NOT NULL DEFAULT now(),
    updated_at       timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT runs_status_allowed CHECK (
        status IN ('pending', 'running', 'completed', 'failed', 'cancelled')
    ),
    CONSTRAINT runs_concurrency_positive CHECK (concurrency > 0),
    CONSTRAINT runs_temperature_non_negative CHECK (temperature >= 0),
    CONSTRAINT runs_top_p_range CHECK (top_p BETWEEN 0 AND 1),
    CONSTRAINT runs_completed_after_started CHECK (
        completed_at IS NULL OR completed_at >= started_at
    ),
    CONSTRAINT runs_name_not_empty CHECK (btrim(run_name) <> '')
);

CREATE TABLE request_metrics (
    metric_id    bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    run_id       int NOT NULL REFERENCES runs (run_id),
    request_seq  int NOT NULL,
    input_tokens int NOT NULL,
    output_tokens int NOT NULL,
    latency_ms   numeric NOT NULL,
    queue_ms     numeric NOT NULL,
    prefill_ms   numeric NOT NULL,
    decode_ms    numeric NOT NULL,
    mtp_proposed int NOT NULL,
    mtp_accepted int NOT NULL,
    recorded_at  timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT request_metrics_request_seq_positive CHECK (request_seq > 0),
    CONSTRAINT request_metrics_tokens_non_negative CHECK (
        input_tokens >= 0 AND output_tokens >= 0
    ),
    CONSTRAINT request_metrics_latency_non_negative CHECK (
        latency_ms >= 0 AND queue_ms >= 0 AND prefill_ms >= 0 AND decode_ms >= 0
    ),
    CONSTRAINT request_metrics_mtp_non_negative CHECK (
        mtp_proposed >= 0 AND mtp_accepted >= 0
    ),
    CONSTRAINT request_metrics_mtp_accepted_not_greater CHECK (
        mtp_accepted <= mtp_proposed
    ),
    CONSTRAINT request_metrics_latency_components_sum CHECK (
        latency_ms = queue_ms + prefill_ms + decode_ms
    ),
    CONSTRAINT request_metrics_run_seq_unique UNIQUE (run_id, request_seq)
);

CREATE INDEX idx_runs_fixture_id

CREATE INDEX idx_runs_started_at

CREATE INDEX idx_runs_model_fixture_start

CREATE INDEX idx_request_metrics_run_id
    ON request_metrics (run_id);

CREATE INDEX idx_request_metrics_run_request_seq
    ON request_metrics (run_id, request_seq);

CREATE INDEX idx_request_metrics_latency_components
    ON request_metrics (latency_ms DESC, prefill_ms DESC, decode_ms DESC);

CREATE INDEX idx_request_metrics_mtp
    ON request_metrics (mtp_proposed DESC, mtp_accepted DESC);

CREATE VIEW v_run_summary AS
SELECT
    m.model_id,
    m.name AS model_name,
    m.version AS model_version,
    f.fixture_id,
    f.kind AS fixture_kind,
    f.name AS fixture_name,
    r.run_id,
    r.run_name,
    r.started_at,
    r.completed_at,
    r.status,
    r.concurrency,
    COUNT(rm.metric_id) AS request_count,
    COALESCE(SUM(rm.input_tokens), 0) AS total_input_tokens,
    COALESCE(SUM(rm.output_tokens), 0) AS total_output_tokens,
    COALESCE(SUM(rm.output_tokens), 0) * 1.0
        / GREATEST(NULLIF(COUNT(rm.metric_id), 0), 1) AS avg_output_tokens_per_request,
    COALESCE(AVG(rm.latency_ms), 0) AS avg_latency_ms,
    COALESCE(AVG(rm.queue_ms), 0) AS avg_queue_ms,
    COALESCE(AVG(rm.prefill_ms), 0) AS avg_prefill_ms,
    COALESCE(AVG(rm.decode_ms), 0) AS avg_decode_ms,
    COALESCE(SUM(rm.mtp_proposed), 0) AS mtp_proposed_total,
    COALESCE(SUM(rm.mtp_accepted), 0) AS mtp_accepted_total,
    CASE
        WHEN COALESCE(SUM(rm.mtp_proposed), 0) = 0 THEN NULL
        ELSE COALESCE(SUM(rm.mtp_accepted), 0) * 100.0 / SUM(rm.mtp_proposed)
    END AS mtp_accept_rate_pct
FROM runs r
JOIN models m ON m.model_id = r.model_id
JOIN fixtures f ON f.fixture_id = r.fixture_id
LEFT JOIN request_metrics rm ON rm.run_id = r.run_id
GROUP BY
    m.model_id,
    m.name,
    m.version,
    f.fixture_id,
    f.kind,
    f.name,
    r.run_id,
    r.run_name,
    r.started_at,
    r.completed_at,
    r.status,
    r.concurrency;
