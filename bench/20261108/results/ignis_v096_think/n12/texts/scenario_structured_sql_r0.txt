CREATE TABLE models (
    model_id bigserial PRIMARY KEY,
    model_name text NOT NULL UNIQUE CHECK (model_name <> ''),
    vendor text CHECK (vendor IS NULL OR vendor <> ''),
    parameter_size_b float8 CHECK (parameter_size_b IS NULL OR parameter_size_b >= 0),
    max_context_tokens int CHECK (max_context_tokens IS NULL OR max_context_tokens > 0),
    created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE fixtures (
    fixture_id bigserial PRIMARY KEY,
    fixture_kind text NOT NULL CHECK (
        fixture_kind IN (
            'code_completion',
            'summarization',
            'long_context_qa',
            'function_calling'
        )
    ),
    fixture_name text NOT NULL UNIQUE CHECK (fixture_name <> ''),
    default_context_tokens int CHECK (default_context_tokens IS NULL OR default_context_tokens > 0),
    created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE runs (
    run_id bigserial PRIMARY KEY,
    model_id bigint NOT NULL REFERENCES models(model_id) ON DELETE RESTRICT,
    fixture_id bigint NOT NULL REFERENCES fixtures(fixture_id) ON DELETE RESTRICT,
    run_label text NOT NULL CHECK (run_label <> ''),
    run_started_at timestamptz NOT NULL DEFAULT now(),
    batch_size int NOT NULL DEFAULT 1 CHECK (batch_size > 0),
    max_output_tokens int NOT NULL DEFAULT 512 CHECK (max_output_tokens > 0),
    temperature float8 NOT NULL DEFAULT 0.0 CHECK (temperature >= 0 AND temperature <= 2),
    top_p float8 NOT NULL DEFAULT 1.0 CHECK (top_p > 0 AND top_p <= 1),
    UNIQUE (model_id, fixture_id, run_label)
);

CREATE TABLE request_metrics (
    metric_id bigserial PRIMARY KEY,
    run_id bigint NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index int NOT NULL CHECK (request_index >= 0),
    input_tokens int NOT NULL CHECK (input_tokens >= 0),
    output_tokens int NOT NULL CHECK (output_tokens >= 0),
    ttft_ms numeric(10, 3) CHECK (ttft_ms IS NULL OR ttft_ms >= 0),
    e2e_latency_ms numeric(12, 3) CHECK (e2e_latency_ms IS NULL OR e2e_latency_ms >= 0),
    tokens_per_second float8 CHECK (tokens_per_second IS NULL OR tokens_per_second >= 0),
    long_context_retention_pct float8 CHECK (
        long_context_retention_pct IS NULL
        OR (long_context_retention_pct >= 0 AND long_context_retention_pct <= 100)
    ),
    mtp_acceptance_rate float8 CHECK (
        mtp_acceptance_rate IS NULL
        OR (mtp_acceptance_rate >= 0 AND mtp_acceptance_rate <= 1)
    ),
    request_finished_at timestamptz NOT NULL DEFAULT now(),
    UNIQUE (run_id, request_index)
);

CREATE VIEW v_run_summary AS
SELECT
    r.run_id,
    r.run_label,
    r.run_started_at,
    r.batch_size,
    r.max_output_tokens,
    r.temperature,
    r.top_p,
    m.model_id,
    m.model_name,
    f.fixture_id,
    f.fixture_kind,
    f.fixture_name,
    COALESCE(SUM(qm.input_tokens), 0)::bigint AS total_input_tokens,
    COALESCE(SUM(qm.output_tokens), 0)::bigint AS total_output_tokens,
    COALESCE(AVG(qm.tokens_per_second), 0)::numeric(10, 3) AS avg_tokens_per_second,
    COALESCE(percentile_cont(0.95) WITHIN GROUP (ORDER BY qm.tokens_per_second), 0)::numeric(10, 3) AS p95_tokens_per_second,
    COALESCE(AVG(qm.e2e_latency_ms), 0)::numeric(12, 3) AS avg_e2e_latency_ms,
    COALESCE(percentile_cont(0.95) WITHIN GROUP (ORDER BY qm.e2e_latency_ms), 0)::numeric(12, 3) AS p95_e2e_latency_ms,
    COALESCE(AVG(qm.mtp_acceptance_rate), 0)::numeric(8, 4) AS avg_mtp_acceptance_rate,
    COALESCE(AVG(qm.long_context_retention_pct), 0)::numeric(8, 3) AS avg_long_context_retention_pct
FROM runs r
JOIN models m
    ON m.model_id = r.model_id
JOIN fixtures f
    ON f.fixture_id = r.fixture_id
LEFT JOIN request_metrics qm
    ON qm.run_id = r.run_id
GROUP BY
    r.run_id,
    r.run_label,
    r.run_started_at,
    r.batch_size,
    r.max_output_tokens,
    r.temperature,
    r.top_p,
    m.model_id,
    m.model_name,
    f.fixture_id,
    f.fixture_kind,
    f.fixture_name;

CREATE VIEW v_long_context_mtp AS
SELECT
    r.run_id,
    r.run_label,
    r.batch_size,
    m.model_id,
    m.model_name,
    f.fixture_id,
    f.fixture_kind,
    f.fixture_name,
    qm.request_index,
    qm.input_tokens,
    qm.output_tokens,
    qm.tokens_per_second,
    qm.long_context_retention_pct,
    qm.mtp_acceptance_rate,
    CASE
        WHEN qm.input_tokens < 2048 THEN 'short'
        WHEN qm.input_tokens < 8192 THEN 'medium'
        ELSE 'long'
    END AS context_bucket,
    CASE
        WHEN f.fixture_kind = 'long_context_qa' THEN TRUE
        ELSE FALSE
    END AS is_long_context_fixture
FROM runs r
JOIN models m
    ON m.model_id = r.model_id
JOIN fixtures f
    ON f.fixture_id = r.fixture_id
JOIN request_metrics qm
    ON qm.run_id = r.run_id;
