/** Bounded, ordered (FIFO) sample of the most recent values in a group. */
export class ValueWindow {
  private readonly buf: number[];
  private head = 0;
  private size = 0;
  constructor(readonly capacity: number) {
    this.buf = new Array<number>(capacity);
  }
  push(value: number): void {
    const i = (this.head + this.size) % this.capacity;
    this.buf[i] = value;
    if (this.size < this.capacity) {
      this.size += 1;
    } else {
      this.head = (this.head + 1) % this.capacity;
    }
  }
  /** Copies in chronological order (oldest retained first). */
  toChronological(): number[] {
    if (this.size === 0) return [];
    const out: number[] = new Array<number>(this.size);
    for (let i = 0; i < this.size; i += 1) {
      const idx = this.head + i;
      out[i] = this.buf[idx < this.capacity ? idx : idx - this.capacity] as number;
    }
    return out;
  }
}
