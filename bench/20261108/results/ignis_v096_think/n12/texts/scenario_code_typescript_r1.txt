/**
 * Shared, strictly-typed domain models for the metrics gateway.
 * No runtime code in this file — types and constants only.
 */

/** One validated newline-delimited metric record. */
export interface MetricRecord {
  /** Identifies the load-test / eval run that produced the record. */
  readonly runId: string;
  /** Model under test, e.g. "gpt-4o-mini". Grouping key (1 of 2). */
  readonly model: string;
  /** Workload fixture, e.g. "summarize-200". Grouping key (2 of 2). */
  readonly fixture: string;
  /** Observed request latency in milliseconds. */
  readonly latencyMs: number;
  /** Throughput in tokens/second. Present only on non-error requests. */
  readonly tokensPerSec: number | null;
  /** HTTP status observed by the client, 100..599. */
  readonly statusCode: number | null;
  /** Free-form error classification ("timeout", "5xx", ...). */
  readonly errorClass: string | null;
  /** UTC epoch milliseconds of the event. */
  readonly tsMs: number;
}

/** A series of records sharing (model, fixture). */
export interface SeriesStats {
  readonly model: string;
  readonly fixture: string;
  readonly count: number;
  readonly errorCount: number;
  readonly successCount: number;
  readonly latency: {
    readonly count: number;
    readonly mean: number;
    readonly median: number;
    /** Sample standard deviation (n-1). 0 when count < 2. */
    readonly sampleStdDev: number;
    readonly min: number;
    readonly max: number;
  };
  readonly tokensPerSec: {
    readonly count: number;
    readonly mean: number;
    readonly median: number;
    readonly sampleStdDev: number;
    readonly min: number;
    readonly max: number;
  } | null;
}

/** One (model, fixture) pair across the entire store. */
export interface GroupStats {
  readonly model: string;
  readonly fixture: string;
  readonly series: SeriesStats[];
  readonly latency: {
    readonly count: number;
    readonly mean: number;
    readonly median: number;
    readonly sampleStdDev: number;
  };
  readonly tokensPerSec: {
    readonly count: number;
    readonly mean: number;
    readonly median: number;
    readonly sampleStdDev: number;
  } | null;
}

export interface SummaryPayload {
  readonly version: number;
  readonly ingested: number;
  readonly droppedByRetention: number;
  readonly recordLimit: number;
  readonly groups: number;
  readonly series: number;
  readonly activeRuns: number;
  readonly latency: {
    readonly count: number;
    readonly mean: number;
    readonly median: number;
    readonly sampleStdDev: number;
  };
  readonly byModel: Array<{
    readonly model: string;
    readonly records: number;
    readonly fixtures: number;
    readonly latencyMean: number;
    readonly latencyMedian: number;
    readonly latencySampleStdDev: number;
  }>;
  readonly groups_?: never;
  readonly groupsDetailed?: never;
  readonly truncated?: never;
}

/** A single per-line validation problem. */
export interface LineError {
  readonly line: number;
  /** 0-based byte offset of the first offending character in the payload. */
  readonly byteOffset: number;
  /** What the parser hit, e.g. "expected object", "JSON: expected value". */
  readonly kind: string;
  /** Where in the payload the parser was, e.g. "line 3, column 14". */
  readonly at: string;
  /** Human-readable explanation. */
  readonly message: string;
}

/** A fully validated record before insertion. */
export interface AcceptedLine {
  readonly line: number;
  readonly byteOffset: number;
  readonly record: MetricRecord;
}

/** Result of parsing + validating one NDJSON payload. */
export interface IngestResult {
  readonly totalLines: number;
  readonly accepted: AcceptedLine[];
  readonly rejected: LineError[];
}

/** A malformed NDJSON payload (no usable records, or oversized/oversized-line). */
export class MalformedInputError extends Error {
  readonly status: number;
  readonly details: unknown;

  constructor(status: number, message: string, details: unknown = null) {
    super(message);
    this.name = "MalformedInputError";
    this.status = status;
    this.details = details;
  }
}
