```python
"""Data models for manifest and runtime state."""

from __future__ import annotations

import enum
import hashlib
import json
from dataclasses import dataclass, field
from typing import Any


class JobStatus(str, enum.Enum):
    PENDING = "pending"
    RUNNING = "running"
    SUCCEEDED = "succeeded"
    FAILED = "failed"
    CANCELLED = "cancelled"
    SKIPPED = "skipped"

    @classmethod
    def from_value(cls, value: str) -> "JobStatus":
        return cls(value)

    def is_terminal(self) -> bool:
        return self in (
            JobStatus.SUCCEEDED,
            JobStatus.FAILED,
            JobStatus.CANCELLED,
            JobStatus.SKIPPED,
        )


@dataclass(frozen=True, slots=True)
class JobSpec:
    """Immutable description of a job from the manifest."""

    id: str
    name: str
    command: tuple[str, ...]
    env: dict[str, str] = field(default_factory=dict)
    gpu: bool = True
    timeout_seconds: float | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "command": list(self.command),
            "env": dict(self.env),
            "gpu": self.gpu,
            "timeout_seconds": self.timeout_seconds,
            "metadata": dict(self.metadata),
        }


@dataclass(frozen=True, slots=True)
class Manifest:
    """Parsed and validated manifest."""

    schema_version: str
    jobs: tuple[JobSpec, ...]
    redaction_secrets: tuple[str, ...]
    redaction_env_names: tuple[str, ...]
    hash: str

    def job_by_id(self, job_id: str) -> JobSpec | None:
        for job in self.jobs:
            if job.id == job_id:
                return job
        return None

    @staticmethod
    def compute_hash(payload: dict[str, Any]) -> str:
        encoded = json.dumps(payload, sort_keys=True, separators=(",", ":")).encode("utf-8")
        return hashlib.sha256(encoded).hexdigest()
```

"""Manifest loading and validation."""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from gpu_job_runner.exceptions import ManifestError
from gpu_job_runner.models import JobSpec, Manifest


def _as_dict(value: Any, path: str) -> dict[str, Any]:
    if not isinstance(value, dict):
        raise ManifestError(f"{path} must be an object")
    return value


def _as_list(value: Any, path: str) -> list[Any]:
    if not isinstance(value, list):
        raise ManifestError(f"{path} must be a list")
    return value


def _as_str(value: Any, path: str) -> str:
    if not isinstance(value, str):
        raise ManifestError(f"{path} must be a string")
    return value


def _as_bool(value: Any, path: str) -> bool:
    if not isinstance(value, bool):
        raise ManifestError(f"{path} must be a boolean")
    return value


def _as_number(value: Any, path: str) -> float | None:
    if value is None:
        return None
    if isinstance(value, bool):
        raise ManifestError(f"{path} must be a number")
    if isinstance(value, (int, float)):
        return float(value)
    raise ManifestError(f"{path} must be a number")


def _parse_job(raw: Any, index: int) -> JobSpec:
    base = f"jobs[{index}]"
    obj = _as_dict(raw, base)

    job_id = _as_str(obj.get("id"), f"{base}.id")
    name = _as_str(obj.get("name", job_id), f"{base}.name")
    command_raw = _as_list(obj.get("command"), f"{base}.command")
    command = tuple(_as_str(item, f"{base}.command[{i}]") for i, item in enumerate(command_raw))
    if not command:
        raise ManifestError(f"{base}.command must not be empty")

    env_raw = obj.get("env", {})
    env = _as_dict(env_raw, f"{base}.env")
    env_parsed = {}
    for key, value in env.items():
        env_parsed[_as_str(key, f"{base}.env key")] = _as_str(value, f"{base}.env[{key}]")

    gpu = _as_bool(obj.get("gpu", True), f"{base}.gpu")
    timeout_seconds = _as_number(obj.get("timeout_seconds"), f"{base}.timeout_seconds")
    if timeout_seconds is not None and timeout_seconds <= 0:
        raise ManifestError(f"{base}.timeout_seconds must be positive")

    metadata_raw = obj.get("metadata", {})
    metadata = _as_dict(metadata_raw, f"{base}.metadata")

    return JobSpec(
        id=job_id,
        name=name,
        command=command,
        env=env_parsed,
        gpu=gpu,
        timeout_seconds=timeout_seconds,
        metadata=metadata,
    )


def _extract_redaction(root: dict[str
