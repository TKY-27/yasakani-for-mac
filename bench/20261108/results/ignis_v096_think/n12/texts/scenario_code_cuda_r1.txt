bool parse_uint64(const std::string& text, std::uint64_t& value)
{
    std::uint64_t parsed = 0;
    bool any = false;

    for (char raw_char : text) {
        if (std::isdigit(static_cast<unsigned char>(raw_char)) == 0) {
            return false;
        }
        const std::uint64_t digit = static_cast<std::uint64_t>(raw_char - '0');
        if (parsed > (std::numeric_limits<std::uint64_t>::max() - digit) / 10ull) {
            return false;
        }
        parsed = parsed * 10ull + digit;
        any = true;
    }

    if (!any) {
        return false;
    }

    value = parsed;
    return true;
}
