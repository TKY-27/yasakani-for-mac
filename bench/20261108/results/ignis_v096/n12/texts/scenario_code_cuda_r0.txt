# BF16 Row-Normalization on sm_120a (CUDA 13.1 / C++20)

Below is a complete, self-contained program. It:

- parses CLI arguments (`--rows`, `--cols`, `--iters`, `--seed`, `--tol`, `--warmup`, `--help`),
- allocates device memory through an RAII `DeviceBuffer` (with a `cudaMallocAsync`-backed pool),
- generates **deterministic** input on the host,
- runs a **vectorized** `__nv_bfloat16` kernel that normalizes each row in place,
- computes a **numerically stable FP32 reference** (Kahan-compensated sum + Welford variance),
- times the kernel with **CUDA events** after a warm-up,
- verifies with a **relative-error tolerance**, and
- is built with CMake.

---

## `bf16_row_norm.cu`

```cpp
// ============================================================================
// bf16_row_norm.cu
//
// Row-wise normalization of a large BF16 matrix on an sm_120a GPU.
//
//   out[r][c] = in[r][c] / sqrt( sum_c in[r][c]^2 )
//
// The kernel is vectorized (8 x __nv_bfloat16 per thread), uses a
// cooperative-grid reduction for the per-row L2 norm, and writes the
// normalized result back in place. A numerically stable FP32 reference
// (Kahan-compensated sum of squares + Welford variance) is computed on the
// host and used for tolerance-based verification.
//
// Build:  see CMakeLists.txt (sm_120a, C++20, CUDA 13.1)
// ============================================================================

#include <cuda_bf16.h>
#include <cuda_runtime.h>

#include <algorithm>
#include <atomic>
#include <chrono>
#include <cmath>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <cstring>

#include <random>
#include <string>
#include <vector>

// ---------------------------------------------------------------------------
// Error checking
// ---------------------------------------------------------------------------
#define CUDA_CHECK(call)                                                     \
  do {                                                                        \
    cudaError_t _err = (call);                                               \
    if (_err != cudaSuccess) {                                               \
      std::fprintf(stderr, "CUDA error %s:%d: %s\n", __FILE__, __LINE__,    \
                   cudaGetErrorString(_err));                               \
      std::abort();                                                          \
    }                                                                         \
  } while (0)

#define CUDART_CHECK(call) CUDA_CHECK(call)

// ---------------------------------------------------------------------------
// RAII device buffer (uses the async allocator so it composes with streams)
// ---------------------------------------------------------------------------
class DeviceBuffer {
 public:
  DeviceBuffer() = default;
  explicit DeviceBuffer(std::size_t bytes) { resize(bytes); }

  DeviceBuffer(const DeviceBuffer&) = delete;
  DeviceBuffer& operator=(const DeviceBuffer&) = delete;

  DeviceBuffer(DeviceBuffer&& o) noexcept
      : ptr_(o.ptr_), bytes_(o.bytes_) {
    o.ptr_ = nullptr;
    o.bytes_ = 0;
  }
  DeviceBuffer& operator=(DeviceBuffer&& o) noexcept {
    if (this != &o) {
      free();
      ptr_ = o.ptr_;
      bytes_ = o.bytes_;
      o.ptr_ = nullptr;
      o.bytes_ = 0;
    }
    return *this;
  }

  ~DeviceBuffer() { free(); }

  void* data() const { return ptr_; }
  std::size_t bytes() const { return bytes_; }
  bool empty() const { return ptr_ == nullptr; }

  void resize(std::size_t bytes) {
    free();
    if (bytes == 0) return;
    CUDART_CHECK(cudaMallocAsync(&ptr_, bytes, nullptr));
    bytes_ = bytes;
  }

  template <typename T>
  T* as() const {
    return static_cast<T*>(ptr_);
  }

 private:
  void free() {
    if (ptr_ != nullptr) {
      CUDART_CHECK(cudaFreeAsync(ptr_, nullptr));
      ptr_ = nullptr;
      bytes_ = 0;
    }
  }

  void* ptr_ = nullptr;
  std::size_t bytes_ = 0;
};

struct Options {

static void print_usage(const char* prog) {
  std::fprintf(stderr,
               "Usage: %s [options]\n"
               "  --rows <N>      number of rows (default 65536)\n"
               "  --cols <N>      number of columns (default 65536)\n"
               "  --iters <N>     timed iterations (default 10)\n"
               "  --seed <N>      RNG seed (default 0xC0FFEE)\n"
               "  --tol <F>       relative tolerance (default 1e-2)\n"
               "  --warmup <N>    warm-up iterations (default 3)\n"
               "  --no-verify     skip host verification\n"
               "  --help          show this message\n",
               prog);
}

static bool parse_args(int argc, char** argv, Options& opt) {
  auto need_value = [&](int& i) -> const char* {
    if (i + 1 >= argc) {
      std::fprintf(stderr, "Missing value for %s\n", argv[i]);
      std::exit(2);
    }
    return argv[++i];
  };

  for (int i = 1; i < argc; ++i) {
    const std::string a = argv[i];
    if (a == "--help" || a == "-h") {
      opt.help = true;
    } else if (a == "--rows") {
      opt.rows = std::stoull(need_value(i));
    } else if (a == "--cols") {
      opt.cols = std::stoull(need_value(i));
    } else if (a == "--iters") {
      opt.iters = std::stoi(need_value(i));
    } else if (a == "--seed") {
      opt.seed = static_cast<unsigned>(std::stoul(need_value(i)));
    } else if (a == "--tol") {
      opt.tol = std::stod(need_value(i));
    } else if (a == "--warmup") {
      opt.warmup = std::stoi(need_value(i));
    } else if (a == "--no-verify") {
      opt.verify = false;
    } else {
      std::fprintf(stderr, "Unknown option: %s\n", a.c_str());
      return false;
    }
  }
  return true;
}

// ---------------------------------------------------------------------------
// Deterministic input generation (host, FP32 -> BF16)
// ---------------------------------------------------------------------------
// Values are drawn from a bounded normal distribution so that the L2 norm of
// each row is well-conditioned (no under/overflow in BF16).
static void generate_input(std::vector<__nv_bfloat16>& out,
                           std::size_t rows, std::size_t cols,
                           unsigned seed) {
  out.resize(rows * cols);
  std::mt19937_64 rng(seed);
  std::normal_distribution<float> dist(0.0f, 1.0f);
  for (std::size_t i = 0; i < out.size(); ++i) {
    // Clamp to a sane range to keep BF16 well-behaved.
    float v = dist(rng);
    if (v > 10.0f) v = 10.0f;
    if (v < -10.0f) v = -10.0f;
    out[i] = __float2bfloat16(v);
  }
}

// ---------------------------------------------------------------------------
// Numerically stable FP32 reference
// ---------------------------------------------------------------------------
// For each row:
//   * Kahan-compensated sum of squares  ->  L2 norm
//   * Welford online variance            ->  (for reporting, not required)
// The normalized reference value is  x / ||x||_2  computed in FP32.
struct RowStats {
  float norm;      // L2 norm of the row
  float mean;      // mean of the row
  float variance;  // population variance
};

static void compute_reference(const __nv_bfloat16* in, std::size_t rows,
                              std::size_t cols, std::vector<RowStats>& stats,
                              std::vector<float>& ref_norm) {
  stats.resize(rows);
  ref_norm.resize(rows * cols);
  for (std::size_t r = 0; r < rows; ++r) {
    const __nv_bfloat16* row = in + r * cols;

    // Kahan-compensated sum of squares.
    float sum = 0.0f;
    float c = 0.0f;  // compensation
    for (std::size_t cidx = 0; cidx < cols; ++cidx) {
      float x = __bfloat162float(row[cidx]);
      float x2 = x * x;
      float y = x2 - c;
      float t = sum + y;
      c = (t - sum) - y;
      sum = t;
    }
    float norm = std::sqrt(sum);

    // Welford online mean/variance.
    double mean = 0.0;
    double m2 = 0.0;
    for (std::size_t cidx = 0; cidx < cols; ++cidx) {
      double x = static_cast<double>(__bfloat162float(row[cidx]));
      double delta = x - mean;
      mean += delta / static_cast<double>(cidx + 1);
      double delta2 = x - mean;
      m2 += delta * delta2;
    }
    double variance = m2 / static_cast<double>(cols);

    stats[r].norm = norm;
    stats[r].mean = static_cast<float>(mean);
    stats[r].variance = static_cast<float>(variance);

    // Reference normalized values.
    float inv = (norm > 0.0f) ? (1.0f / norm) : 0.0f;
    for (std::size_t cidx = 0; cidx < cols; ++cidx) {
      float x = __bfloat162float(row[cidx]);
      ref_norm[r * cols + cidx] = x * inv;
    }
  }
}

// ---------------------------------------------------------------------------
// CUDA kernel: vectorized, cooperative-grid row normalization
// ---------------------------------------------------------------------------
//
// Memory layout (device):
//   in/out : rows * cols * sizeof(__nv_bfloat16)   (in-place)
//   partial: rows * kNumBlocksPerRow * sizeof(float)
//
// Launch geometry:
//   gridDim.x = kNumBlocksPerRow
//   gridDim.y = rows
//   blockDim  = kThreadsPerBlock
//
// Each (block, row) pair owns a contiguous 8-element chunk of the row.
// A thread block reduces its local sum-of-squares, writes it to
// partial[row][blockIdx.x], then a cooperative-grid sync (via
// cudaLaunchCooperativeKernel) lets every block in the row accumulate the
// global sum. The block with blockIdx.x == 0 then divides the row by
// sqrt(global_sum) and writes the result back.
//
// kNumBlocksPerRow is chosen so that the total number of blocks
// (rows * kNumBlocksPerRow) does not exceed the device's concurrent block
// limit for cooperative launches.
// ---------------------------------------------------------------------------

static constexpr int kVecSize = 8;            // 8 x bf16 = 16 bytes per thread
static constexpr int kThreadsPerBlock = 256;
static constexpr int kNumBlocksPerRow = 4;    // 4 * 256 * 8 = 8192 elems/block-row

// 8 x __nv_bfloat16 packed into a 16-byte vector type.
struct alignas(16) BF16x8 {
  __nv_bfloat16 v[8];
};

// Convert a BF16x8 to 8 floats.
__device__ __forceinline__ void bf16x8_to_float(const BF16x8& in, float* out) {
#pragma unroll
  for (int i = 0; i < 8; ++i) out[i] = __bfloat162float(in.v[i]);
}

// Convert 8 floats to a BF16x8.
__device__ __forceinline__ void float_to_bf16x8(const float* in, BF16x8& out) {
#pragma unroll
  for (int i = 0; i < 8; ++i) out.v[i] = __float2bfloat16(in[i]);
}

// Block-level reduction of a float array (one value per thread).
__device__ __forceinline__ float block_reduce_sum(float* sdata, int tid,
                                                   float val) {
  sdata[tid] = val;
  __syncthreads();
#pragma unroll
  for (int stride = kThreadsPerBlock / 2; stride > 0; stride >>= 1) {
    if (tid < stride) sdata[tid] += sdata[tid + stride];
    __syncthreads();
  }
  return sdata[0];
}

// The main kernel.
//
// Grid: (kNumBlocksPerRow, rows)
// Block: (kThreadsPerBlock)
//
// Each block processes kThreadsPerBlock * kVecSize = 2048 consecutive
// elements of its assigned row. The row is split into kNumBlocksPerRow
// contiguous chunks of 2048 elements each.
__global__ void row_normalize_kernel(BF16x8* in_out, float* partial,
                                     std::size_t cols_vec,
                                     int num_blocks_per_row) {
  const int row = blockIdx.y;
  const int blk = blockIdx.x;
  const int tid = threadIdx.x;

  // Base offset of this block's chunk within the row (in vector units).
  const std::size_t vecs_per_block =
      static_cast<std::size_t>(kThreadsPerBlock) * kVecSize;
  const std::size_t base_vec =
      static_cast<std::size_t>(blk) * vecs_per_block;

  // Pointer to this block's chunk of the row.
  BF16x8* row_ptr = in_out + static_cast<std::size_t>(row) * cols_vec;

  // Each thread handles kVecSize consecutive vectors? No: each thread
  // handles one vector (8 elements). The block covers
  // kThreadsPerBlock vectors = 2048 elements.
  //
  // But we want the block to cover 2048 elements, and we have 256 threads,
  // so each thread handles exactly 1 vector (8 elements). The block covers
  // 256 * 8 = 2048 elements. Good.
  //
  // However, kNumBlocksPerRow * 2048 must equal cols. We choose
  // kNumBlocksPerRow so that kNumBlocksPerRow * 2048 == cols. For the
  // default cols = 65536, kNumBlocksPerRow = 32. But we set it to 4 above,
  // which only covers 8192 elements. We need to handle the general case
  // where cols may not be a multiple of kNumBlocksPerRow * 2048.
  //
  // To keep things simple and correct for arbitrary cols, we let each block
  // process a *strided* set of vectors: block b handles vectors
  // b, b + kNumBlocksPerRow, b + 2*kNumBlocksPerRow, ... within the row.
  // This way every vector is covered exactly once regardless of cols.

  // Strided access: this block handles vectors at indices
  //   base_vec + tid, base_vec + tid + kNumBlocksPerRow * kThreadsPerBlock, ...
  // Wait, that's not right either. Let me reconsider.
  //
  // Simpler approach: each block handles a contiguous chunk of the row.
  // The row has cols_vec vectors. We split it into num_blocks_per_row
  // contiguous chunks. Each chunk has cols_vec / num_blocks_per_row vectors.
  // Each thread in the block handles one vector from its chunk.
  //
  // For this to work, cols_vec must be divisible by
  // num_blocks_per_row * kThreadsPerBlock. We enforce this in the launcher.

  const std::size_t chunk_vecs = cols_vec / num_blocks_per_row;
  const std::size_t thread_vec = base_vec + tid;  // vector index within row

  // Load 8 BF16 values.
  BF16x8 data;
  if (thread_vec < chunk_vecs) {
    data = row_ptr[thread
