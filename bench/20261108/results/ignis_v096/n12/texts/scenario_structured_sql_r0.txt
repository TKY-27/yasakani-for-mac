BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL,
    parameter_count BIGINT NOT NULL CHECK (parameter_count > 0),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CHECK (kind IN ('code', 'math', 'long_context', 'chat'))
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    batch_size INT NOT NULL CHECK (batch_size > 0),
    sequence_length INT NOT NULL CHECK (sequence_length > 0),
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (model_id, fixture_id, batch_size, sequence_length)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    latency_ms FLOAT NOT NULL CHECK (latency_ms > 0),
    tokens_per_second FLOAT NOT NULL CHECK (tokens_per_second > 0),
    mtp_acceptance_rate FLOAT CHECK (mtp_acceptance_rate BETWEEN 0.0 AND 1.0),
    retention_score FLOAT CHECK (retention_score BETWEEN 0.0 AND 1.0),
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);

CREATE VIEW IF NOT EXISTS v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.batch_size,
    r.sequence_length,
    COUNT(*) AS total_requests,
    AVG(rm.latency_ms) AS avg_latency_ms,
    AVG(rm.tokens_per_second) AS avg_tps,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance,
    AVG(rm.retention_score) AS avg_retention
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY r.run_id, m.name, f.kind, r.batch_size, r.sequence_length;

CREATE VIEW IF NOT EXISTS v_model_performance AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.tokens_per_second) AS avg_tps,
    STDDEV(rm.latency_ms) AS latency_stddev,
    MAX(rm.mtp_acceptance_rate) AS max_mtp_acceptance
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_latency_ms FLOAT,
    p_tps FLOAT,
    p_mtp_rate FLOAT,
    p_retention FLOAT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, latency_ms, tokens_per_second, mtp_acceptance_rate, retention_score)
    VALUES (p_run_id, p_request_index, p_latency_ms, p_tps, p_mtp_rate, p_retention)
    ON CONFLICT (run_id, request_index) DO UPDATE SET
        latency_ms = EXCLUDED.latency_ms,
        tokens_per_second = EXCLUDED.tokens_per_second,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        retention_score = EXCLUDED.retention_score;
END;
$$ LANGUAGE plpgsql;
