BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    architecture VARCHAR(100) NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL,
    description TEXT,
    unique_identifier VARCHAR(255) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_runs_status ON runs(status);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id BIGSERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    timestamp_ms INT NOT NULL CHECK (timestamp_ms >= 0),
    tokens_generated INT NOT NULL CHECK (tokens_generated >= 0),
    latency_ms FLOAT NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted_tokens INT NOT NULL DEFAULT 0 CHECK (mtp_accepted_tokens >= 0),
    mtp_proposed_tokens INT NOT NULL DEFAULT 0 CHECK (mtp_proposed_tokens >= 0),
    long_context_retention_score FLOAT CHECK (long_context_retention_score BETWEEN 0.0 AND 1.0)
);

CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_timestamp ON request_metrics(timestamp_ms);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_proposed_tokens, mtp_accepted_tokens);

-- View for aggregated run statistics
CREATE OR REPLACE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.status,
    COUNT(rm.metric_id) AS total_requests,
    SUM(rm.tokens_generated) AS total_tokens,
    AVG(rm.latency_ms) AS avg_latency_ms,
    MAX(rm.long_context_retention_score) AS max_retention_score,
    CASE WHEN SUM(rm.mtp_proposed_tokens) > 0 THEN SUM(rm.mtp_accepted_tokens)::FLOAT / SUM(rm.mtp_proposed_tokens) ELSE 0 END AS mtp_acceptance_rate
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics rm ON r.run_id = rm.run_id
GROUP BY r.run_id, m.name, f.kind, r.status;

-- View for detailed throughput analysis
CREATE OR REPLACE VIEW v_throughput_analysis AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    rm.timestamp_ms,
    rm.tokens_generated,
    rm.latency_ms,
    CASE WHEN rm.latency_ms > 0 THEN rm.tokens_generated::FLOAT / (rm.latency_ms / 1000.0) ELSE 0 END AS tokens_per_second
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
WHERE r.status = 'completed';

-- Transaction-safe upsert function for request metrics
CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_timestamp_ms INT,
    p_tokens_generated INT,
    p_latency_ms FLOAT,
    p_mtp_accepted INT,
    p_mtp_proposed INT,
    p_retention_score FLOAT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, timestamp_ms, tokens_generated, latency_ms, mtp_accepted_tokens, mtp_proposed_tokens, long_context_retention_score)
    VALUES (p_run_id, p_timestamp_ms, p_tokens_generated, p_latency_ms, p_mtp_accepted, p_mtp_proposed, p_retention_score)
    ON CONFLICT (run_id, timestamp_ms) DO UPDATE SET
        tokens_generated = EXCLUDED.tokens_generated,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted_tokens = EXCLUDED.mtp_accepted_tokens,
        mtp_proposed_tokens = EXCLUDED.mtp_proposed_tokens,
        long_context_retention_score = EXCLUDED.long_context_retention_score;
END;
$$ LANGUAGE plpgsql;
