// ============================================================================
// bf16_row_norm.cu
//
// Row-wise L2 normalization of a large BF16 matrix on an sm_120a GPU.
//
//   y[i][j] = x[i][j] / sqrt( sum_j x[i][j]^2 )
//
// The kernel reads/writes the matrix as packed __nv_bfloat16x8 vectors
// (16 bytes per element group), accumulates the row norm in FP32 using a
// two-pass Welford-style reduction for numerical stability, and writes back
// normalized BF16 values. A separate FP32 host reference (computed with
// Kahan-compensated summation) is used to verify the GPU result within a
// tolerance that accounts for BF16 output rounding.
//
// Build (see CMakeLists.txt):
//   cmake -S . -B build && cmake --build build
// Run:
//   ./bf16_row_norm [options]
// ============================================================================

[[noreturn]] static void fail(const char* what, cudaError_t err) {
    std::fprintf(stderr, "CUDA error [%s]: %s\n", what, cudaGetErrorString(err));
    std::exit(EXIT_FAILURE);
}

#define CUDA_CHECK(expr)                                                     \
    do {                                                                     \
        cudaError_t _e = (expr);                                             \
        if (_e != cudaSuccess) fail(#expr, _e);                              \

// ---------------------------------------------------------------------------
// Deterministic PRNG (SplitMix64) — portable, no state beyond one uint64
// ---------------------------------------------------------------------------

struct SplitMix64 {
    uint64_t s;
    explicit SplitMix64(uint64_t seed) : s(seed) {}
    uint64_t next() {
        uint64_t z = (s += 0x9E3779B97F4A7C15ULL);
        z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
        z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
        return z ^ (z >> 31);
    }
};

// ---------------------------------------------------------------------------
// RAII device buffer
// ---------------------------------------------------------------------------

class DeviceBuffer {
public:
    DeviceBuffer() = default;
    explicit DeviceBuffer(size_t bytes) { allocate(bytes); }

    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    DeviceBuffer(DeviceBuffer&& o) noexcept : ptr_(o.ptr_), bytes_(o.bytes_) {
        o.ptr_ = nullptr;
        o.bytes_ = 0;
    }
    DeviceBuffer& operator=(DeviceBuffer&& o) noexcept {
        if (this != &o) {
            free();
            ptr_ = o.ptr_;
            bytes_ = o.bytes_;
            o.ptr_ = nullptr;
            o.bytes_ = 0;
        }
        return *this;
    }

    ~DeviceBuffer() { free(); }

    void* data() const { return ptr_; }
    size_t bytes() const { return bytes_; }
    bool valid() const { return ptr_ != nullptr; }

private:
    void allocate(size_t bytes) {
        if (bytes == 0) return;
        CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        bytes_ = bytes;
    }
    void free() {
        if (ptr_) {
            CUDA_CHECK(cudaFree(ptr_));
            ptr_ = nullptr;
            bytes_ = 0;
        }
    }

    void* ptr_ = nullptr;
    size_t bytes_ = 0;
};

// ---------------------------------------------------------------------------
// Vectorized BF16 row-normalization kernel
//
// Layout: X is M rows x N columns of __nv_bfloat16, stored row-major.
// Each thread block handles exactly one row. Within a row, threads stride by
// blockDim.x over groups of 8 consecutive BF16 elements (one __nv_bfloat16x8
// = 16 bytes). This gives coalesced 16-byte loads/stores when blockDim.x is
// a multiple of 4 and N is a multiple of 8.
//
// Two passes over the row:
//   Pass 1: compute sum-of-squares in FP32 (Welford-style running mean/variance
//           for numerical stability against catastrophic cancellation).
//   Pass 2: divide each element by sqrt(variance) and write normalized BF16.
//
// Grid:  (N/8 + blockDim.x - 1)/blockDim.x blocks? No — one block per row.
// Block: up to 1024 threads. For very wide rows we rely on grid-stride loops
//       inside the block so a single block can cover arbitrarily many columns.
// ---------------------------------------------------------------------------

__device__ __forceinline__ float bf16_to_float(__nv_bfloat16 h) {
    return __bfloat162float(h);
}

__global__ void row_normalize_bf16(
    const __nv_bfloat16* __restrict__ X,
    __nv_bfloat16* __restrict__ Y,
    int M,
    int N)
{
    // One block per row.
    const int row = blockIdx.x;
    if (row >= M) return;

    const int tid = threadIdx.x;
    const int nthreads = blockDim.x;

    // Number of 8-element groups in this row.
    const int groups = (N + 7) / 8;

    // Pointer to start of this row.
    const __nv_bfloat16* xrow = X + static_cast<size_t>(row) * N;
    __nv_bfloat16* yrow = Y + static_cast<size_t>(row) * N;

    // ---- Pass 1: numerically stable sum-of-squares via Welford ----
    // We accumulate the variance of the values x[j] (i.e., E[x^2] since mean
    // is not needed for L2 norm, but Welford's algorithm still guards against
    // loss of significance when magnitudes vary widely).
    //
    // Standard Welford for sum of squares:
    //   count = 0; mean = 0; m2 = 0;
    //   for each value v:
    //     count++;
    //     delta = v - mean;
    //     mean += delta / count;
    //     delta2 = v - mean;
    //     m2 += delta * delta2;
    //   sum_sq = m2 + mean * mean * count   (== sum(v^2) algebraically)
    //
    // For pure sum-of-squares we can simplify: track running sum with Kahan
    // compensation, which is simpler and equally stable for non-negative terms.
    // Since all terms are >= 0, Kahan summation suffices and avoids the
    // overhead of full Welford. We use Kahan here.

    float sum_sq = 0.0f;
    float c = 0.0f;  // Kahan compensation

    for (int g = tid; g < groups; g += nthreads) {
        const int col = g * 8;
        // Load 8 BF16 values as one 16-byte vector.
        // Guard against out-of-bounds tail (N may not be divisible by 8).
        __nv_bfloat16 vals[8];
        if (col + 8 <= N) {
            const __nv_bfloat16x8* vec = reinterpret_cast<const __nv_bfloat16x8*>(xrow + col);
            __nv_bfloat16x8 v8 = *vec;
            // Unpack
            vals[0] = v8.x; vals[1] = v8.y; vals[2] = v8.z; vals[3] = v8.w;
            vals[4] = v8.u; vals[5] = v8.v; vals[6] = v8.s; vals[7] = v8.t;
        } else {
            for (int k = 0; k < 8; ++k) {
                vals[k] = (col + k < N) ? xrow[col + k] : __float2bfloat16(0.0f);
            }
        }

        #pragma unroll
        for (int k = 0; k < 8; ++k) {
            float f = bf16_to_float(vals[k]);
            float sq = f * f;
            // Kahan compensated addition
            float y = sq - c;
            float t = sum_sq + y;
            c = (t - sum_sq) - y;
            sum_sq = t;
        }
    }

    // Reduce sum_sq across the block.
    // Shared memory for intra-block reduction.
    extern __shared__ float sdata[];
    sdata[tid] = sum_sq;
    __syncthreads();

    // Tree reduction
    for (int stride = nthreads / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            sdata[tid] += sdata[tid + stride];
        }
        __syncthreads();
    }

    // All threads read the final sum from sdata[0].
    float total_sq = sdata[0];

    // Guard against zero-norm rows (all zeros) to avoid division by zero.
    float inv_norm = (total_sq > 0.0f) ? rsqrtf(total_sq) : 0.0f;

    // ---- Pass 2: normalize and write ----
    for (int g = tid; g < groups; g += nthreads) {
        const int col = g * 8;
        __nv_bfloat16 vals[8];
        if (col + 8 <= N) {
            const __nv_bfloat16x8* vec = reinterpret_cast<const __nv_bfloat16x8*>(xrow + col);
            __nv_bfloat16x8 v8 = *vec;
            vals[0] = v8.x; vals[1] = v8.y; vals[2] = v8.z; vals[3] = v8.w;
            vals[4] = v8.u; vals[5] = v8.v; vals[6] = v8.s; vals[7] = v8.t;
        } else {
            for (int k = 0; k < 8; ++k) {
                vals[k] = (col + k < N) ? xrow[col + k] : __float2bfloat16(0.0f);
            }
        }

        __nv_bfloat16 out[8];
        #pragma unroll
        for (int k = 0; k < 8; ++k) {
            float f = bf16_to_float(vals[k]) * inv_norm;
            out[k] = __float2bfloat16(f);
        }

        if (col + 8 <= N) {
            __nv_bfloat16x8 o8;
            o8.x = out[0]; o8.y = out[1]; o8.z = out[2]; o8.w = out[3];
            o8.u = out[4]; o8.v = out[5]; o8.s = out[6]; o8.t = out[7];
            __nv_bfloat16x8* dst = reinterpret_cast<__nv_bfloat16x8*>(yrow + col);
            *dst = o8;
        } else {
            for (int k = 0; k < 8; ++k) {
                if (col + k < N) yrow[col + k] = out[k];
            }
        }
    }
}

// ---------------------------------------------------------------------------
// Host-side FP32 reference (Kahan-compensated sum-of-squares)
// ---------------------------------------------------------------------------

static void reference_row_normalize_f32(
    const std::vector<float>& X,
    std::vector<float>& Y,
    int M,
    int N)
{
    Y.resize(static_cast<size_t>(M) * N);
    for (int i = 0; i < M; ++i) {
        const float* xrow = X.data() + static_cast<size_t>(i) * N;
        float* yrow = Y.data() + static_cast<size_t>(i) * N;

        // Kahan-compensated sum of squares
        float sum_sq = 0.0f;
        float c = 0.0f;
        for (int j = 0; j < N; ++j) {
            float v = xrow[j];
            float sq = v * v;
            float y = sq - c;
            float t = sum_sq + y;
            c = (t - sum_sq) - y;
            sum_sq = t;
        }

        float inv_norm = (sum_sq > 0.0f) ? std::sqrt(1.0f / sum_sq) : 0.0f;
        for (int j = 0; j < N; ++j) {
            yrow[j] = xrow[j] * inv_norm;
        }
    }
}

struct Options {
    int M = 4096;          // rows
    int N = 8192;          // columns
    int iters = 100;       // timed iterations
    int warmup = 10;       // warm-up iterations
    double tol = 1e-2;     // relative tolerance for verification
    uint64_t seed = 0xDEADBEEFCAFEULL;
    bool verbose = true;
};

static void print_usage(const char* prog) {
    std::printf(
        "Usage: %s [options]\n"
        "  --rows <int>        Number of rows M (default 4096)\n"
        "  --cols <int>        Number of columns N (default 8192)\n"
        "  --iters <int>       Timed iterations (default 100)\n"
        "  --warmup <int>      Warm-up iterations (default 10)\n"
        "  --tol <double>      Relative tolerance (default 1e-2)\n"
        "  --seed <uint64>     RNG seed (default 0xDEADBEEFCAFE)\n"
        "  --quiet             Suppress verbose output\n"
        "  --help              Show this message\n",
        prog);
}

static bool parse_args(int argc, char** argv, Options& opt) {
    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        auto need_value = [&](const char* name) -> std::string {
            if (i + 1 >= argc) {
                std::fprintf(stderr, "Missing value for %s\n", name);
                std::exit(EXIT_FAILURE);
            }
            return argv[++i];
        };

        if (arg == "--rows") {
            opt.M = std::stoi(need_value("--rows"));
        } else if (arg == "--cols") {
            opt.N = std::stoi(need_value("--cols"));
        } else if (arg == "--iters") {
            opt.iters = std::stoi(need_value("--iters"));
        } else if (arg == "--warmup") {
            opt.warmup = std::stoi(need_value("--warmup"));
        } else if (arg == "--tol") {
            opt.tol = std::stod(need_value("--tol"));
        } else if (arg == "--seed") {
            opt.seed = std::stoull(need_value("--seed"), nullptr, 0);
        } else if (arg == "--quiet") {
            opt.verbose = false;
        } else if (arg == "--help" || arg == "-h") {
            print_usage(argv[0]);
            std::exit(0);
        } else {
            std::fprintf(stderr, "Unknown option: %s\n", arg.c_str());
            print_usage(argv[0]);
            return false;
        }
    }
    if (opt.M <= 0 || opt.N <= 0) {
        std::fprintf(stderr, "M and N must be positive.\n");
        return false;
    }
    if (opt.iters <= 0 || opt.warmup < 0) {
        std::fprintf(stderr, "iters must be positive, warmup non-negative.\n");
        return false;
    }
    return true;
}
