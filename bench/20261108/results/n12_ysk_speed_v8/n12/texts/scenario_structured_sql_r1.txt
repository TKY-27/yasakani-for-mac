BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    finished_at TIMESTAMPTZ,
    status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted INT NOT NULL DEFAULT 0 CHECK (mtp_accepted >= 0),
    mtp_proposed INT NOT NULL DEFAULT 0 CHECK (mtp_proposed >= 0),
    UNIQUE (run_id, request_index)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_runs_started ON runs(started_at);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted INT,
    p_mtp_proposed INT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, request_index, input_tokens, output_tokens, latency_ms, mtp_accepted, mtp_proposed)
    VALUES (p_run_id, p_request_index, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted, p_mtp_proposed)
    ON CONFLICT (run_id, request_index) DO UPDATE
    SET input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_proposed = EXCLUDED.mtp_proposed;
END;
$$ LANGUAGE plpgsql;

CREATE VIEW v_run_summary AS
SELECT 
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    r.started_at,
    r.status,
    COUNT(m.metric_id) AS total_requests,
    SUM(m.input_tokens) AS total_input_tokens,
    SUM(m.output_tokens) AS total_output_tokens,
    AVG(m.latency_ms) AS avg_latency_ms,
    SUM(m.mtp_accepted) AS total_mtp_accepted,
    SUM(m.mtp_proposed) AS total_mtp_proposed
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics m ON r.run_id = m.run_id
GROUP BY r.run_id, m.name, f.kind, r.started_at, r.status;

CREATE VIEW v_model_performance AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS run_count,
    AVG(m.latency_ms) AS avg_latency_ms,
    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY m.latency_ms) AS median_latency_ms,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY m.latency_ms) AS p95_latency_ms,
    SUM(m.output_tokens) * 1000.0 / NULLIF(SUM(m.latency_ms), 0) AS throughput_tokens_per_sec,
    CASE WHEN SUM(m.mtp_proposed) > 0 THEN SUM(m.mtp_accepted)::DOUBLE PRECISION / SUM(m.mtp_proposed) ELSE 0 END AS mtp_acceptance_rate
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics m ON r.run_id = m.run_id
GROUP BY m.name, f.kind;
