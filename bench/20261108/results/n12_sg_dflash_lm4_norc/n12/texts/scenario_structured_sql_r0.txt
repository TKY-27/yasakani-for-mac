BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'ultra')),
    token_count INT NOT NULL CHECK (token_count > 0),
    description TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP WITH TIME ZONE,
    status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    request_index INT NOT NULL CHECK (request_index >= 0),
    input_tokens INT NOT NULL CHECK (input_tokens > 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms DOUBLE PRECISION NOT NULL CHECK (latency_ms >= 0),
    mtp_accepted INT NOT NULL DEFAULT 0 CHECK (mtp_accepted >= 0),
    mtp_total INT NOT NULL DEFAULT 0 CHECK (mtp_total >= 0),
    recorded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (run_id, request_index)
);

CREATE INDEX IF NOT EXISTS idx_models_name ON models(name);
CREATE INDEX IF NOT EXISTS idx_fixtures_kind ON fixtures(kind);
CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_status ON runs(status);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX IF NOT EXISTS idx_metrics_mtp ON request_metrics(mtp_accepted, mtp_total);

CREATE OR REPLACE VIEW v_run_summary AS
SELECT
    r.run_id,
    m.name AS model_name,
    f.kind AS fixture_kind,
    f.token_count AS fixture_tokens,
    r.status,
    r.started_at,
    r.finished_at,
    COUNT(mr.metric_id) AS total_requests,
    COALESCE(SUM(mr.input_tokens), 0) AS total_input_tokens,
    COALESCE(SUM(mr.output_tokens), 0) AS total_output_tokens,
    COALESCE(AVG(mr.latency_ms), 0) AS avg_latency_ms,
    COALESCE(SUM(mr.mtp_accepted), 0) AS total_mtp_accepted,
    COALESCE(SUM(mr.mtp_total), 0) AS total_mtp_total
FROM runs r
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
LEFT JOIN request_metrics mr ON r.run_id = mr.run_id
GROUP BY r.run_id, m.name, f.kind, f.token_count, r.status, r.started_at, r.finished_at;

CREATE OR REPLACE VIEW v_model_performance AS
SELECT
    m.model_id,
    m.name AS model_name,
    m.context_length,
    COUNT(DISTINCT r.run_id) AS total_runs,
    COUNT(DISTINCT CASE WHEN r.status = 'completed' THEN r.run_id END) AS completed_runs,
    COALESCE(AVG(rs.avg_latency_ms), 0) AS overall_avg_latency_ms,
    COALESCE(SUM(rs.total_output_tokens), 0) AS total_output_tokens,
    CASE 
        WHEN COALESCE(SUM(rs.total_mtp_total), 0) = 0 THEN 0
        ELSE ROUND(100.0 * SUM(rs.total_mtp_accepted) / SUM(rs.total_mtp_total), 2)
    END AS mtp_acceptance_rate_pct
FROM models m
LEFT JOIN runs r ON m.model_id = r.model_id
LEFT JOIN v_run_summary rs ON r.run_id = rs.run_id
GROUP BY m.model_id, m.name, m.context_length;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_request_index INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms DOUBLE PRECISION,
    p_mtp_accepted INT DEFAULT 0,
    p_mtp_total INT DEFAULT 0
) RETURNS INT AS $$
DECLARE
    v_metric_id INT;
BEGIN
    INSERT INTO request_metrics (
        run_id,
        request_index,
        input_tokens,
        output_tokens,
        latency_ms,
        mtp_accepted,
        mtp_total
    )
    VALUES (
        p_run_id,
        p_request_index,
        p_input_tokens,
        p_output_tokens,
        p_latency_ms,
        p_mtp_accepted,
        p_mtp_total
    )
    ON CONFLICT (run_id, request_index)
    DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted,
        mtp_total = EXCLUDED.mtp_total,
        recorded_at = CURRENT_TIMESTAMP
    RETURNING metric_id INTO v_metric_id;
    
    RETURN v_metric_id;
END;
$$ LANGUAGE plpgsql;
