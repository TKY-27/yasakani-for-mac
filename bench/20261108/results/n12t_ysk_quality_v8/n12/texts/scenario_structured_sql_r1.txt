CREATE VIEW v_run_summary AS
SELECT r.run_id, m.name AS model_name, m.supports_mtp, f.name AS fixture_name, f.kind AS fixture_kind,
r.batch_size, r.max_output_tokens, r.temperature, r.started_at, r.ended_at,
COUNT(q.metric_id) AS request_count,
SUM(q.completion_tokens) AS completion_tokens,
SUM(q.prompt_tokens) AS prompt_tokens,
AVG(q.latency_ms) AS avg_latency_ms,
PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY q.latency_ms) AS p95_latency_ms,
AVG(q.output_tokens_per_s) AS avg_output_tps,
AVG(q.throughput_tps) AS avg_throughput_tps,

SUM(q.mtp_tokens_accepted) AS mtp_accepted,
CASE WHEN SUM(q.mtp_tokens_proposed) > 0 THEN SUM(q.mtp_tokens_accepted)::numeric / SUM(q.mtp_tokens_proposed) END AS mtp_acceptance_rate,
AVG(q.retention_score) FILTER (WHERE q.retention_score IS NOT NULL) AS avg_retention_score
FROM runs r JOIN models m... JOIN fixtures f... LEFT JOIN request_metrics q ON q.run_id = r.run_id
GROUP BY r.run_id, ...;
