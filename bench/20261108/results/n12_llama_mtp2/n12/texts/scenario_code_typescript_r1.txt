```typescript
/**
 * Core domain types for the metrics service.
 * All types are strict; no `any` is used.
 */

/** A single validated request-metric record. */
export interface MetricRecord {
  /** Unique identifier for the run (e.g. a UUID or run name). */
  runId: string;
  /** Model identifier (e.g. "gpt-4", "llama-7b"). */
  model: string;
  /** Fixture / dataset identifier (e.g. "mmlu", "human-eval"). */
  fixture: string;
  /** Latency in milliseconds. Must be finite and >= 0. */
  latencyMs: number;
  /** Token count. Must be a non-negative integer. */
  tokens: number;
  /** Unix epoch timestamp in milliseconds. */
  timestampMs: number;
  /** Optional free-form label for the request. */
  label?: string;
}

/** A single line of the NDJSON input (before validation). */
export interface RawLine {
  /** The raw string content of the line (trimmed). */
  raw: string;
  /** 1-based line number in the stream. */
  line: number;
}

/** Parsed but not yet validated JSON object from a single line. */
export type ParsedLine = Record<string, unknown>;

/** Aggregated statistics for a (model, fixture) group. */
export interface GroupStats {
  model: string;
  fixture: string;
  count: number;
  mean: number;
  median: number;
  /** Sample standard deviation (n-1 denominator). 0 when count < 2. */
  sampleStdDev: number;
  min: number;
  max: number;
  sum: number;
}

/** Per-run summary. */
export interface RunSummary {
  runId: string;
  totalRecords: number;
  groups: GroupStats[];
  firstTimestampMs: number;
  lastTimestampMs: number;
}

/** Overall service summary across all runs. */
export interface ServiceSummary {
  totalRuns: number;
  totalRecords: number;
  runs: RunSummary[];
}

/** Health check response. */
export interface HealthResponse {
  status: "ok" | "degraded";
  uptimeMs: number;
  totalRecords: number;
  totalRuns: number;
  memory: {
    rss: number;
    heapUsed: number;
    heapTotal: number;
  };
}

/** Error returned for malformed input. */
export interface MalformedLineError {
  line: number;
  raw: string;
  reason: string;
}

/** Result of processing a batch of lines. */
export interface ProcessResult {
  accepted: number;
  rejected: MalformedLineError[];
}

/** Configuration for the in-memory store. */
export interface StoreConfig {
  /** Maximum number of records to retain. Oldest are evicted. */
  maxRecords: number;
  /** Maximum number of distinct runs to retain. Oldest runs are evicted. */
  maxRuns: number;
}
```

```typescript
/**
 * Custom error types for the metrics service.

export class MalformedInputError extends Error {
  public readonly line: number;
  public readonly raw: string;
  public readonly reason: string;

  constructor(line: number, raw: string, reason: string) {
    super(`Line ${line}: ${reason}`);
    this.name = "MalformedInputError";
    this.line = line;
    this.raw = raw;
    this.reason = reason;
  }
}

export class StoreFullError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "StoreFullError";
  }
}

export class RunNotFoundError extends Error {
  public readonly runId: string;

  constructor(runId: string) {
    super(`Run "${runId}" not found`);
    this.name = "RunNotFoundError";
    this.runId = runId;
  }
}
```

## `src/parser.ts`

```typescript
/**
 * Incremental NDJSON parser with strict validation.
 * No third-party schema libraries — all validation is hand-rolled.
 */

import type { MetricRecord, ParsedLine, ProcessResult, RawLine } from "./types.js";
import { MalformedInputError } from "./errors.js";

/** Maximum length of a single line (bytes). Prevents memory blow-up. */
const MAX_LINE_LENGTH = 65_536;

/** Maximum length for string fields. */
const MAX_STRING_LENGTH = 512;

/**
 * Incremental line splitter.
 * Feeds it chunks of text; it yields complete lines.
 * Handles partial lines that span chunk boundaries.
 */
export class LineSplitter {
  private buffer: string = "";
  private lineCounter: number = 0;

  /**
   * Feed a chunk of text. Returns an array of complete lines.
   * The last line may be incomplete and is retained in the buffer.
   */
  feed(chunk: string): RawLine[] {
    this.buffer += chunk;
    const lines: RawLine[] = [];
    let start = 0;

    while (true) {
      const newlineIdx = this.buffer.indexOf("\n", start);
      if (newlineIdx === -1) {
        break;
      }
      const lineContent = this.buffer.slice(start, newlineIdx);
      start = newlineIdx + 1;
      this.lineCounter += 1;
      const trimmed = lineContent.replace(/\r$/, ""); // strip trailing \r for CRLF
      if (trimmed.length > 0) {
        lines.push({ raw: trimmed, line: this.lineCounter });
      }
    }

    // Keep the remainder (incomplete line) in the buffer.
    this.buffer = this.buffer.slice(start);

    // Guard against unbounded buffer growth.
    if (this.buffer.length > MAX_LINE_LENGTH) {
      // Discard the oversized buffer; the next feed will start fresh.
      this.buffer = "";
    }

    return lines;
  }

  /**
   * Flush any remaining buffered content as a final line.
   * Call this when the stream ends.
   */
  flush(): RawLine[] {
    if (this.buffer.length === 0) {
      return [];
    }
    this.lineCounter += 1;
    const trimmed = this.buffer.replace(/\r$/, "");
    this.buffer = "";
    if (trimmed.length > 0) {
      return [{ raw: trimmed, line: this.lineCounter }];
    }
    return [];
  }

  /** Reset the splitter (e.g. for a new stream). */
  reset(): void {
    this.buffer = "";
    this.lineCounter = 0;
  }
}

/**
 * Validate a parsed JSON object against the MetricRecord schema.
 * Returns the validated record or throws MalformedInputError.
 */
export function validateRecord(parsed: ParsedLine, line: number, raw: string): MetricRecord {
  // --- runId ---
  const runId = parsed["runId"];
  if (typeof runId !== "string" || runId.length === 0 || runId.length > MAX_STRING_LENGTH) {
    throw new MalformedInputError(line, raw, "runId must be a non-empty string (max 512 chars)");
  }

  // --- model ---
  const model = parsed["model"];
  if (typeof model !== "string" || model.length === 0 || model.length > MAX_STRING_LENGTH) {
    throw new MalformedInputError(line, raw, "model must be a non-empty string (max 512 chars)");
  }

  // --- fixture ---
  const fixture = parsed["fixture"];
  if (typeof fixture !== "string" || fixture.length === 0 || fixture.length > MAX_STRING_LENGTH) {
    throw new MalformedInputError(line, raw, "fixture must be a non-empty string (max 512 chars)");
  }

  // --- latencyMs ---
  const latencyMs = parsed["latencyMs"];
  if (typeof latencyMs !== "number" || !Number.isFinite(latencyMs) || latencyMs < 0) {
    throw new MalformedInputError(line, raw, "latencyMs must be a finite number >= 0");
  }

  // --- tokens ---
  const tokens = parsed["tokens"];
  if (typeof tokens !== "number" || !Number.isInteger(tokens) || tokens < 0) {
    throw new MalformedInputError(line, raw, "tokens must be a non-negative integer");
  }

  // --- timestampMs ---
  const timestampMs = parsed["timestampMs"];
  if (typeof timestampMs !== "number" || !Number.isFinite(timestampMs) || timestampMs < 0) {
    throw new MalformedInputError(line, raw, "timestampMs must be a finite number >= 0");
  }

  // --- label (optional) ---
  let label: string | undefined;
  if (parsed["label"] !== undefined) {
    if (typeof parsed["label"] !== "string" || parsed["label"].length > MAX_STRING_LENGTH) {
      throw new MalformedInputError(line, raw, "label must be a string (max 512 chars) if present");
    }
    label = parsed["label"];
  }

  return {
    runId,
    model,
    fixture,
    latencyMs,
    tokens,
    timestampMs,
    ...(label !== undefined ? { label } : {}),
  };
}

/**
 * Parse and validate a single NDJSON line.
 * Returns the validated record, or throws MalformedInputError.
 */
export function parseLine(raw: string, line: number): MetricRecord {
  let parsed: ParsedLine;
  try {
    const obj: unknown = JSON.parse(raw);
    if (typeof obj !== "object" || obj === null || Array.isArray(obj)) {
      throw new MalformedInputError(line, raw, "top-level value must be a JSON object");
    }
    parsed = obj as ParsedLine;
  } catch (err) {
    if (err instanceof MalformedInputError) {
      throw err;
    }
    throw new MalformedInputError(line, raw, `invalid JSON: ${(err as Error).message}`);
  }

  return validateRecord(parsed, line, raw);
}

/**
 * Process a batch of raw lines, returning accepted records and rejected errors.
 * This is the main entry point for incremental ingestion.
 */
export function processLines(lines: RawLine[]): { records: MetricRecord[]; result: ProcessResult } {
  const records: MetricRecord[] = [];
  const rejected: ProcessResult["rejected"] = [];

  for (const line of lines) {
    try {
      const record = parseLine(line.raw, line.line);
      records.push(record);
    } catch (err) {
      if (err instanceof MalformedInputError) {
        rejected.push({ line: err.line, raw: err.raw, reason: err.reason });
      } else {
        rejected.push({ line: line.line, raw: line.raw, reason: "unexpected internal error" });
      }
    }
  }

  return {
    records,
    result: { accepted: records.length, rejected },
  };
}
```

## `src/stats.ts`

```typescript
/**
 * Statistical computations: mean, median, sample standard deviation.
 * All functions are pure and operate on number arrays.
 */

/**
 * Compute the arithmetic mean. Returns 0 for an empty array.
 */
export function mean(values: readonly number[]): number {
  if (values.length === 0) return 0;
  let sum = 0;
  for (const v of values) sum += v;
  return sum / values.length;
}

/**
 * Compute the median.
 * For even-length arrays, returns the average of the two middle values.
 * Returns 0 for an empty array.
 * Does not mutate the input.
 */
export function median(values: readonly number[]): number {
  if (values.length === 0) return 0;
  const sorted = [...values].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  if (sorted.length % 2 === 1) {
    return sorted[mid]!;
  }
  return (sorted[mid - 1]! + sorted[mid]!) / 2;
}

/**
 * Compute the sample standard deviation (n-1 denominator, Bessel's correction).
 * Returns 0 when n < 2.
 */
export function sampleStdDev(values: readonly number[]): number {
  const n = values.length;
  if (n < 2) return 0;
  const m = mean(values);
  let sumSqDiff = 0;
  for (const v of values) {
    const diff = v - m;
    sumSqDiff += diff * diff;
  }
  return Math.sqrt(sumSqDiff / (n - 1));
}

/**
 * Compute min and max. Returns { min: 0, max: 0 } for an empty array.
 */
export function minMax(values: readonly number[]): { min: number; max: number } {
  if (values.length === 0) return { min: 0, max: 0 };
  let min = values[0]!;
  let max = values[0]!;
  for (let i = 1; i < values.length; i++) {
    const v = values[i]!;
    if (v < min) min = v;
    if (v > max) max = v;
  }
  return { min, max };
}

/**
 * Compute all statistics for a group of latency values.
 */
export function computeGroupStats(
  model: string,
  fixture: string,
  latencies: readonly number[],
): {
  model: string;
  fixture: string;
  count: number;
  mean: number;
  median: number;
  sampleStdDev: number;
  min: number;
  max: number;
  sum: number;
} {
  const { min, max } = minMax(latencies);
  const sum = latencies.reduce((acc, v) => acc + v, 0);
  return {
    model,
    fixture,
    count: latencies.length,
    mean: mean(latencies),
    median: median(latencies),
    sampleStdDev: sampleStdDev(latencies),
    min,
    max,
    sum,
  };
}
```
