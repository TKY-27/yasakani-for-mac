BEGIN;

CREATE TABLE IF NOT EXISTS models (
    model_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    architecture TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind TEXT NOT NULL CHECK (kind IN ('synthetic', 'real_world', 'adversarial', 'multilingual')),
    description TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS runs (
    run_id SERIAL PRIMARY KEY,
    model_id INTEGER NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INTEGER NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE,
    status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed')),
    UNIQUE (model_id, fixture_id, started_at)
);

CREATE TABLE IF NOT EXISTS request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INTEGER NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    timestamp TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    tokens_per_second REAL CHECK (tokens_per_second >= 0),
    latency_ms REAL CHECK (latency_ms >= 0),
    context_length INTEGER CHECK (context_length > 0),
    retention_score REAL CHECK (retention_score BETWEEN 0 AND 1),
    mtp_acceptance_rate REAL CHECK (mtp_acceptance_rate BETWEEN 0 AND 1),
    error_count INTEGER NOT NULL DEFAULT 0 CHECK (error_count >= 0)
);

CREATE INDEX IF NOT EXISTS idx_runs_model ON runs(model_id);
CREATE INDEX IF NOT EXISTS idx_runs_fixture ON runs(fixture_id);
CREATE INDEX IF NOT EXISTS idx_runs_started ON runs(started_at);
CREATE INDEX IF NOT EXISTS idx_metrics_run ON request_metrics(run_id);
CREATE INDEX IF NOT EXISTS idx_metrics_timestamp ON request_metrics(timestamp);
CREATE INDEX IF NOT EXISTS idx_metrics_tps ON request_metrics(tokens_per_second);

CREATE OR REPLACE VIEW v_model_performance AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(DISTINCT r.run_id) AS total_runs,
    AVG(rm.tokens_per_second) AS avg_tps,
    STDDEV(rm.tokens_per_second) AS tps_stddev,
    AVG(rm.retention_score) AS avg_retention,
    AVG(rm.mtp_acceptance_rate) AS avg_mtp_acceptance
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE OR REPLACE VIEW v_long_context_analysis AS
SELECT 
    m.name AS model_name,
    CASE 
        WHEN rm.context_length < 4096 THEN 'short'
        WHEN rm.context_length < 32768 THEN 'medium'
        ELSE 'long'
    END AS context_category,
    COUNT(*) AS sample_count,
    AVG(rm.retention_score) AS avg_retention,
    MIN(rm.retention_score) AS min_retention,
    MAX(rm.retention_score) AS max_retention
FROM models m
JOIN runs r ON m.model_id = r.model_id
JOIN request_metrics rm ON r.run_id = rm.run_id
WHERE r.status = 'completed'
GROUP BY m.name, context_category;

CREATE OR REPLACE FUNCTION upsert_request_metrics(
    p_run_id INTEGER,
    p_timestamp TIMESTAMP WITH TIME ZONE,
    p_tps REAL,
    p_latency REAL,
    p_context INTEGER,
    p_retention REAL,
    p_mtp REAL,
    p_errors INTEGER
) RETURNS INTEGER AS $$
DECLARE
    v_metric_id INTEGER;
BEGIN
    INSERT INTO request_metrics (
        run_id, timestamp, tokens_per_second, latency_ms, 
        context_length, retention_score, mtp_acceptance_rate, error_count
    ) VALUES (
        p_run_id, p_timestamp, p_tps, p_latency, 
        p_context, p_retention, p_mtp, p_errors
    )
    ON CONFLICT (run_id, timestamp) DO UPDATE SET
        tokens_per_second = EXCLUDED.tokens_per_second,
        latency_ms = EXCLUDED.latency_ms,
        context_length = EXCLUDED.context_length,
        retention_score = EXCLUDED.retention_score,
        mtp_acceptance_rate = EXCLUDED.mtp_acceptance_rate,
        error_count = EXCLUDED.error_count
    RETURNING metric_id INTO v_metric_id;
    
    RETURN v_metric_id;
END;
$$ LANGUAGE plpgsql;
