**src/djr/redaction.py**

```python
class Redactor:
    def __init__(self, values: Iterable[str]):
        # only non-empty values, sort by length descending to redact longer substrings first
        self._values = sorted({v for v in values if v}, key=len, reverse=True)
    def redact(self, text: str) -> str:
        for v in self._values:
            if v in text:
                text = text.replace(v, "[REDACTED]")
        return text
```

This is deterministic: sort by length descending, and for determinism tiebreak by string (use key=lambda v: (-len(v), v)).

Atomic write helper:

```python
def atomic_write_json(path: Path, data: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=str(path.parent), prefix=path.name + ".", suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2, sort_keys=True)
            fh.write("\n")
            fh.flush()
            os.fsync(fh.fileno())
        os.replace(tmp, path)
    except:
        try: os.unlink(tmp)
        except OSError: pass
        raise
```
