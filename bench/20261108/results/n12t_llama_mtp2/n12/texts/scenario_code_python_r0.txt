    def __init__(self, manifest, state_path, event_stream, redactor, loop):
        self.semaphore = asyncio.Semaphore(manifest.max_gpu_jobs) # default 1
        self.cancel_event = asyncio.Event()
        self._loop = loop
        self._signal_handler_installed = False

    async def run(self) -> RunnerResult:
        install signal handler
        state = self.state.load()
        self.events.emit(runner_started, manifest=..., resume=...)
        for job in manifest.jobs:
            if state.status_for(job.id) == completed: continue
            if self.cancel_event.is_set(): break
            state.set_status(job.id, running, save=True)
            self.events.emit(job_started, job=job)
            try:
                async with self.semaphore:
                    await self._execute_job(job)
                state.set_status(job.id, completed, ...)
                self.events.emit(job_completed)
            except asyncio.CancelledError:
                state.set_status(job.id, canceled)
                self.events.emit(job_canceled)
                # if due to SIGINT break; else re-raise?

                state.set_status(job.id, failed, error=str(exc))
                self.events.emit(job_failed)
                if manifest.stop_on_error: break
        self.events.emit(runner_completed, summary=...)
        return result
