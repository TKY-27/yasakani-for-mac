CREATE TABLE models (
  model_id smallint PRIMARY KEY CHECK (model_id > 0),
  model_name text NOT NULL UNIQUE,
  family text NOT NULL,
  param_count_b numeric(10,2) NOT NULL CHECK (param_count_b > 0),
  context_limit int NOT NULL CHECK (context_limit > 0),
  quantization text NOT NULL DEFAULT 'bf16',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE fixtures (
  fixture_id smallint PRIMARY KEY CHECK(fixture_id >0),
  fixture_kind text NOT NULL CHECK (fixture_kind IN ('short_chat','long_doc','rag_multihop','streaming_tool')),
  description text NOT NULL,
  min_context_tokens int NOT NULL CHECK(min_context_tokens >0),
  max_context_tokens int NOT NULL CHECK(max_context_tokens >= min_context_tokens),
  target_output_tokens int NOT NULL CHECK(target_output_tokens >0),
  difficulty int NOT NULL CHECK(difficulty BETWEEN 1 AND 5)
);

CREATE TABLE runs (
  run_id int PRIMARY KEY CHECK(run_id >0),
  model_id smallint NOT NULL REFERENCES models(model_id),
  fixture_id smallint NOT NULL REFERENCES fixtures(fixture_id),
  batch_size int NOT NULL CHECK(batch_size >0),
  max_tokens_per_seq int NOT NULL CHECK(max_tokens_per_seq >0),
  speculative_tokens int NOT NULL DEFAULT 0 CHECK(speculative_tokens >=0),
  gpu text NOT NULL DEFAULT 'A100-80GB',
  started_at timestamptz NOT NULL,
  finished_at timestamptz,
  status text NOT NULL CHECK (status IN ('completed','failed','aborted')),
  notes text,
  CHECK (finished_at IS NULL OR finished_at >= started_at)
);
